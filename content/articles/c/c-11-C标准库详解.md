## 概述

C标准库是C语言的核心组成部分，提供了输入输出、内存管理、字符串处理、数学运算等基础功能。

**常用标准库头文件**:

| 头文件 | 功能 |
|--------|------|
| `<stdio.h>` | 输入输出 |
| `<stdlib.h>` | 通用工具：内存、转换、随机数 |
| `<string.h>` | 字符串操作 |
| `<math.h>` | 数学函数 |
| `<time.h>` | 时间日期 |
| `<ctype.h>` | 字符分类 |
| `<stdint.h>` | 定宽整数类型 |
| `<stdbool.h>` | 布尔类型 (C99) |
| `<stddef.h>` | 常用定义 (NULL, size_t) |
| `<limits.h>` | 类型限制 |
| `<errno.h>` | 错误处理 |
| `<assert.h>` | 断言 |
| `<signal.h>` | 信号处理 |
| `<setjmp.h>` | 非局部跳转 |

## stdio.h - 输入输出

### 文件操作

```c
// 打开/关闭文件
FILE *fopen(const char *filename, const char *mode);
int fclose(FILE *stream);

// 模式: "r"读, "w"写, "a"追加, "rb"二进制读, "w+"读写

// 读写函数
int fgetc(FILE *stream);
int fputc(int c, FILE *stream);
char *fgets(char *s, int n, FILE *stream);
int fputs(const char *s, FILE *stream);

size_t fread(void *ptr, size_t size, size_t nmemb, FILE *stream);
size_t fwrite(const void *ptr, size_t size, size_t nmemb, FILE *stream);

// 定位
int fseek(FILE *stream, long offset, int whence);  // SEEK_SET, SEEK_CUR, SEEK_END
long ftell(FILE *stream);
void rewind(FILE *stream);

// 错误检查
int feof(FILE *stream);
int ferror(FILE *stream);
void clearerr(FILE *stream);
```

### 格式化输入输出

```c
// 输出
int printf(const char *format, ...);
int fprintf(FILE *stream, const char *format, ...);
int sprintf(char *str, const char *format, ...);
int snprintf(char *str, size_t size, const char *format, ...);  // 推荐

// 输入
int scanf(const char *format, ...);
int fscanf(FILE *stream, const char *format, ...);
int sscanf(const char *str, const char *format, ...);
```

### 格式说明符

```
常用格式说明符:
%d, %i  - int
%u      - unsigned int
%ld     - long
%lld    - long long
%f      - float/double (printf), float (scanf)
%lf     - double (scanf)
%e, %E  - 科学计数法
%x, %X  - 十六进制
%o      - 八进制
%c      - 字符
%s      - 字符串
%p      - 指针
%zu     - size_t
%td     - ptrdiff_t
%%      - 百分号

修饰符:
%-10d   - 左对齐，宽度10
%+d     - 显示正负号
%010d   - 前导零填充
%.2f    - 精度2位小数
%*d     - 动态宽度
```

## stdlib.h - 通用工具

### 内存管理

```c
void *malloc(size_t size);
void *calloc(size_t nmemb, size_t size);  // 初始化为0
void *realloc(void *ptr, size_t size);
void free(void *ptr);

// 对齐分配 (C11)
void *aligned_alloc(size_t alignment, size_t size);
```

### 类型转换

```c
int atoi(const char *nptr);
long atol(const char *nptr);
double atof(const char *nptr);

// 更安全的版本
long strtol(const char *nptr, char **endptr, int base);
unsigned long strtoul(const char *nptr, char **endptr, int base);
double strtod(const char *nptr, char **endptr);
long long strtoll(const char *nptr, char **endptr, int base);
```

**示例**:
```c
const char *str = "42abc";
char *end;
long val = strtol(str, &end, 10);
printf("值: %ld, 剩余: '%s'\n", val, end);  // 值: 42, 剩余: 'abc'
```

### 随机数

```c
int rand(void);           // 返回 0 到 RAND_MAX
void srand(unsigned int seed);

// 典型用法
srand((unsigned)time(NULL));
int r = rand() % 100;     // 0-99
```

### 程序控制

```c
void exit(int status);
void abort(void);
int atexit(void (*func)(void));  // 注册退出处理函数
void _Exit(int status);          // 立即退出，不调用 atexit

int system(const char *command);
char *getenv(const char *name);
```

### 搜索与排序

```c
void qsort(void *base, size_t nmemb, size_t size,
           int (*compar)(const void *, const void *));

void *bsearch(const void *key, const void *base,
              size_t nmemb, size_t size,
              int (*compar)(const void *, const void *));
```

**示例**:
```c
int compare_int(const void *a, const void *b) {
    return (*(int*)a - *(int*)b);
}

int arr[] = {5, 2, 8, 1, 9};
qsort(arr, 5, sizeof(int), compare_int);

int key = 8;
int *found = bsearch(&key, arr, 5, sizeof(int), compare_int);
```

## time.h - 时间日期

```c
// 类型
typedef long time_t;
struct tm {
    int tm_sec;    // 0-60
    int tm_min;    // 0-59
    int tm_hour;   // 0-23
    int tm_mday;   // 1-31
    int tm_mon;    // 0-11
    int tm_year;   // 年份 - 1900
    int tm_wday;   // 0-6 (周日=0)
    int tm_yday;   // 0-365
    int tm_isdst;  // 夏令时
};

// 获取时间
time_t time(time_t *tloc);
clock_t clock(void);  // CPU时间

// 转换
struct tm *localtime(const time_t *timep);
struct tm *gmtime(const time_t *timep);
time_t mktime(struct tm *tm);

// 格式化
char *ctime(const time_t *timep);
char *asctime(const struct tm *tm);
size_t strftime(char *s, size_t max, const char *format, const struct tm *tm);

// 差值
double difftime(time_t time1, time_t time0);
```

**示例**:
```c
time_t now = time(NULL);
struct tm *local = localtime(&now);

char buf[64];
strftime(buf, sizeof(buf), "%Y-%m-%d %H:%M:%S", local);
printf("当前时间: %s\n", buf);
```

## errno.h - 错误处理

```c
extern int errno;  // 全局错误码
```

**示例**:

## signal.h - 信号处理

```c
// 信号类型
SIGINT   // Ctrl+C
SIGTERM  // 终止请求
SIGSEGV  // 段错误
SIGFPE   // 浮点异常
SIGABRT  // abort()

// 信号处理
typedef void (*sighandler_t)(int);
sighandler_t signal(int signum, sighandler_t handler);

// 特殊处理函数
SIG_DFL  // 默认处理
SIG_IGN  // 忽略信号

// 发送信号
int raise(int sig);
```

**示例**:
```c
#include <signal.h>
#include <stdio.h>

volatile sig_atomic_t flag = 0;

void handler(int sig) {
    flag = 1;
}

int main(void) {
    signal(SIGINT, handler);
    
    while (!flag) {
        printf("运行中...\n");
        sleep(1);
    }
    
    printf("收到信号，退出\n");
    return 0;
}
```

## setjmp.h - 非局部跳转

```c
#include <setjmp.h>

jmp_buf env;

void deep_function(void) {
    // 出错时跳回
    longjmp(env, 1);
}

int main(void) {
    if (setjmp(env) == 0) {
        // 正常路径
        deep_function();
    } else {
        // 错误处理路径
        printf("从错误中恢复\n");
    }
    return 0;
}
```

**注意**: `longjmp` 不会执行 C++ 析构函数或清理本地资源。

## 最佳实践

### 1. 缓冲区安全

```c
// 不安全
char buf[10];
strcpy(buf, user_input);  // 可能溢出!

// 安全
char buf[10];
strncpy(buf, user_input, sizeof(buf) - 1);
buf[sizeof(buf) - 1] = '\0';

// 更好 (如果可用)
snprintf(buf, sizeof(buf), "%s", user_input);
```

### 2. 错误检查

```c
// 始终检查返回值
FILE *f = fopen("file.txt", "r");
if (f == NULL) {
    perror("fopen");
    return -1;
}

void *p = malloc(size);
if (p == NULL) {
    fprintf(stderr, "内存分配失败\n");
    return -1;
}
```

### 3. 资源清理

```c
FILE *f = NULL;
char *buf = NULL;

f = fopen("file.txt", "r");
if (!f) goto cleanup;

buf = malloc(1024);
if (!buf) goto cleanup;

// 使用资源...

cleanup:
    free(buf);
    if (f) fclose(f);
```
