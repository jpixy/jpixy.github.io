## 条件语句

### if-else

```c
// 基本形式
if (condition) {
    // 条件为真时执行
}

// if-else
if (condition) {
    // 真
} else {
    // 假
}

// else-if 链
if (score >= 90) {
    grade = 'A';
} else if (score >= 80) {
    grade = 'B';
} else if (score >= 70) {
    grade = 'C';
} else {
    grade = 'F';
}
```

### 悬空 else 问题

```c
// 歧义代码
if (a > 0)
    if (b > 0)
        printf("a>0 && b>0\n");
else
    printf("???\n");  // 属于哪个 if?

// 实际: else 匹配最近的 if
// 建议: 始终使用大括号
if (a > 0) {
    if (b > 0) {
        printf("a>0 && b>0\n");
    }
} else {
    printf("a<=0\n");
}
```

### 条件表达式陷阱

```c
// 陷阱1: 赋值vs比较
if (x = 5) { ... }   // 赋值! 总是为真
if (x == 5) { ... }  // 比较

// 防御性编程
if (5 == x) { ... }  // 常量在左，误写成=会报错

// 陷阱2: 浮点比较
if (f == 0.1) { ... }  // 危险! 浮点精度问题
if (fabs(f - 0.1) < 1e-9) { ... }  // 正确
```

## switch 语句

### 基本语法

```c
switch (expression) {
    case constant1:
        // 语句
        break;
    case constant2:
    case constant3:  // 多值匹配
        // 语句
        break;
    default:
        // 默认情况
        break;
}
```

### 穿透行为

```c
int day = 2;
switch (day) {
    case 1:
        printf("Monday\n");
        // 没有 break，会穿透!
    case 2:
        printf("Tuesday\n");
        break;
    default:
        printf("Other\n");
}
// day=1 时输出: Monday Tuesday (穿透)
```

**利用穿透**:
```c
// 判断是否为元音
switch (c) {
    case 'a':
    case 'e':
    case 'i':
    case 'o':
    case 'u':
        printf("Vowel\n");
        break;
    default:
        printf("Consonant\n");
}
```

### switch 限制

```c
// 1. expression 必须是整数类型
switch (1.5) { ... }   // 错误!
switch ("abc") { ... } // 错误!

// 2. case 必须是编译时常量
int x = 10;
switch (n) {
    case x:     // 错误! x 不是常量
    case 10:    // 正确
    case 1+2:   // 正确: 编译时计算
}

// 3. 不能有重复的 case 值
switch (n) {
    case 1:
    case 1:  // 错误!
}
```

## 循环语句

### for 循环

```c
// 基本形式
for (初始化; 条件; 更新) {
    // 循环体
}

// 示例
for (int i = 0; i < n; i++) {
    printf("%d\n", i);
}

// 多变量
for (int i = 0, j = n - 1; i < j; i++, j--) {
    swap(&arr[i], &arr[j]);
}

// 无限循环
for (;;) {
    // ...
    if (exit_condition) break;
}
```

### while 循环

```c
// 前测试循环
while (condition) {
    // 循环体
}

// 示例: 读取输入直到EOF
int c;
while ((c = getchar()) != EOF) {
    putchar(c);
}
```

### do-while 循环

```c
// 后测试循环 (至少执行一次)
do {
    // 循环体
} while (condition);

// 示例: 输入验证
int n;
do {
    printf("Enter positive number: ");
    scanf("%d", &n);
} while (n <= 0);
```

### 循环对比

| 类型 | 检查时机 | 适用场景 |
|------|----------|----------|
| for | 循环前 | 已知迭代次数 |
| while | 循环前 | 未知次数，可能0次 |
| do-while | 循环后 | 至少执行1次 |

## 跳转语句

### break

```c
// 跳出最内层循环或switch
for (int i = 0; i < n; i++) {
    if (arr[i] == target) {
        found = 1;
        break;  // 跳出 for
    }
}

// 跳出嵌套循环需要用标志或goto
int found = 0;
for (int i = 0; i < m && !found; i++) {
    for (int j = 0; j < n; j++) {
        if (matrix[i][j] == target) {
            found = 1;
            break;  // 只跳出内层
        }
    }
}
```

### continue

```c
// 跳过本次迭代，继续下一次
for (int i = 0; i < n; i++) {
    if (arr[i] < 0) {
        continue;  // 跳过负数
    }
    sum += arr[i];
}
```

### goto

```c
// 无条件跳转 (慎用!)
// 合理用法: 跳出多层嵌套、集中错误处理

// 示例: 错误处理
int func(void) {
    int *p1 = NULL, *p2 = NULL;
    
    p1 = malloc(100);
    if (!p1) goto cleanup;
    
    p2 = malloc(200);
    if (!p2) goto cleanup;
    
    // 正常处理...
    
    free(p2);
    free(p1);
    return 0;
    
cleanup:
    free(p2);  // NULL 可以安全 free
    free(p1);
    return -1;
}

// 跳出嵌套循环
for (int i = 0; i < m; i++) {
    for (int j = 0; j < n; j++) {
        if (matrix[i][j] == target) {
            goto found;
        }
    }
}
found:
    // 找到后的处理
```

### return

```c
// 从函数返回
int max(int a, int b) {
    if (a > b) {
        return a;
    }
    return b;  // else 分支
}

// void 函数
void process(int *data) {
    if (data == NULL) {
        return;  // 提前返回
    }
    // 处理...
}
```

## 循环优化技巧

### 减少循环内计算

```c
// 差: 每次迭代都计算 strlen
for (int i = 0; i < strlen(s); i++) { ... }

// 好: 预先计算
int len = strlen(s);
for (int i = 0; i < len; i++) { ... }
```

### 循环展开

```c
// 原始
for (int i = 0; i < n; i++) {
    sum += arr[i];
}

// 展开 (减少分支预测失败)
int i;
for (i = 0; i + 3 < n; i += 4) {
    sum += arr[i] + arr[i+1] + arr[i+2] + arr[i+3];
}
for (; i < n; i++) {
    sum += arr[i];
}
```

### Duff's Device

```c
// 著名的循环展开技巧 (C语言奇技淫巧)
void send(char *to, char *from, int count) {
    int n = (count + 7) / 8;
    switch (count % 8) {
        case 0: do { *to++ = *from++;
        case 7:      *to++ = *from++;
        case 6:      *to++ = *from++;
        case 5:      *to++ = *from++;
        case 4:      *to++ = *from++;
        case 3:      *to++ = *from++;
        case 2:      *to++ = *from++;
        case 1:      *to++ = *from++;
        } while (--n > 0);
    }
}
```

## 面试题

### 题目1: 输出1-100中的素数

```c
#include <stdio.h>
#include <stdbool.h>

bool is_prime(int n) {
    if (n < 2) return false;
    if (n == 2) return true;
    if (n % 2 == 0) return false;
    for (int i = 3; i * i <= n; i += 2) {
        if (n % i == 0) return false;
    }
    return true;
}

int main(void) {
    for (int i = 1; i <= 100; i++) {
        if (is_prime(i)) {
            printf("%d ", i);
        }
    }
    printf("\n");
    return 0;
}
```

### 题目2: 打印菱形

```c
void print_diamond(int n) {
    // 上半部分
    for (int i = 1; i <= n; i++) {
        for (int j = 0; j < n - i; j++) printf(" ");
        for (int j = 0; j < 2 * i - 1; j++) printf("*");
        printf("\n");
    }
    // 下半部分
    for (int i = n - 1; i >= 1; i--) {
        for (int j = 0; j < n - i; j++) printf(" ");
        for (int j = 0; j < 2 * i - 1; j++) printf("*");
        printf("\n");
    }
}
```

### 题目3: 使用位运算替代循环

判断 n 是否为 2 的幂:
```c
// 循环方法
bool is_power_of_2_loop(int n) {
    if (n <= 0) return false;
    while (n > 1) {
        if (n % 2 != 0) return false;
        n /= 2;
    }
    return true;
}

// 位运算方法 (O(1))
bool is_power_of_2(int n) {
    return n > 0 && (n & (n - 1)) == 0;
}
```
