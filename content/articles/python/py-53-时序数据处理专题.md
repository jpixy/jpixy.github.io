## 二、重采样

### 2.1 降采样

```python
# 分钟数据重采样为小时
minute_data = pd.DataFrame({
    'price': np.random.random(390),
    'volume': np.random.randint(100, 1000, 390)
}, index=pd.date_range('2024-01-01 09:30', periods=390, freq='min'))

# 基本重采样
hourly = minute_data.resample('h').agg({
    'price': 'last',
    'volume': 'sum'
})

# OHLCV重采样
ohlcv = minute_data['price'].resample('h').ohlc()
ohlcv['volume'] = minute_data['volume'].resample('h').sum()

# 自定义聚合
def vwap(df):
    return (df['price'] * df['volume']).sum() / df['volume'].sum()

hourly_vwap = minute_data.resample('h').apply(vwap)
```

```python
# 日数据升采样为分钟

# 前向填充

# 线性插值

minute = daily.resample('min').interpolate(method='time')
```

### 3.3 高级滑动窗口

```python
# 扩展窗口（累计）
df['cummax'] = df['price'].expanding().max()
df['cummin'] = df['price'].expanding().min()
df['cummean'] = df['price'].expanding().mean()

# 自定义滑动函数
def zscore(x):
    return (x[-1] - x.mean()) / x.std()

df['zscore'] = df['price'].rolling(20).apply(zscore, raw=True)

# 使用numba加速
from numba import jit

@jit(nopython=True)
def fast_rolling_zscore(arr, window):
    result = np.empty(len(arr))
    result[:window-1] = np.nan
    for i in range(window-1, len(arr)):
        window_data = arr[i-window+1:i+1]
        mean = window_data.mean()
        std = window_data.std()
        result[i] = (arr[i] - mean) / std
    return result

df['zscore_fast'] = fast_rolling_zscore(df['price'].values, 20)
```

### 5.1 Tick数据处理

```python
# Tick数据结构
ticks = pd.DataFrame({
    'timestamp': pd.to_datetime([
        '2024-01-02 09:30:00.123456',
        '2024-01-02 09:30:00.234567',
        '2024-01-02 09:30:00.345678',
    ]),
    'price': [100.01, 100.02, 100.01],
    'volume': [100, 200, 150],
    'side': ['B', 'S', 'B']  # Buy/Sell
})
ticks = ticks.set_index('timestamp')

# 计算tick级统计
ticks['return'] = ticks['price'].pct_change()
ticks['spread'] = ticks.groupby(ticks.index.floor('s'))['price'].transform(
    lambda x: x.max() - x.min()
)
```

### 5.2 Tick转Bar

```python
def ticks_to_bars(ticks, freq='1min'):
    """Tick数据转K线"""
    bars = ticks.resample(freq).agg({
        'price': ['first', 'max', 'min', 'last'],
        'volume': 'sum'
    })
    bars.columns = ['open', 'high', 'low', 'close', 'volume']
    
    # 计算VWAP
    ticks['value'] = ticks['price'] * ticks['volume']
    bars['vwap'] = ticks['value'].resample(freq).sum() / bars['volume']
    
    return bars

def ticks_to_volume_bars(ticks, volume_threshold):
    """基于成交量的Bar"""
    ticks['cumvol'] = ticks['volume'].cumsum()
    ticks['bar_id'] = ticks['cumvol'] // volume_threshold
    
    bars = ticks.groupby('bar_id').agg({
        'price': ['first', 'max', 'min', 'last'],
        'volume': 'sum'
    })
    bars.columns = ['open', 'high', 'low', 'close', 'volume']
    bars.index = ticks.groupby('bar_id').first().index
    
    return bars
```

### 5.3 订单流分析

```python
def compute_order_flow(ticks):
    """计算订单流指标"""
    # 分类买卖
    ticks['signed_volume'] = np.where(
        ticks['side'] == 'B',
        ticks['volume'],
        -ticks['volume']
    )
    
    # 订单不平衡
    resampled = ticks.resample('1min')
    buy_volume = ticks[ticks['side'] == 'B']['volume'].resample('1min').sum()
    sell_volume = ticks[ticks['side'] == 'S']['volume'].resample('1min').sum()
    
    imbalance = (buy_volume - sell_volume) / (buy_volume + sell_volume)
    
    return imbalance

def compute_trades_stats(ticks):
    """交易统计"""
    resampled = ticks.resample('1min')
    
    stats = pd.DataFrame({
        'num_trades': resampled['price'].count(),
        'avg_trade_size': resampled['volume'].mean(),
        'total_volume': resampled['volume'].sum(),
        'price_range': resampled['price'].max() - resampled['price'].min(),
    })
    
    return stats
```

### 6.2 技术指标

```python
def compute_indicators(df):
    """计算常用技术指标"""
    # 移动平均
    df['sma_20'] = df['close'].rolling(20).mean()
    df['ema_20'] = df['close'].ewm(span=20).mean()
    
    # 布林带
    df['bb_mid'] = df['close'].rolling(20).mean()
    df['bb_std'] = df['close'].rolling(20).std()
    df['bb_upper'] = df['bb_mid'] + 2 * df['bb_std']
    df['bb_lower'] = df['bb_mid'] - 2 * df['bb_std']
    
    # RSI
    delta = df['close'].diff()
    gain = delta.where(delta > 0, 0).rolling(14).mean()
    loss = (-delta.where(delta < 0, 0)).rolling(14).mean()
    rs = gain / loss
    df['rsi'] = 100 - (100 / (1 + rs))
    
    # MACD
    ema12 = df['close'].ewm(span=12).mean()
    ema26 = df['close'].ewm(span=26).mean()
    df['macd'] = ema12 - ema26
    df['macd_signal'] = df['macd'].ewm(span=9).mean()
    df['macd_hist'] = df['macd'] - df['macd_signal']
    
    return df
```

### 6.3 时间特征

```python
def add_time_features(df):
    """添加时间特征"""
    df['hour'] = df.index.hour
    df['minute'] = df.index.minute
    df['dayofweek'] = df.index.dayofweek
    df['dayofmonth'] = df.index.day
    df['month'] = df.index.month
    df['quarter'] = df.index.quarter
    
    # 周期性编码
    df['hour_sin'] = np.sin(2 * np.pi * df['hour'] / 24)
    df['hour_cos'] = np.cos(2 * np.pi * df['hour'] / 24)
    df['dow_sin'] = np.sin(2 * np.pi * df['dayofweek'] / 7)
    df['dow_cos'] = np.cos(2 * np.pi * df['dayofweek'] / 7)
    
    # 交易时段
    df['is_open'] = (df.index.hour == 9) & (df.index.minute < 60)
    df['is_close'] = df.index.hour == 15
    
    return df
```

3. 处理多市场数据统一转UTC
4. 高频数据考虑使用Numba加速
5. 特征工程注意避免前瞻偏差
