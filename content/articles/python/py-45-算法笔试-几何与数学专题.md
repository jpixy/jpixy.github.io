# 一、基础向量运算

## 1.1 点与向量

```python
import math
from typing import Tuple, List

# 点的表示
Point = Tuple[float, float]
Vector = Tuple[float, float]


def vector(p1: Point, p2: Point) -> Vector:
    """
    创建从p1指向p2的向量
    
    参数:
        p1: Point - 起点 (x1, y1)
        p2: Point - 终点 (x2, y2)
    返回:
        Vector - 向量 (x2-x1, y2-y1)
    """
    return (p2[0] - p1[0], p2[1] - p1[1])


def add(v1: Vector, v2: Vector) -> Vector:
    """向量加法"""
    return (v1[0] + v2[0], v1[1] + v2[1])


def subtract(v1: Vector, v2: Vector) -> Vector:
    """向量减法"""
    return (v1[0] - v2[0], v1[1] - v2[1])


def scale(v: Vector, k: float) -> Vector:
    """向量数乘"""
    return (v[0] * k, v[1] * k)


def dot(v1: Vector, v2: Vector) -> float:
    """
    向量点积（内积）
    
    公式: v1 · v2 = x1*x2 + y1*y2 = |v1|*|v2|*cos(θ)
    
    应用:
        - 判断夹角: >0 锐角, =0 垂直, <0 钝角
        - 计算投影长度
    """
    return v1[0] * v2[0] + v1[1] * v2[1]


def cross(v1: Vector, v2: Vector) -> float:
    """
    向量叉积（外积的z分量）
    
    公式: v1 × v2 = x1*y2 - x2*y1 = |v1|*|v2|*sin(θ)
    
    应用:
        - 判断旋转方向: >0 逆时针, =0 共线, <0 顺时针
        - 计算平行四边形面积（三角形面积的2倍）
        - 判断点在向量的左侧还是右侧
    """
    return v1[0] * v2[1] - v1[1] * v2[0]


def magnitude(v: Vector) -> float:
    """向量模长"""
    return math.sqrt(v[0] ** 2 + v[1] ** 2)


def normalize(v: Vector) -> Vector:
    """单位向量"""
    m = magnitude(v)
    if m == 0:
        return (0, 0)
    return (v[0] / m, v[1] / m)


def angle(v1: Vector, v2: Vector) -> float:
    """
    两向量夹角（弧度）
    
    返回:
        float - 夹角 [0, π]
    """
    cos_theta = dot(v1, v2) / (magnitude(v1) * magnitude(v2))
    # 处理浮点误差
    cos_theta = max(-1, min(1, cos_theta))
    return math.acos(cos_theta)


def rotate(v: Vector, theta: float) -> Vector:
    """
    向量旋转
    
    参数:
        v: Vector - 原向量
        theta: float - 旋转角度（弧度，逆时针为正）
    返回:
        Vector - 旋转后的向量
    """
    cos_t = math.cos(theta)
    sin_t = math.sin(theta)
    return (
        v[0] * cos_t - v[1] * sin_t,
        v[0] * sin_t + v[1] * cos_t
    )
```

## 1.2 距离计算

```python
def distance(p1: Point, p2: Point) -> float:
    """
    两点间欧几里得距离
    
    公式: d = √((x2-x1)² + (y2-y1)²)
    """
    return math.sqrt((p2[0] - p1[0]) ** 2 + (p2[1] - p1[1]) ** 2)


def distance_squared(p1: Point, p2: Point) -> float:
    """
    两点间距离的平方
    
    用途: 避免开方运算，用于比较距离大小
    """
    return (p2[0] - p1[0]) ** 2 + (p2[1] - p1[1]) ** 2


def manhattan_distance(p1: Point, p2: Point) -> float:
    """
    曼哈顿距离
    
    公式: d = |x2-x1| + |y2-y1|
    应用: 网格移动（只能水平或垂直移动）
    """
    return abs(p2[0] - p1[0]) + abs(p2[1] - p1[1])


def chebyshev_distance(p1: Point, p2: Point) -> float:
    """
    切比雪夫距离
    
    公式: d = max(|x2-x1|, |y2-y1|)
    应用: 棋盘上国王的移动距离
    """
    return max(abs(p2[0] - p1[0]), abs(p2[1] - p1[1]))


def point_to_line_distance(p: Point, a: Point, b: Point) -> float:
    """
    点到直线的距离
    
    参数:
        p: Point - 目标点
        a, b: Point - 直线上两点
    返回:
        float - 点p到直线ab的距离
    
    公式: d = |AP × AB| / |AB|
    """
    ap = vector(a, p)
    ab = vector(a, b)
    return abs(cross(ap, ab)) / magnitude(ab)


def point_to_segment_distance(p: Point, a: Point, b: Point) -> float:
    """
    点到线段的距离
    
    参数:
        p: Point - 目标点
        a, b: Point - 线段端点
    返回:
        float - 点p到线段ab的最短距离
    """
    ap = vector(a, p)
    ab = vector(a, b)
    
    ab_len_sq = dot(ab, ab)
    if ab_len_sq == 0:  # a和b重合
        return magnitude(ap)
    
    # 投影比例
    t = dot(ap, ab) / ab_len_sq
    
    if t < 0:  # 最近点是a
        return distance(p, a)
    elif t > 1:  # 最近点是b
        return distance(p, b)
    else:  # 最近点在线段上
        # 投影点
        proj = (a[0] + t * ab[0], a[1] + t * ab[1])
        return distance(p, proj)
```

# 二、线段与直线

## 2.1 线段相交判断

```python
def sign(x: float) -> int:
    """符号函数"""
    if x > 1e-9:
        return 1
    elif x < -1e-9:
        return -1
    return 0


def segments_intersect(a: Point, b: Point, c: Point, d: Point) -> bool:
    """
    判断线段AB和CD是否相交
    
    LeetCode相关: 面试题16.03
    
    方法: 跨立实验
        1. AB跨立CD: C和D在AB两侧（或在AB上）
        2. CD跨立AB: A和B在CD两侧（或在CD上）
        3. 两个条件都满足则相交
    
    时间复杂度: O(1)
    """
    # AB向量
    ab = vector(a, b)
    # AC, AD向量
    ac = vector(a, c)
    ad = vector(a, d)
    
    # CD向量
    cd = vector(c, d)
    # CA, CB向量
    ca = vector(c, a)
    cb = vector(c, b)
    
    # 跨立实验
    d1 = cross(ab, ac)
    d2 = cross(ab, ad)
    d3 = cross(cd, ca)
    d4 = cross(cd, cb)
    
    # C和D在AB两侧
    if sign(d1) * sign(d2) < 0 and sign(d3) * sign(d4) < 0:
        return True
    
    # 处理共线情况
    if sign(d1) == 0 and on_segment(a, b, c):
        return True
    if sign(d2) == 0 and on_segment(a, b, d):
        return True
    if sign(d3) == 0 and on_segment(c, d, a):
        return True
    if sign(d4) == 0 and on_segment(c, d, b):
        return True
    
    return False


def on_segment(a: Point, b: Point, p: Point) -> bool:
    """
    判断点p是否在线段ab上（假设三点共线）
    """
    return (min(a[0], b[0]) <= p[0] <= max(a[0], b[0]) and
            min(a[1], b[1]) <= p[1] <= max(a[1], b[1]))


def line_intersection(a: Point, b: Point, c: Point, d: Point) -> Point:
    """
    计算两直线的交点
    
    参数:
        a, b: 直线1上两点
        c, d: 直线2上两点
    返回:
        交点坐标，若平行返回None
    
    方法: 使用参数方程求解
    """
    ab = vector(a, b)
    cd = vector(c, d)
    ac = vector(a, c)
    
    denom = cross(ab, cd)
    if abs(denom) < 1e-9:  # 平行
        return None
    
    t = cross(ac, cd) / denom
    
    return (a[0] + t * ab[0], a[1] + t * ab[1])
```

## 2.2 直线方程

```python
def line_equation(p1: Point, p2: Point) -> Tuple[float, float, float]:
    """
    根据两点求直线方程 ax + by + c = 0
    
    返回:
        (a, b, c) - 直线系数
    """
    a = p2[1] - p1[1]
    b = p1[0] - p2[0]
    c = p2[0] * p1[1] - p1[0] * p2[1]
    return (a, b, c)


def parallel(line1: Tuple, line2: Tuple) -> bool:
    """判断两直线是否平行"""
    a1, b1, _ = line1
    a2, b2, _ = line2
    return abs(a1 * b2 - a2 * b1) < 1e-9


def perpendicular(line1: Tuple, line2: Tuple) -> bool:
    """判断两直线是否垂直"""
    a1, b1, _ = line1
    a2, b2, _ = line2
    return abs(a1 * a2 + b1 * b2) < 1e-9
```

# 三、多边形

## 3.1 多边形面积

```python
def polygon_area(points: List[Point]) -> float:
    """
    计算多边形面积（Shoelace公式）
    
    参数:
        points: List[Point] - 多边形顶点（按顺序排列）
    返回:
        float - 面积（始终为正）
    
    公式: S = 0.5 * |Σ(xi * yi+1 - xi+1 * yi)|
    
    时间复杂度: O(n)
    """
    n = len(points)
    if n < 3:
        return 0
    
    area = 0
    for i in range(n):
        j = (i + 1) % n
        area += points[i][0] * points[j][1]
        area -= points[j][0] * points[i][1]
    
    return abs(area) / 2


def triangle_area(a: Point, b: Point, c: Point) -> float:
    """
    三角形面积
    
    公式: S = 0.5 * |AB × AC|
    """
    ab = vector(a, b)
    ac = vector(a, c)
    return abs(cross(ab, ac)) / 2


def triangle_area_heron(a: float, b: float, c: float) -> float:
    """
    海伦公式求三角形面积
    
    参数:
        a, b, c: float - 三边长度
    返回:
        float - 面积
    
    公式: S = √(s(s-a)(s-b)(s-c)), s = (a+b+c)/2
    """
    s = (a + b + c) / 2
    area_sq = s * (s - a) * (s - b) * (s - c)
    if area_sq < 0:  # 无效三角形
        return 0
    return math.sqrt(area_sq)
```

## 3.2 点与多边形关系

```python
def point_in_polygon(p: Point, polygon: List[Point]) -> int:
    """
    判断点与多边形的关系（射线法）
    
    参数:
        p: Point - 目标点
        polygon: List[Point] - 多边形顶点
    返回:
        1: 点在多边形内部
        0: 点在多边形边上
        -1: 点在多边形外部
    
    方法: 从点向右发射一条射线，统计与多边形边的交点数
          奇数个交点在内部，偶数个在外部
    
    时间复杂度: O(n)
    """
    n = len(polygon)
    if n < 3:
        return -1
    
    count = 0
    for i in range(n):
        a = polygon[i]
        b = polygon[(i + 1) % n]
        
        # 检查点是否在边上
        if on_segment_with_collinear_check(a, b, p):
            return 0
        
        # 射线与边相交判断
        if a[1] <= p[1]:
            if b[1] > p[1]:  # 向上穿越
                if cross(vector(a, b), vector(a, p)) > 0:
                    count += 1
        else:
            if b[1] <= p[1]:  # 向下穿越
                if cross(vector(a, b), vector(a, p)) < 0:
                    count += 1
    
    return 1 if count % 2 == 1 else -1


def on_segment_with_collinear_check(a: Point, b: Point, p: Point) -> bool:
    """检查点p是否在线段ab上"""
    # 先检查共线
    if abs(cross(vector(a, b), vector(a, p))) > 1e-9:
        return False
    # 再检查在线段范围内
    return on_segment(a, b, p)


def point_in_triangle(p: Point, a: Point, b: Point, c: Point) -> bool:
    """
    判断点是否在三角形内（含边界）
    
    方法: 面积法
        S(PAB) + S(PBC) + S(PCA) == S(ABC)
    """
    def sign(p1, p2, p3):
        return cross(vector(p1, p2), vector(p1, p3))
    
    d1 = sign(p, a, b)
    d2 = sign(p, b, c)
    d3 = sign(p, c, a)
    
    has_neg = (d1 < 0) or (d2 < 0) or (d3 < 0)
    has_pos = (d1 > 0) or (d2 > 0) or (d3 > 0)
    
    return not (has_neg and has_pos)
```

# 四、凸包

## 4.1 Graham扫描法

```python
def convex_hull_graham(points: List[Point]) -> List[Point]:
    """
    凸包 - Graham扫描法
    
    LeetCode 587. Erect the Fence
    
    参数:
        points: List[Point] - 点集
    返回:
        List[Point] - 凸包顶点（逆时针顺序）
    
    时间复杂度: O(n log n)
    空间复杂度: O(n)
    
    步骤:
        1. 找到最下最左的点作为起点
        2. 按极角排序
        3. 使用栈构建凸包
    """
    n = len(points)
    if n <= 3:
        return points
    
    # 找最下最左的点
    start = min(range(n), key=lambda i: (points[i][1], points[i][0]))
    pivot = points[start]
    
    # 按极角排序
    def polar_angle_key(p):
        if p == pivot:
            return (-float('inf'), 0)
        angle = math.atan2(p[1] - pivot[1], p[0] - pivot[0])
        dist = distance_squared(pivot, p)
        return (angle, dist)
    
    sorted_points = sorted(points, key=polar_angle_key)
    
    # 处理共线点（最后一组按距离降序）
    # 找最后一个极角相同的组
    i = n - 1
    while i > 0:
        v1 = vector(pivot, sorted_points[i])
        v2 = vector(pivot, sorted_points[i - 1])
        if abs(cross(v1, v2)) > 1e-9:
            break
        i -= 1
    
    # 反转最后一组
    sorted_points[i:] = sorted_points[i:][::-1]
    
    # Graham扫描
    stack = []
    for p in sorted_points:
        while len(stack) >= 2:
            v1 = vector(stack[-2], stack[-1])
            v2 = vector(stack[-1], p)
            if cross(v1, v2) < 0:  # 右转，弹出
                stack.pop()
            else:
                break
        stack.append(p)
    
    return stack
```

## 4.2 Andrew算法

```python
def convex_hull_andrew(points: List[Point]) -> List[Point]:
    """
    凸包 - Andrew算法（单调链算法）
    
    参数:
        points: List[Point] - 点集
    返回:
        List[Point] - 凸包顶点（逆时针顺序）
    
    时间复杂度: O(n log n)
    
    步骤:
        1. 按x坐标排序（x相同按y）
        2. 从左到右构建下凸包
        3. 从右到左构建上凸包
    """
    points = sorted(set(points))  # 去重并排序
    n = len(points)
    
    if n <= 2:
        return points
    
    # 构建下凸包
    lower = []
    for p in points:
        while len(lower) >= 2:
            v1 = vector(lower[-2], lower[-1])
            v2 = vector(lower[-1], p)
            if cross(v1, v2) <= 0:  # 右转或共线
                lower.pop()
            else:
                break
        lower.append(p)
    
    # 构建上凸包
    upper = []
    for p in reversed(points):
        while len(upper) >= 2:
            v1 = vector(upper[-2], upper[-1])
            v2 = vector(upper[-1], p)
            if cross(v1, v2) <= 0:
                upper.pop()
            else:
                break
        upper.append(p)
    
    # 合并（去掉重复的端点）
    return lower[:-1] + upper[:-1]
```

## 4.3 凸包应用

```python
def rotating_calipers(hull: List[Point]) -> float:
    """
    旋转卡壳 - 求凸包直径（最远点对距离）
    
    参数:
        hull: List[Point] - 凸包顶点（按顺序）
    返回:
        float - 最远点对距离
    
    时间复杂度: O(n)
    """
    n = len(hull)
    if n <= 1:
        return 0
    if n == 2:
        return distance(hull[0], hull[1])
    
    max_dist = 0
    j = 1  # 对踵点
    
    for i in range(n):
        # 找当前边的最远点
        while True:
            next_j = (j + 1) % n
            # 比较j+1和j到边(i, i+1)的距离
            edge = vector(hull[i], hull[(i + 1) % n])
            to_j = vector(hull[i], hull[j])
            to_next_j = vector(hull[i], hull[next_j])
            
            if cross(edge, to_next_j) > cross(edge, to_j):
                j = next_j
            else:
                break
        
        max_dist = max(max_dist, distance(hull[i], hull[j]))
        max_dist = max(max_dist, distance(hull[(i + 1) % n], hull[j]))
    
    return max_dist


def min_enclosing_rectangle(hull: List[Point]) -> float:
    """
    最小外接矩形面积（旋转卡壳）
    
    时间复杂度: O(n)
    """
    n = len(hull)
    if n < 3:
        return 0
    
    min_area = float('inf')
    j = k = m = 1  # 对踵点
    
    for i in range(n):
        # 边向量
        edge = vector(hull[i], hull[(i + 1) % n])
        edge_len = magnitude(edge)
        unit_edge = normalize(edge)
        
        # 找最远点（高度）
        while True:
            next_j = (j + 1) % n
            if abs(cross(edge, vector(hull[i], hull[next_j]))) > \
               abs(cross(edge, vector(hull[i], hull[j]))):
                j = next_j
            else:
                break
        
        # 找最右点
        while True:
            next_k = (k + 1) % n
            if dot(edge, vector(hull[i], hull[next_k])) > \
               dot(edge, vector(hull[i], hull[k])):
                k = next_k
            else:
                break
        
        if i == 0:
            m = j
        
        # 找最左点
        while True:
            next_m = (m + 1) % n
            if dot(edge, vector(hull[i], hull[next_m])) < \
               dot(edge, vector(hull[i], hull[m])):
                m = next_m
            else:
                break
        
        # 计算面积
        height = abs(cross(edge, vector(hull[i], hull[j]))) / edge_len
        width = (dot(edge, vector(hull[i], hull[k])) - 
                 dot(edge, vector(hull[i], hull[m]))) / edge_len
        
        min_area = min(min_area, height * width)
    
    return min_area
```

# 五、圆

## 5.1 圆的基本运算

```python
def circle_area(r: float) -> float:
    """圆面积"""
    return math.pi * r * r


def circle_perimeter(r: float) -> float:
    """圆周长"""
    return 2 * math.pi * r


def point_in_circle(p: Point, center: Point, r: float) -> int:
    """
    点与圆的关系
    
    返回:
        1: 点在圆内
        0: 点在圆上
        -1: 点在圆外
    """
    d = distance(p, center)
    if abs(d - r) < 1e-9:
        return 0
    return 1 if d < r else -1


def circle_intersection(c1: Point, r1: float, c2: Point, r2: float) -> int:
    """
    两圆的位置关系
    
    返回:
        0: 内含
        1: 内切
        2: 相交
        3: 外切
        4: 相离
    """
    d = distance(c1, c2)
    
    if d < abs(r1 - r2) - 1e-9:
        return 0  # 内含
    elif abs(d - abs(r1 - r2)) < 1e-9:
        return 1  # 内切
    elif d < r1 + r2 - 1e-9:
        return 2  # 相交
    elif abs(d - (r1 + r2)) < 1e-9:
        return 3  # 外切
    else:
        return 4  # 相离


def circle_intersection_points(c1: Point, r1: float, 
                                c2: Point, r2: float) -> List[Point]:
    """
    计算两圆交点
    
    返回:
        交点列表（0、1或2个点）
    """
    d = distance(c1, c2)
    
    # 无交点的情况
    if d > r1 + r2 + 1e-9 or d < abs(r1 - r2) - 1e-9:
        return []
    
    # 一个交点（相切）
    if abs(d - (r1 + r2)) < 1e-9 or abs(d - abs(r1 - r2)) < 1e-9:
        t = r1 / d
        return [(c1[0] + t * (c2[0] - c1[0]), 
                 c1[1] + t * (c2[1] - c1[1]))]
    
    # 两个交点
    a = (r1 * r1 - r2 * r2 + d * d) / (2 * d)
    h = math.sqrt(r1 * r1 - a * a)
    
    # 中点
    px = c1[0] + a * (c2[0] - c1[0]) / d
    py = c1[1] + a * (c2[1] - c1[1]) / d
    
    # 两个交点
    dx = h * (c2[1] - c1[1]) / d
    dy = h * (c2[0] - c1[0]) / d
    
    return [(px + dx, py - dy), (px - dx, py + dy)]
```

## 5.2 最小覆盖圆

```python
import random

def min_enclosing_circle(points: List[Point]) -> Tuple[Point, float]:
    """
    最小覆盖圆（Welzl算法）
    
    参数:
        points: List[Point] - 点集
    返回:
        (center, radius) - 圆心和半径
    
    时间复杂度: 期望O(n)
    """
    points = list(points)
    random.shuffle(points)
    
    def circle_from_3(p1, p2, p3):
        """三点确定一个圆"""
        ax, ay = p1
        bx, by = p2
        cx, cy = p3
        
        d = 2 * (ax * (by - cy) + bx * (cy - ay) + cx * (ay - by))
        if abs(d) < 1e-9:
            return None, float('inf')
        
        ux = ((ax*ax + ay*ay) * (by - cy) + 
              (bx*bx + by*by) * (cy - ay) + 
              (cx*cx + cy*cy) * (ay - by)) / d
        uy = ((ax*ax + ay*ay) * (cx - bx) + 
              (bx*bx + by*by) * (ax - cx) + 
              (cx*cx + cy*cy) * (bx - ax)) / d
        
        center = (ux, uy)
        radius = distance(center, p1)
        return center, radius
    
    def circle_from_2(p1, p2):
        """两点确定一个圆（直径）"""
        center = ((p1[0] + p2[0]) / 2, (p1[1] + p2[1]) / 2)
        radius = distance(p1, p2) / 2
        return center, radius
    
    def in_circle(p, center, r):
        return distance(p, center) <= r + 1e-9
    
    def welzl(P, R):
        if len(P) == 0 or len(R) == 3:
            if len(R) == 0:
                return (0, 0), 0
            elif len(R) == 1:
                return R[0], 0
            elif len(R) == 2:
                return circle_from_2(R[0], R[1])
            else:
                return circle_from_3(R[0], R[1], R[2])
        
        p = P[0]
        center, r = welzl(P[1:], R)
        
        if in_circle(p, center, r):
            return center, r
        
        return welzl(P[1:], R + [p])
    
    return welzl(points, [])
```

## 6.1 最近点对

```python
def closest_pair(points: List[Point]) -> float:
    """
    最近点对（分治法）
    
    LeetCode相关问题
    
    参数:
        points: List[Point] - 点集
    返回:
        float - 最近点对距离
    
    时间复杂度: O(n log n)
    """
    def brute_force(pts):
        min_dist = float('inf')
        n = len(pts)
        for i in range(n):
            for j in range(i + 1, n):
                min_dist = min(min_dist, distance(pts[i], pts[j]))
        return min_dist
    
    def closest_util(pts_x, pts_y):
        n = len(pts_x)
        
        # 小规模直接暴力
        if n <= 3:
            return brute_force(pts_x)
        
        mid = n // 2
        mid_point = pts_x[mid]
        
        # 分割
        pts_y_left = []
        pts_y_right = []
        for p in pts_y:
            if p[0] <= mid_point[0]:
                pts_y_left.append(p)
            else:
                pts_y_right.append(p)
        
        # 递归
        dl = closest_util(pts_x[:mid], pts_y_left)
        dr = closest_util(pts_x[mid:], pts_y_right)
        d = min(dl, dr)
        
        # 合并：检查跨区域的点对
        strip = [p for p in pts_y if abs(p[0] - mid_point[0]) < d]
        
        for i in range(len(strip)):
            j = i + 1
            while j < len(strip) and strip[j][1] - strip[i][1] < d:
                d = min(d, distance(strip[i], strip[j]))
                j += 1
        
        return d
    
    pts_x = sorted(points, key=lambda p: p[0])
    pts_y = sorted(points, key=lambda p: p[1])
    
    return closest_util(pts_x, pts_y)
```

## 6.2 判断矩形重叠

```python
def rectangles_overlap(r1: List[int], r2: List[int]) -> bool:
    """
    判断两矩形是否重叠
    
    LeetCode 836. Rectangle Overlap
    
    参数:
        r1, r2: [x1, y1, x2, y2] - 左下角和右上角坐标
    返回:
        bool - 是否重叠
    """
    # 不重叠的条件（取反）
    return not (r1[2] <= r2[0] or  # r1在r2左边
                r1[0] >= r2[2] or  # r1在r2右边
                r1[3] <= r2[1] or  # r1在r2下边
                r1[1] >= r2[3])    # r1在r2上边


def rectangle_intersection_area(r1: List[int], r2: List[int]) -> int:
    """
    计算两矩形交集面积
    
    LeetCode 223. Rectangle Area
    """
    x_overlap = max(0, min(r1[2], r2[2]) - max(r1[0], r2[0]))
    y_overlap = max(0, min(r1[3], r2[3]) - max(r1[1], r2[1]))
    return x_overlap * y_overlap
```

## 6.3 有效数字判断

```python
def valid_number(s: str) -> bool:
    """
    LeetCode 65. Valid Number
    
    判断字符串是否是有效数字
    
    有效形式:
        - 整数: "2", "-3", "+5"
        - 小数: "3.14", ".5", "2."
        - 科学计数法: "1e10", "3.14E-5"
    """
    s = s.strip()
    if not s:
        return False
    
    # 状态机或直接解析
    seen_digit = False
    seen_dot = False
    seen_e = False
    
    for i, c in enumerate(s):
        if c.isdigit():
            seen_digit = True
        elif c in ['+', '-']:
            # 符号只能在开头或e后面
            if i > 0 and s[i-1].lower() != 'e':
                return False
        elif c == '.':
            # 小数点不能在e后面，且只能出现一次
            if seen_dot or seen_e:
                return False
            seen_dot = True
        elif c.lower() == 'e':
            # e前必须有数字，且只能出现一次
            if seen_e or not seen_digit:
                return False
            seen_e = True
            seen_digit = False  # e后面必须有数字
        else:
            return False
    
    return seen_digit
```

## 总结

### 几何算法分类

| 类型 | 算法/问题 | 时间复杂度 |
|------|----------|-----------|
| 基础 | 向量运算、点积叉积 | O(1) |
| 距离 | 点到线距离、点到线段距离 | O(1) |
| 线段 | 线段相交判断 | O(1) |
| 多边形 | 面积计算、点在多边形内 | O(n) |
| 凸包 | Graham/Andrew算法 | O(n log n) |
| 旋转卡壳 | 最远点对、最小外接矩形 | O(n) |
| 圆 | 最小覆盖圆 | 期望O(n) |
| 分治 | 最近点对 | O(n log n) |

### 关键公式

| 运算 | 公式 | 含义 |
|------|------|------|
| 点积 | x1×x2 + y1×y2 | 投影、夹角 |
| 叉积 | x1×y2 - x2×y1 | 面积、方向 |
| 距离 | √((x2-x1)² + (y2-y1)²) | 欧几里得距离 |
| 多边形面积 | 0.5×\|Σ(xi×yi+1 - xi+1×yi)\| | Shoelace公式 |

### 注意事项

1. **浮点精度**: 使用eps=1e-9进行比较
2. **边界情况**: 共线、平行、重合
3. **方向判断**: 叉积>0逆时针，<0顺时针
4. **凸包**: 注意去重和共线点处理
