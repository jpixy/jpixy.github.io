## 1.2 常用位运算技巧

```python
# 判断奇偶
n & 1 == 0    # 偶数
n & 1 == 1    # 奇数

# 乘除2的幂
n << k        # n * 2^k
n >> k        # n // 2^k

# 判断是否2的幂
n > 0 and (n & (n - 1)) == 0

# 获取最低位的1
n & (-n)      # lowbit

# 去掉最低位的1
n & (n - 1)

# 设置第k位为1（从0开始）
n | (1 << k)

# 设置第k位为0
n & ~(1 << k)

# 翻转第k位
n ^ (1 << k)

# 获取第k位的值
(n >> k) & 1

# 将最后k位设为0
n & ~((1 << k) - 1)

# 将最后k位设为1
n | ((1 << k) - 1)

# 取反最后k位
n ^ ((1 << k) - 1)

# 判断第k位是否为1
(n >> k) & 1 == 1
# 或
n & (1 << k) != 0
```

## 1.3 常见位运算问题

```python
# 1. 计算二进制中1的个数（Brian Kernighan算法）
def count_bits(n):
    count = 0
    while n:
        n &= n - 1  # 去掉最低位的1
        count += 1
    return count

# 内置方法
n.bit_count()      # Python 3.10+
bin(n).count('1')  # 通用方法

# 2. 只出现一次的数字（其他数字出现两次）
def single_number(nums):
    result = 0
    for num in nums:
        result ^= num
    return result

# 3. 只出现一次的数字II（其他数字出现三次）
def single_number_ii(nums):
    ones, twos = 0, 0
    for num in nums:
        ones = (ones ^ num) & ~twos
        twos = (twos ^ num) & ~ones
    return ones

# 4. 两个只出现一次的数字
def single_numbers(nums):
    xor = 0
    for num in nums:
        xor ^= num
    
    # 找到任意一个不同的位
    diff = xor & (-xor)
    
    a, b = 0, 0
    for num in nums:
        if num & diff:
            a ^= num
        else:
            b ^= num
    
    return [a, b]

# 5. 不用加减法实现加法
def add(a, b):
    MASK = 0xFFFFFFFF
    MAX_INT = 0x7FFFFFFF
    
    while b != 0:
        carry = ((a & b) << 1) & MASK
        a = (a ^ b) & MASK
        b = carry
    
    return a if a <= MAX_INT else ~(a ^ MASK)

# 6. 位运算求平均（避免溢出）
def average(a, b):
    return (a & b) + ((a ^ b) >> 1)

# 7. 交换两个数
a, b = b, a  # Python推荐
# 或
a ^= b
b ^= a
a ^= b

# 8. 判断两数符号是否相同
def same_sign(a, b):
    return (a ^ b) >= 0
```

## 1.4 位掩码与状态压缩

```python
# 使用位掩码表示集合
# 集合 {0, 2, 3} 可以表示为二进制 1101 = 13

# 添加元素i
mask |= (1 << i)

# 删除元素i
mask &= ~(1 << i)

# 检查元素i是否存在
(mask >> i) & 1 == 1

# 遍历所有子集
def enumerate_subsets(n):
    """枚举n个元素的所有子集"""
    for mask in range(1 << n):
        subset = []
        for i in range(n):
            if mask & (1 << i):
                subset.append(i)
        yield subset

# 遍历某个集合的所有子集
def enumerate_subsets_of(mask):
    """枚举mask的所有子集（包括空集）"""
    subset = mask
    while subset:
        yield subset
        subset = (subset - 1) & mask
    yield 0

# 示例：旅行商问题（TSP）
def tsp(dist):
    """状态压缩DP解决TSP"""
    n = len(dist)
    dp = [[float('inf')] * n for _ in range(1 << n)]
    dp[1][0] = 0  # 从城市0开始
    
    for mask in range(1 << n):
        for u in range(n):
            if not (mask & (1 << u)):
                continue
            for v in range(n):
                if mask & (1 << v):
                    continue
                new_mask = mask | (1 << v)
                dp[new_mask][v] = min(
                    dp[new_mask][v],
                    dp[mask][u] + dist[u][v]
                )
    
    # 返回起点
    full_mask = (1 << n) - 1
    return min(dp[full_mask][u] + dist[u][0] for u in range(1, n))
```

# 二、数论基础

## 2.1 最大公约数与最小公倍数

```python
import math

# 最大公约数 GCD
math.gcd(12, 18)           # 6
math.gcd(12, 18, 24)       # 6 (Python 3.9+)

# 欧几里得算法
def gcd(a, b):
    while b:
        a, b = b, a % b
    return a

# 递归版本
def gcd_recursive(a, b):
    return a if b == 0 else gcd_recursive(b, a % b)

# 最小公倍数 LCM
math.lcm(4, 6)             # 12 (Python 3.9+)

def lcm(a, b):
    return a * b // math.gcd(a, b)

# 多个数的GCD/LCM
from functools import reduce

def gcd_list(nums):
    return reduce(math.gcd, nums)

def lcm_list(nums):
    return reduce(lcm, nums)

# 扩展欧几里得算法
def extended_gcd(a, b):
    """返回 (gcd, x, y) 使得 ax + by = gcd"""
    if b == 0:
        return a, 1, 0
    gcd, x1, y1 = extended_gcd(b, a % b)
    x = y1
    y = x1 - (a // b) * y1
    return gcd, x, y

# 示例：求解 ax + by = gcd(a, b)
gcd, x, y = extended_gcd(35, 15)
# gcd=5, x=1, y=-2
# 验证: 35*1 + 15*(-2) = 35 - 30 = 5 ✓
```

## 2.2 素数相关

```python
import math

# 判断素数
def is_prime(n):
    if n < 2:
        return False
    if n == 2:
        return True
    if n % 2 == 0:
        return False
    for i in range(3, int(math.sqrt(n)) + 1, 2):
        if n % i == 0:
            return False
    return True

# 埃氏筛法（Sieve of Eratosthenes）
def sieve(n):
    """返回所有小于n的素数"""
    is_prime = [True] * n
    is_prime[0] = is_prime[1] = False
    
    for i in range(2, int(math.sqrt(n)) + 1):
        if is_prime[i]:
            for j in range(i * i, n, i):
                is_prime[j] = False
    
    return [i for i in range(n) if is_prime[i]]

# 线性筛法（欧拉筛）
def linear_sieve(n):
    """O(n)时间复杂度的筛法"""
    is_prime = [True] * n
    primes = []
    
    for i in range(2, n):
        if is_prime[i]:
            primes.append(i)
        for p in primes:
            if i * p >= n:
                break
            is_prime[i * p] = False
            if i % p == 0:
                break
    
    return primes

# 分解质因数
def prime_factors(n):
    """返回n的质因数分解 {因子: 次数}"""
    factors = {}
    d = 2
    while d * d <= n:
        while n % d == 0:
            factors[d] = factors.get(d, 0) + 1
            n //= d
        d += 1
    if n > 1:
        factors[n] = factors.get(n, 0) + 1
    return factors

# 示例
prime_factors(360)  # {2: 3, 3: 2, 5: 1} 即 2³ × 3² × 5

# 所有因子
def get_factors(n):
    """返回n的所有因子"""
    factors = []
    for i in range(1, int(math.sqrt(n)) + 1):
        if n % i == 0:
            factors.append(i)
            if i != n // i:
                factors.append(n // i)
    return sorted(factors)

# 因子个数
def count_factors(n):
    """计算n的因子个数"""
    count = 0
    i = 1
    while i * i <= n:
        if n % i == 0:
            count += 1 if i * i == n else 2
        i += 1
    return count
```

## 2.3 模运算

```python
MOD = 10**9 + 7

# 基本模运算
(a + b) % MOD
(a - b + MOD) % MOD  # 避免负数
(a * b) % MOD

# 快速幂
def power_mod(base, exp, mod):
    result = 1
    base %= mod
    while exp > 0:
        if exp & 1:
            result = result * base % mod
        base = base * base % mod
        exp >>= 1
    return result

# Python内置
pow(base, exp, mod)

# 模逆元（费马小定理，mod必须是素数）
def mod_inverse(a, mod):
    return pow(a, mod - 2, mod)

# 模逆元（扩展欧几里得，适用于任意mod）
def mod_inverse_extended(a, mod):
    gcd, x, _ = extended_gcd(a, mod)
    if gcd != 1:
        return None  # 逆元不存在
    return x % mod

# 模除法
def mod_div(a, b, mod):
    return a * mod_inverse(b, mod) % mod

# 组合数取模
def comb_mod(n, r, mod):
    """计算 C(n, r) % mod"""
    if r > n or r < 0:
        return 0
    
    numerator = 1
    denominator = 1
    for i in range(r):
        numerator = numerator * (n - i) % mod
        denominator = denominator * (i + 1) % mod
    
    return numerator * pow(denominator, mod - 2, mod) % mod

# 预处理阶乘（大量组合数计算）
def precompute_factorial(n, mod):
    fact = [1] * (n + 1)
    for i in range(1, n + 1):
        fact[i] = fact[i-1] * i % mod
    
    inv_fact = [1] * (n + 1)
    inv_fact[n] = pow(fact[n], mod - 2, mod)
    for i in range(n - 1, -1, -1):
        inv_fact[i] = inv_fact[i+1] * (i+1) % mod
    
    return fact, inv_fact

# 使用预处理计算组合数
fact, inv_fact = precompute_factorial(100000, MOD)
def C(n, r):
    if r > n or r < 0:
        return 0
    return fact[n] * inv_fact[r] % MOD * inv_fact[n-r] % MOD
```

## 2.4 其他数论算法

```python
# 中国剩余定理（CRT）
def chinese_remainder_theorem(remainders, mods):
    """
    求解 x ≡ r[i] (mod m[i]) 的最小非负整数解
    要求所有m[i]两两互素
    """
    from functools import reduce
    
    M = reduce(lambda a, b: a * b, mods)
    x = 0
    for r, m in zip(remainders, mods):
        Mi = M // m
        _, inv, _ = extended_gcd(Mi, m)
        x += r * Mi * inv
    
    return x % M

# 欧拉函数 φ(n)
def euler_phi(n):
    """计算小于n且与n互素的正整数个数"""
    result = n
    p = 2
    while p * p <= n:
        if n % p == 0:
            while n % p == 0:
                n //= p
            result -= result // p
        p += 1
    if n > 1:
        result -= result // n
    return result

# 筛法计算欧拉函数
def euler_phi_sieve(n):
    """计算1到n-1的所有欧拉函数值"""
    phi = list(range(n))
    for i in range(2, n):
        if phi[i] == i:  # i是素数
            for j in range(i, n, i):
                phi[j] -= phi[j] // i
    return phi

# 离散对数（Baby-step Giant-step）
def discrete_log(a, b, m):
    """求解 a^x ≡ b (mod m) 的最小非负整数x"""
    import math
    
    n = int(math.ceil(math.sqrt(m)))
    
    # Baby step: 计算 a^0, a^1, ..., a^(n-1)
    table = {}
    power = 1
    for j in range(n):
        if power == b:
            return j
        table[power] = j
        power = power * a % m
    
    # Giant step: 计算 a^(-n)
    factor = pow(a, n * (m - 2), m)  # a^(-n) mod m
    
    gamma = b
    for i in range(n):
        if gamma in table:
            return i * n + table[gamma]
        gamma = gamma * factor % m
    
    return -1  # 无解
```

## 3.1 约瑟夫问题

```python
# 约瑟夫问题：n个人围成圈，每次数到第k个人出列
def josephus(n, k):
    """返回最后剩下的人的编号（0-indexed）"""
    if n == 1:
        return 0
    return (josephus(n - 1, k) + k) % n

# 迭代版本
def josephus_iterative(n, k):
    result = 0
    for i in range(2, n + 1):
        result = (result + k) % i
    return result

# 如果编号从1开始
def josephus_1indexed(n, k):
    return josephus(n, k) + 1
```

```python

# 大整数加法
a = 12345678901234567890

c = a + b

d = a * b

# 字符串形式的大数运算（某些OJ可能需要）
def add_strings(num1, num2):
    """字符串形式的大数加法"""
    i, j = len(num1) - 1, len(num2) - 1
    carry = 0
    result = []
    
    while i >= 0 or j >= 0 or carry:
        x = int(num1[i]) if i >= 0 else 0
        y = int(num2[j]) if j >= 0 else 0
        total = x + y + carry
        result.append(str(total % 10))
        carry = total // 10
        i -= 1
        j -= 1
    
    return ''.join(reversed(result))

def multiply_strings(num1, num2):
    """字符串形式的大数乘法"""
    if num1 == "0" or num2 == "0":
        return "0"
    
    n1, n2 = len(num1), len(num2)
    result = [0] * (n1 + n2)
    
    for i in range(n1 - 1, -1, -1):
        for j in range(n2 - 1, -1, -1):
            mul = int(num1[i]) * int(num2[j])
            p1, p2 = i + j, i + j + 1
            total = mul + result[p2]
            result[p2] = total % 10
            result[p1] += total // 10
    
    # 去除前导零
    result_str = ''.join(map(str, result)).lstrip('0')
    return result_str if result_str else "0"
```

```python
def to_base(n, base):
    """将十进制n转换为base进制字符串"""
    if n == 0:
        return '0'
    
    digits = "0123456789ABCDEFGHIJKLMNOPQRSTUVWXYZ"
    result = []
    negative = n < 0
    n = abs(n)
    
    while n:
        result.append(digits[n % base])
        n //= base
    
    if negative:
        result.append('-')
    
    return ''.join(reversed(result))

def from_base(s, base):
    """将base进制字符串s转换为十进制"""
    return int(s, base)

# 任意进制转任意进制
def convert_base(s, from_base, to_base):
    decimal = int(s, from_base)
    return to_base(decimal, to_base) if to_base != 10 else str(decimal)
```

## 总结

### 位运算速查

| 操作 | 代码 | 说明 |
|------|------|------|
| 判断奇偶 | `n & 1` | 1为奇，0为偶 |
| 判断2的幂 | `n & (n-1) == 0` | n>0且结果为0 |
| 最低位的1 | `n & (-n)` | lowbit |
| 去掉最低位1 | `n & (n-1)` | Brian Kernighan |
| 设置第k位 | `n \| (1<<k)` | 置1 |
| 清除第k位 | `n & ~(1<<k)` | 置0 |
| 获取第k位 | `(n>>k) & 1` | 返回0或1 |

### 数论公式

| 公式 | 说明 |
|------|------|
| `gcd(a,b) * lcm(a,b) = a * b` | GCD与LCM关系 |
| `a^(p-1) ≡ 1 (mod p)` | 费马小定理（p为素数） |
| `a^(-1) ≡ a^(p-2) (mod p)` | 模逆元 |
| `φ(n) = n * ∏(1-1/p)` | 欧拉函数 |

### 关键记忆

1. 位运算优先级低于比较运算符：`(n & 1) == 0`
2. Python整数无限精度，取反需注意
3. 模运算：`(a - b + MOD) % MOD`避免负数
4. 快速幂用`pow(base, exp, mod)`
5. 素数判断只需检查到`sqrt(n)`
