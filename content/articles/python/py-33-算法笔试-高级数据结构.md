# 一、Trie字典树

## 1.1 基础实现

```python
class TrieNode:
    """
    Trie节点类
    
    属性:
        children: Dict[str, TrieNode] - 子节点映射
        is_end: bool - 是否是某个单词的结尾
        count: int - 以此节点结尾的单词数量（可选）
    """
    def __init__(self):
        self.children = {}
        self.is_end = False
        self.count = 0  # 可选：记录单词出现次数

class Trie:
    """
    Trie（前缀树/字典树）
    
    用途:
        - 字符串的快速查找
        - 前缀匹配
        - 自动补全
        - 拼写检查
    
    时间复杂度:
        - 插入: O(m)，m是字符串长度
        - 查找: O(m)
        - 前缀查找: O(m)
    
    空间复杂度: O(字符集大小 × 字符串总长度)
    """
    
    def __init__(self):
        """初始化Trie"""
        self.root = TrieNode()
    
    def insert(self, word: str) -> None:
        """
        插入单词
        
        参数:
            word: str - 要插入的单词
        返回:
            None
        
        时间复杂度: O(m)，m是word长度
        """
        node = self.root
        for char in word:
            if char not in node.children:
                node.children[char] = TrieNode()
            node = node.children[char]
        node.is_end = True
        node.count += 1
    
    def search(self, word: str) -> bool:
        """
        搜索完整单词
        
        参数:
            word: str - 要搜索的单词
        返回:
            bool - 单词是否存在
        
        时间复杂度: O(m)
        """
        node = self._find_node(word)
        return node is not None and node.is_end
    
    def starts_with(self, prefix: str) -> bool:
        """
        检查是否存在以prefix为前缀的单词
        
        参数:
            prefix: str - 前缀
        返回:
            bool - 是否存在该前缀
        
        时间复杂度: O(m)
        """
        return self._find_node(prefix) is not None
    
    def _find_node(self, prefix: str) -> TrieNode:
        """
        查找前缀对应的节点（内部方法）
        
        返回:
            TrieNode - 前缀末尾的节点，不存在返回None
        """
        node = self.root
        for char in prefix:
            if char not in node.children:
                return None
            node = node.children[char]
        return node
    
    def count_prefix(self, prefix: str) -> int:
        """
        统计以prefix为前缀的单词数量
        
        返回:
            int - 前缀匹配的单词数
        """
        node = self._find_node(prefix)
        if node is None:
            return 0
        
        # DFS统计所有以此节点为根的单词
        count = 0
        def dfs(n):
            nonlocal count
            count += n.count
            for child in n.children.values():
                dfs(child)
        
        dfs(node)
        return count
    
    def get_words_with_prefix(self, prefix: str) -> list:
        """
        获取所有以prefix为前缀的单词
        
        返回:
            List[str] - 匹配的单词列表
        """
        node = self._find_node(prefix)
        if node is None:
            return []
        
        results = []
        def dfs(n, path):
            if n.is_end:
                results.append(path)
            for char, child in n.children.items():
                dfs(child, path + char)
        
        dfs(node, prefix)
        return results


# 使用示例
trie = Trie()
trie.insert("apple")
trie.insert("app")
trie.insert("application")

trie.search("app")           # True
trie.search("ap")            # False
trie.starts_with("app")      # True
trie.get_words_with_prefix("app")  # ['app', 'apple', 'application']
```

class TrieArray:
    """
    使用二维数组实现的Trie（更快）
    
    适用于字符集较小（如只有小写字母）的情况
    """
    
    def __init__(self, max_nodes=100001):
        """
        参数:
            max_nodes: int - 最大节点数
        """
        self.next = [[0] * 26 for _ in range(max_nodes)]
        self.is_end = [False] * max_nodes
        self.cnt = 0  # 当前节点数
    
    def insert(self, word: str) -> None:
        cur = 0
        for char in word:
            idx = ord(char) - ord('a')
            if self.next[cur][idx] == 0:
                self.cnt += 1
                self.next[cur][idx] = self.cnt
            cur = self.next[cur][idx]
        self.is_end[cur] = True
    
    def search(self, word: str) -> bool:
        cur = 0
        for char in word:
            idx = ord(char) - ord('a')
            if self.next[cur][idx] == 0:
                return False
            cur = self.next[cur][idx]
        return self.is_end[cur]
    
    def starts_with(self, prefix: str) -> bool:
        cur = 0
        for char in prefix:
            idx = ord(char) - ord('a')
            if self.next[cur][idx] == 0:
                return False
            cur = self.next[cur][idx]
        return True

class WordDictionary:
    """
    支持 . 通配符的字典
    
    LeetCode 211. Design Add and Search Words Data Structure
    """
    def __init__(self):
        self.root = TrieNode()
    
    def addWord(self, word: str) -> None:
        node = self.root
        for char in word:
            if char not in node.children:
                node.children[char] = TrieNode()
            node = node.children[char]
        node.is_end = True
    
    def search(self, word: str) -> bool:
        def dfs(node, i):
            if i == len(word):
                return node.is_end
            
            char = word[i]
            if char == '.':
                # 匹配任意字符
                for child in node.children.values():
                    if dfs(child, i + 1):
                        return True
                return False
            else:
                if char not in node.children:
                    return False
                return dfs(node.children[char], i + 1)
        
        return dfs(self.root, 0)

# 3. 最大异或对
def find_maximum_xor(nums: list) -> int:
    """
    找到数组中两个数的最大异或值
    
    LeetCode 421. Maximum XOR of Two Numbers in an Array
    使用二进制Trie
    """
    # 构建二进制Trie
    root = {}
    for num in nums:
        node = root
        for i in range(31, -1, -1):
            bit = (num >> i) & 1
            if bit not in node:
                node[bit] = {}
            node = node[bit]
    
    # 查找最大异或
    max_xor = 0
    for num in nums:
        node = root
        xor_val = 0
        for i in range(31, -1, -1):
            bit = (num >> i) & 1
            toggle = 1 - bit  # 尽量走相反的路
            if toggle in node:
                xor_val |= (1 << i)
                node = node[toggle]
            else:
                node = node[bit]
        max_xor = max(max_xor, xor_val)
    
    return max_xor

# 二、并查集 Union-Find

## 2.1 基础实现

```python
class UnionFind:
    """
    并查集（Union-Find / Disjoint Set Union）
    
    用途:
        - 判断两个元素是否在同一集合
        - 合并两个集合
        - 统计连通分量
    
    时间复杂度:
        - find: O(α(n)) ≈ O(1)，α是阿克曼函数的反函数
        - union: O(α(n)) ≈ O(1)
    
    空间复杂度: O(n)
    """
    
    def __init__(self, n: int):
        """
        初始化并查集
        
        参数:
            n: int - 元素数量（0到n-1）
        """
        self.parent = list(range(n))  # 每个元素的父节点
        self.rank = [0] * n           # 树的秩（用于按秩合并）
        self.count = n                # 连通分量数量
    
    def find(self, x: int) -> int:
        """
        查找x的根节点（代表元素）
        
        参数:
            x: int - 要查找的元素
        返回:
            int - x所在集合的根节点
        
        使用路径压缩优化
        """
        if self.parent[x] != x:
            self.parent[x] = self.find(self.parent[x])  # 路径压缩
        return self.parent[x]
    
    def union(self, x: int, y: int) -> bool:
        """
        合并x和y所在的集合
        
        参数:
            x, y: int - 要合并的两个元素
        返回:
            bool - True表示成功合并，False表示已在同一集合
        
        使用按秩合并优化
        """
        px, py = self.find(x), self.find(y)
        if px == py:
            return False
        
        # 按秩合并：将较矮的树接到较高的树下
        if self.rank[px] < self.rank[py]:
            px, py = py, px
        self.parent[py] = px
        if self.rank[px] == self.rank[py]:
            self.rank[px] += 1
        
        self.count -= 1
        return True
    
    def connected(self, x: int, y: int) -> bool:
        """
        判断x和y是否在同一集合
        
        返回:
            bool - 是否连通
        """
        return self.find(x) == self.find(y)
    
    def get_count(self) -> int:
        """
        获取连通分量数量
        
        返回:
            int - 当前连通分量数
        """
        return self.count


# 使用示例
uf = UnionFind(5)
uf.union(0, 1)
uf.union(2, 3)
print(uf.connected(0, 1))  # True
print(uf.connected(0, 2))  # False
print(uf.get_count())      # 3
uf.union(1, 3)
print(uf.connected(0, 2))  # True
print(uf.get_count())      # 2
```

## 2.2 带权并查集

```python
class WeightedUnionFind:
    """
    带权并查集
    
    用途: 维护元素之间的相对关系（如相对大小、距离等）
    
    weight[x] 表示 x 到 parent[x] 的权值
    """
    
    def __init__(self, n: int):
        self.parent = list(range(n))
        self.weight = [0.0] * n  # x到parent[x]的权值
    
    def find(self, x: int) -> int:
        """
        查找并进行路径压缩，同时更新权值
        """
        if self.parent[x] != x:
            root = self.find(self.parent[x])
            self.weight[x] += self.weight[self.parent[x]]
            self.parent[x] = root
        return self.parent[x]
    
    def union(self, x: int, y: int, w: float) -> bool:
        """
        合并x和y，设置 x/y = w（即 x 到 y 的权值为 w）
        
        参数:
            x, y: int - 要合并的元素
            w: float - x到y的权值（x = w * y）
        """
        px, py = self.find(x), self.find(y)
        if px == py:
            return False
        
        # weight[px] = ?
        # x -> px: weight[x]
        # y -> py: weight[y]
        # x / y = w
        # px / py = w * weight[y] / weight[x]
        self.parent[px] = py
        self.weight[px] = w * self.weight[y] - self.weight[x]
        return True
    
    def query(self, x: int, y: int) -> float:
        """
        查询 x / y 的值
        
        返回:
            float - x/y 的值，不连通返回 -1
        """
        if self.find(x) != self.find(y):
            return -1.0
        return self.weight[x] - self.weight[y]


# 应用：LeetCode 399. Evaluate Division
def calc_equation(equations, values, queries):
    """
    计算除法表达式的值
    
    equations = [["a","b"],["b","c"]]
    values = [2.0, 3.0]  # a/b=2, b/c=3
    queries = [["a","c"]]  # 求 a/c
    """
    # 构建变量到索引的映射
    var_to_idx = {}
    idx = 0
    for a, b in equations:
        if a not in var_to_idx:
            var_to_idx[a] = idx
            idx += 1
        if b not in var_to_idx:
            var_to_idx[b] = idx
            idx += 1
    
    uf = WeightedUnionFind(idx)
    
    for (a, b), val in zip(equations, values):
        uf.union(var_to_idx[a], var_to_idx[b], val)
    
    results = []
    for a, b in queries:
        if a not in var_to_idx or b not in var_to_idx:
            results.append(-1.0)
        else:
            # a / b = weight[a] / weight[b] (相对于同一根)
            pa = uf.find(var_to_idx[a])
            pb = uf.find(var_to_idx[b])
            if pa != pb:
                results.append(-1.0)
            else:
                results.append(uf.weight[var_to_idx[a]] / uf.weight[var_to_idx[b]])
    
    return results
```

# 1. 岛屿数量（二维网格）
def num_islands(grid: list) -> int:
    """
    LeetCode 200. Number of Islands
    """
    if not grid:
        return 0
    
    m, n = len(grid), len(grid[0])
    uf = UnionFind(m * n)
    
    def idx(i, j):
        return i * n + j
    
    land_count = 0
    for i in range(m):
        for j in range(n):
            if grid[i][j] == '1':
                land_count += 1
                # 与相邻陆地合并
                if i > 0 and grid[i-1][j] == '1':
                    if uf.union(idx(i, j), idx(i-1, j)):
                        land_count -= 1
                if j > 0 and grid[i][j-1] == '1':
                    if uf.union(idx(i, j), idx(i, j-1)):
                        land_count -= 1
    
    return land_count

def monotonic_stack_next_greater(nums: list) -> list:
    """
    单调栈：找到每个元素右边第一个更大的元素
    
    参数:
        nums: List[int] - 输入数组
    返回:
        List[int] - 每个位置右边第一个更大元素的索引，不存在为-1
    
    时间复杂度: O(n)
    空间复杂度: O(n)
    
    原理: 维护一个单调递减的栈（存索引），
          遇到更大元素时，弹出栈中所有更小的元素
    """
    n = len(nums)
    result = [-1] * n
    stack = []  # 存储索引，对应元素单调递减
    
    for i in range(n):
        # 当前元素比栈顶大，栈顶元素找到了答案
        while stack and nums[i] > nums[stack[-1]]:
            j = stack.pop()
            result[j] = i
        stack.append(i)
    
    return result

# 变体：找左边第一个更大的元素
def next_greater_left(nums: list) -> list:
    n = len(nums)
    result = [-1] * n
    stack = []
    
    for i in range(n):
        while stack and nums[stack[-1]] <= nums[i]:
            stack.pop()
        if stack:
            result[i] = stack[-1]
        stack.append(i)
    
    return result

# 变体：找右边第一个更小的元素
def next_smaller(nums: list) -> list:
    n = len(nums)
    result = [-1] * n
    stack = []  # 单调递增栈
    
    for i in range(n):
        while stack and nums[i] < nums[stack[-1]]:
            j = stack.pop()
            result[j] = i
        stack.append(i)
    
    return result


# 使用示例

# 1. 柱状图中最大的矩形
def largest_rectangle_area(heights: list) -> int:
    """
    LeetCode 84. Largest Rectangle in Histogram
    
    关键思路: 对于每个柱子，找到左右第一个更矮的柱子，
             计算以当前柱子高度为高的矩形面积
    """
    heights = [0] + heights + [0]  # 哨兵
    stack = []
    max_area = 0
    
    for i, h in enumerate(heights):
        while stack and heights[stack[-1]] > h:
            j = stack.pop()
            height = heights[j]
            width = i - stack[-1] - 1
            max_area = max(max_area, height * width)
        stack.append(i)
    
    return max_area

def daily_temperatures(temperatures: list) -> list:

# 3. 接雨水
def trap(height: list) -> int:
    """
    LeetCode 42. Trapping Rain Water
    
    使用单调栈：当遇到更高的柱子时，可以接水
    """
    stack = []
    water = 0
    
    for i, h in enumerate(height):
        while stack and height[stack[-1]] < h:
            bottom = stack.pop()
            if not stack:
                break
            left = stack[-1]
            width = i - left - 1
            bounded_height = min(height[left], h) - height[bottom]
            water += width * bounded_height
        stack.append(i)
    
    return water

## 3.3 单调队列

```python
from collections import deque

def max_sliding_window(nums: list, k: int) -> list:
    """
    滑动窗口最大值
    
    LeetCode 239. Sliding Window Maximum
    
    参数:
        nums: List[int] - 输入数组
        k: int - 窗口大小
    返回:
        List[int] - 每个窗口的最大值
    
    时间复杂度: O(n)
    空间复杂度: O(k)
    
    原理: 维护一个单调递减的双端队列，
          队首始终是当前窗口的最大值
    """
    dq = deque()  # 存储索引，对应元素单调递减
    result = []
    
    for i, num in enumerate(nums):
        # 移除超出窗口的元素
        while dq and dq[0] < i - k + 1:
            dq.popleft()
        
        # 移除比当前元素小的元素（它们不可能成为最大值）
        while dq and nums[dq[-1]] < num:
            dq.pop()
        
        dq.append(i)
        
        # 窗口形成后开始记录结果
        if i >= k - 1:
            result.append(nums[dq[0]])
    
    return result


# 滑动窗口最小值
def min_sliding_window(nums: list, k: int) -> list:
    """
    滑动窗口最小值（单调递增队列）
    """
    dq = deque()  # 单调递增
    result = []
    
    for i, num in enumerate(nums):
        while dq and dq[0] < i - k + 1:
            dq.popleft()
        
        while dq and nums[dq[-1]] > num:  # 注意：这里是>
            dq.pop()
        
        dq.append(i)
        
        if i >= k - 1:
            result.append(nums[dq[0]])
    
    return result


# 使用示例
nums = [1, 3, -1, -3, 5, 3, 6, 7]
k = 3
print(max_sliding_window(nums, k))  # [3, 3, 5, 5, 6, 7]
```

## 3.4 单调队列优化DP

```python
def constrained_subset_sum(nums: list, k: int) -> int:
    """
    LeetCode 1425. Constrained Subsequence Sum
    
    选择子序列，相邻两个选中元素的索引差不超过k，求最大和
    
    dp[i] = max(dp[j]) + nums[i], 其中 i-k <= j < i
    使用单调队列优化，O(nk) -> O(n)
    """
    n = len(nums)
    dp = nums.copy()
    dq = deque([0])  # 单调递减队列
    
    for i in range(1, n):
        # 移除超出范围的
        while dq and dq[0] < i - k:
            dq.popleft()
        
        # dp[i] = max(0, dp[dq[0]]) + nums[i]
        if dq and dp[dq[0]] > 0:
            dp[i] = dp[dq[0]] + nums[i]
        
        # 维护单调性
        while dq and dp[dq[-1]] <= dp[i]:
            dq.pop()
        dq.append(i)
    
    return max(dp)
```

```python
class SegmentTree:
    """
    线段树：支持区间查询和单点更新
    
    用途:
        - 区间求和/最值查询
        - 单点/区间更新
    
    时间复杂度:
        - 构建: O(n)
        - 查询: O(log n)
        - 更新: O(log n)
    
    空间复杂度: O(n)
    """
    
    def __init__(self, nums: list):
        """
        初始化线段树
        
        参数:
            nums: List[int] - 初始数组
        """
        self.n = len(nums)
        self.tree = [0] * (4 * self.n)  # 4n空间足够
        self._build(nums, 0, 0, self.n - 1)
    
    def _build(self, nums: list, node: int, start: int, end: int) -> None:
        """
        递归构建线段树
        
        参数:
            node: int - 当前节点索引
            start, end: int - 当前节点管辖的区间
        """
        if start == end:
            self.tree[node] = nums[start]
            return
        
        mid = (start + end) // 2
        left, right = 2 * node + 1, 2 * node + 2
        self._build(nums, left, start, mid)
        self._build(nums, right, mid + 1, end)
        self.tree[node] = self.tree[left] + self.tree[right]  # 区间和
    
    def update(self, idx: int, val: int) -> None:
        """
        单点更新
        
        参数:
            idx: int - 要更新的索引
            val: int - 新值
        """
        self._update(0, 0, self.n - 1, idx, val)
    
    def _update(self, node: int, start: int, end: int, idx: int, val: int) -> None:
        if start == end:
            self.tree[node] = val
            return
        
        mid = (start + end) // 2
        left, right = 2 * node + 1, 2 * node + 2
        
        if idx <= mid:
            self._update(left, start, mid, idx, val)
        else:
            self._update(right, mid + 1, end, idx, val)
        
        self.tree[node] = self.tree[left] + self.tree[right]
    
    def query(self, l: int, r: int) -> int:
        """
        区间查询
        
        参数:
            l, r: int - 查询区间 [l, r]
        返回:
            int - 区间和
        """
        return self._query(0, 0, self.n - 1, l, r)
    
    def _query(self, node: int, start: int, end: int, l: int, r: int) -> int:
        if r < start or l > end:
            return 0  # 区间不相交
        
        if l <= start and end <= r:
            return self.tree[node]  # 区间完全包含
        
        mid = (start + end) // 2
        left, right = 2 * node + 1, 2 * node + 2
        return self._query(left, start, mid, l, r) + \
               self._query(right, mid + 1, end, l, r)


# 使用示例
nums = [1, 3, 5, 7, 9, 11]
st = SegmentTree(nums)
print(st.query(1, 3))  # 15 (3+5+7)
st.update(2, 10)       # 将nums[2]改为10
print(st.query(1, 3))  # 20 (3+10+7)
```

## 4.2 带懒标记的线段树

```python
class LazySegmentTree:
    """
    带懒标记的线段树：支持区间更新
    
    用途: 区间加/区间赋值 + 区间查询
    """
    
    def __init__(self, nums: list):
        self.n = len(nums)
        self.tree = [0] * (4 * self.n)
        self.lazy = [0] * (4 * self.n)  # 懒标记
        self._build(nums, 0, 0, self.n - 1)
    
    def _build(self, nums: list, node: int, start: int, end: int) -> None:
        if start == end:
            self.tree[node] = nums[start]
            return
        mid = (start + end) // 2
        left, right = 2 * node + 1, 2 * node + 2
        self._build(nums, left, start, mid)
        self._build(nums, right, mid + 1, end)
        self.tree[node] = self.tree[left] + self.tree[right]
    
    def _push_down(self, node: int, start: int, end: int) -> None:
        """下传懒标记"""
        if self.lazy[node] == 0:
            return
        
        mid = (start + end) // 2
        left, right = 2 * node + 1, 2 * node + 2
        
        # 更新子节点
        self.tree[left] += self.lazy[node] * (mid - start + 1)
        self.tree[right] += self.lazy[node] * (end - mid)
        
        # 传递懒标记
        self.lazy[left] += self.lazy[node]
        self.lazy[right] += self.lazy[node]
        self.lazy[node] = 0
    
    def update_range(self, l: int, r: int, val: int) -> None:
        """区间加"""
        self._update_range(0, 0, self.n - 1, l, r, val)
    
    def _update_range(self, node: int, start: int, end: int, l: int, r: int, val: int) -> None:
        if r < start or l > end:
            return
        
        if l <= start and end <= r:
            self.tree[node] += val * (end - start + 1)
            self.lazy[node] += val
            return
        
        self._push_down(node, start, end)
        
        mid = (start + end) // 2
        left, right = 2 * node + 1, 2 * node + 2
        self._update_range(left, start, mid, l, r, val)
        self._update_range(right, mid + 1, end, l, r, val)
        self.tree[node] = self.tree[left] + self.tree[right]
    
    def query(self, l: int, r: int) -> int:
        return self._query(0, 0, self.n - 1, l, r)
    
    def _query(self, node: int, start: int, end: int, l: int, r: int) -> int:
        if r < start or l > end:
            return 0
        
        if l <= start and end <= r:
            return self.tree[node]
        
        self._push_down(node, start, end)
        
        mid = (start + end) // 2
        left, right = 2 * node + 1, 2 * node + 2
        return self._query(left, start, mid, l, r) + \
               self._query(right, mid + 1, end, l, r)
```

# 五、树状数组

```python
class BinaryIndexedTree:
    """
    树状数组（Binary Indexed Tree / Fenwick Tree）
    
    用途:
        - 单点更新 + 前缀和查询
        - 比线段树更简洁，常数更小
    
    时间复杂度:
        - 更新: O(log n)
        - 查询: O(log n)
    
    空间复杂度: O(n)
    
    核心操作:
        - lowbit(x): 取x的最低位1，即 x & (-x)
    """
    
    def __init__(self, n: int):
        """
        初始化树状数组
        
        参数:
            n: int - 数组大小
        
        注意: 内部使用1-indexed
        """
        self.n = n
        self.tree = [0] * (n + 1)
    
    def update(self, i: int, delta: int) -> None:
        """
        单点更新（增加delta）
        
        参数:
            i: int - 索引（0-indexed）
            delta: int - 增加的值
        """
        i += 1  # 转为1-indexed
        while i <= self.n:
            self.tree[i] += delta
            i += i & (-i)  # i += lowbit(i)
    
    def query(self, i: int) -> int:
        """
        前缀和查询 [0, i]
        
        参数:
            i: int - 索引（0-indexed）
        返回:
            int - 前缀和
        """
        i += 1
        result = 0
        while i > 0:
            result += self.tree[i]
            i -= i & (-i)  # i -= lowbit(i)
        return result
    
    def range_query(self, l: int, r: int) -> int:
        """
        区间查询 [l, r]
        """
        if l == 0:
            return self.query(r)
        return self.query(r) - self.query(l - 1)


# 从数组构建
def build_bit(nums: list) -> BinaryIndexedTree:
    bit = BinaryIndexedTree(len(nums))
    for i, num in enumerate(nums):
        bit.update(i, num)
    return bit


# 应用：逆序对计数
def count_inversions(nums: list) -> int:
    """
    统计数组中的逆序对数量
    
    逆序对：i < j 且 nums[i] > nums[j]
    """
    # 离散化
    sorted_nums = sorted(set(nums))
    rank = {v: i for i, v in enumerate(sorted_nums)}
    
    bit = BinaryIndexedTree(len(sorted_nums))
    count = 0
    
    for num in nums:
        r = rank[num]
        # 统计已插入的、比当前数大的数量
        count += bit.query(len(sorted_nums) - 1) - bit.query(r)
        bit.update(r, 1)
    
    return count
```

## 总结

### 数据结构选择

| 场景 | 推荐数据结构 | 时间复杂度 |
|------|--------------|------------|
| 字符串前缀匹配 | Trie | O(m) |
| 集合合并/查询 | 并查集 | O(α(n)) |
| 找左右更大/更小 | 单调栈 | O(n) |
| 滑动窗口最值 | 单调队列 | O(n) |
| 区间查询+单点更新 | 树状数组 | O(log n) |
| 区间查询+区间更新 | 线段树(懒标记) | O(log n) |

### 关键记忆

1. **Trie**: 子节点用dict，记得标记is_end
2. **并查集**: 路径压缩 + 按秩合并
3. **单调栈**: 找右边第一个更大 → 单调递减栈
4. **单调队列**: 用deque，队首是最值
5. **树状数组**: lowbit = x & (-x)
