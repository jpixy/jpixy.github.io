# 一、random随机模块

## 1.1 基础随机函数

```python
import random

# 随机浮点数 [0.0, 1.0)
random.random()

# 随机浮点数 [a, b]
random.uniform(1.5, 5.5)

# 随机整数 [a, b]（包含两端）
random.randint(1, 10)

# 随机整数 [a, b)（不包含b）
random.randrange(1, 10)
random.randrange(0, 10, 2)  # [0, 2, 4, 6, 8] 中随机

# 从序列中随机选择
random.choice([1, 2, 3, 4, 5])
random.choice('abcdef')

# 随机选择k个（可重复）
random.choices([1, 2, 3, 4, 5], k=3)  # 如 [2, 2, 5]

# 随机选择k个（不重复）
random.sample([1, 2, 3, 4, 5], k=3)  # 如 [3, 1, 5]

# 随机打乱（原地修改）
arr = [1, 2, 3, 4, 5]
random.shuffle(arr)  # arr被打乱

# 打乱但不修改原列表
arr_copy = random.sample(arr, len(arr))
```

## 1.4 算法题应用

```python
import random

# 1. 水塘抽样（Reservoir Sampling）
def reservoir_sample(stream, k):
    """从数据流中等概率抽取k个元素"""
    reservoir = []
    for i, item in enumerate(stream):
        if i < k:
            reservoir.append(item)
        else:
            j = random.randint(0, i)
            if j < k:
                reservoir[j] = item
    return reservoir

# 2. 洗牌算法（Fisher-Yates）
def shuffle(arr):
    """等概率打乱数组"""
    arr = arr.copy()
    n = len(arr)
    for i in range(n - 1, 0, -1):
        j = random.randint(0, i)
        arr[i], arr[j] = arr[j], arr[i]
    return arr

# 3. 随机选择算法（Quick Select）
def quick_select(arr, k):
    """找到第k小的元素"""
    if len(arr) == 1:
        return arr[0]
    
    pivot = random.choice(arr)
    left = [x for x in arr if x < pivot]
    mid = [x for x in arr if x == pivot]
    right = [x for x in arr if x > pivot]
    
    if k <= len(left):
        return quick_select(left, k)
    elif k <= len(left) + len(mid):
        return pivot
    else:
        return quick_select(right, k - len(left) - len(mid))

# 4. 随机化快速排序
def randomized_quicksort(arr):
    if len(arr) <= 1:
        return arr
    
    pivot = random.choice(arr)
    left = [x for x in arr if x < pivot]
    mid = [x for x in arr if x == pivot]
    right = [x for x in arr if x > pivot]
    
    return randomized_quicksort(left) + mid + randomized_quicksort(right)

# 5. 链表随机节点
class RandomNode:
    def __init__(self, head):
        self.head = head
    
    def get_random(self):
        """等概率返回链表中的一个节点值"""
        result = None
        node = self.head
        i = 0
        while node:
            i += 1
            if random.randint(1, i) == 1:
                result = node.val
            node = node.next
        return result

# 6. 生成随机测试数据
def generate_random_array(n, low=0, high=100):
    return [random.randint(low, high) for _ in range(n)]

def generate_random_tree(n):
    """生成随机树的边"""
    edges = []
    for i in range(1, n):
        parent = random.randint(0, i - 1)
        edges.append((parent, i))
    return edges
```

# 二、copy复制模块

## 2.1 浅拷贝与深拷贝

```python
import copy

# 浅拷贝：只复制第一层
original = [[1, 2], [3, 4]]
shallow = copy.copy(original)
shallow = original.copy()         # 等价
shallow = original[:]             # 等价
shallow = list(original)          # 等价

# 修改内层会影响原对象
shallow[0][0] = 100
print(original)  # [[100, 2], [3, 4]]

# 深拷贝：递归复制所有层
original = [[1, 2], [3, 4]]
deep = copy.deepcopy(original)

# 修改不会影响原对象
deep[0][0] = 100
print(original)  # [[1, 2], [3, 4]]
```

## 2.2 常见数据结构的复制

```python
import copy

# 列表
arr = [1, 2, 3]
arr_copy = arr.copy()
arr_copy = arr[:]
arr_copy = list(arr)

# 二维列表（必须深拷贝！）
matrix = [[1, 2], [3, 4]]
matrix_copy = copy.deepcopy(matrix)
# 或手动
matrix_copy = [row.copy() for row in matrix]

# 字典
d = {'a': 1, 'b': [2, 3]}
d_shallow = d.copy()
d_deep = copy.deepcopy(d)

# 集合
s = {1, 2, 3}
s_copy = s.copy()

# 元组（不可变，通常不需要复制）
t = (1, [2, 3])
t_deep = copy.deepcopy(t)  # 内部可变对象需要深拷贝
```

## 2.3 算法题中的复制

```python
import copy

# 回溯时保存状态
def backtrack(path, result):
    if 条件满足:
        result.append(path.copy())  # 必须复制！
        # 或 result.append(path[:])
        return
    
    for choice in choices:
        path.append(choice)
        backtrack(path, result)
        path.pop()

# 分支时复制状态
def solve(state):
    if 终止条件:
        return process(state)
    
    new_state = copy.deepcopy(state)
    # 修改new_state
    return solve(new_state)

# 避免不必要的深拷贝（性能优化）
# 如果只修改第一层，用浅拷贝即可
def process_matrix(matrix):
    # 只需要复制行的引用
    result = matrix.copy()
    result[0] = matrix[0].copy()  # 只深拷贝要修改的行
    result[0][0] = 100
    return result
```

# 三、typing类型提示

## 3.1 基础类型提示

```python
from typing import List, Dict, Set, Tuple, Optional, Union, Any

# 基础类型
def greet(name: str) -> str:
    return f"Hello, {name}"

def add(a: int, b: int) -> int:
    return a + b

# 列表
def sum_list(nums: List[int]) -> int:
    return sum(nums)

# 字典
def count_words(text: str) -> Dict[str, int]:
    words = text.split()
    return {w: words.count(w) for w in set(words)}

# 集合
def unique_elements(arr: List[int]) -> Set[int]:
    return set(arr)

# 元组
def get_range(nums: List[int]) -> Tuple[int, int]:
    return min(nums), max(nums)

# 可选类型（可能为None）
def find_max(nums: List[int]) -> Optional[int]:
    return max(nums) if nums else None

# 联合类型
def process(data: Union[str, int]) -> str:
    return str(data)

# Python 3.10+ 简化语法
def process_v2(data: str | int) -> str:
    return str(data)

def find_max_v2(nums: list[int]) -> int | None:
    return max(nums) if nums else None
```

## 3.2 高级类型

```python
from typing import (
    Callable, Iterator, Generator, Iterable,
    TypeVar, Generic, Protocol
)

# 可调用类型
def apply(func: Callable[[int, int], int], a: int, b: int) -> int:
    return func(a, b)

# 迭代器
def process_items(items: Iterable[int]) -> List[int]:
    return [x * 2 for x in items]

# 生成器
def count_up(n: int) -> Generator[int, None, None]:
    for i in range(n):
        yield i

# 类型变量（泛型）
T = TypeVar('T')

def first(items: List[T]) -> T:
    return items[0]

def last(items: List[T]) -> Optional[T]:
    return items[-1] if items else None

# 泛型类
class Stack(Generic[T]):
    def __init__(self):
        self.items: List[T] = []
    
    def push(self, item: T) -> None:
        self.items.append(item)
    
    def pop(self) -> T:
        return self.items.pop()
```

# 四、dataclasses数据类

## 4.1 基础用法

```python
from dataclasses import dataclass, field
from typing import List

# 基础数据类
@dataclass
class Point:
    x: int
    y: int

p = Point(1, 2)
print(p)           # Point(x=1, y=2)
print(p.x, p.y)    # 1 2

# 默认值
@dataclass
class Student:
    name: str
    age: int = 18
    scores: List[int] = field(default_factory=list)

s1 = Student("Alice")
s2 = Student("Bob", 20, [90, 85])

# 自动生成的方法
# __init__, __repr__, __eq__
```

## 4.2 高级特性

```python
from dataclasses import dataclass, field
from typing import List

# 不可变数据类
@dataclass(frozen=True)
class ImmutablePoint:
    x: int
    y: int

# 可以用作字典键或集合元素
p = ImmutablePoint(1, 2)
d = {p: "point"}
s = {p}

# 排序
@dataclass(order=True)
class Person:
    name: str = field(compare=False)  # 不参与比较
    age: int

people = [Person("Alice", 30), Person("Bob", 25)]
people.sort()  # 按age排序

# 后初始化处理
@dataclass
class Rectangle:
    width: int
    height: int
    area: int = field(init=False)
    
    def __post_init__(self):
        self.area = self.width * self.height

# 继承
@dataclass
class ColoredPoint(Point):
    color: str = "black"
```

## 4.3 算法题应用

```python
from dataclasses import dataclass, field
from typing import List, Tuple
import heapq

# 堆中的元素
@dataclass(order=True)
class HeapItem:
    priority: int
    item: str = field(compare=False)

heap = []
heapq.heappush(heap, HeapItem(3, "task1"))
heapq.heappush(heap, HeapItem(1, "task2"))
top = heapq.heappop(heap)  # HeapItem(priority=1, item='task2')

# 图的边
@dataclass
class Edge:
    u: int
    v: int
    weight: int = 1

# 状态表示
@dataclass(frozen=True)  # frozen使其可哈希
class State:
    position: Tuple[int, int]
    keys: frozenset

visited = set()
state = State((0, 0), frozenset())
visited.add(state)

# Dijkstra状态
@dataclass(order=True)
class DijkstraState:
    distance: int
    node: int = field(compare=False)
```

## 5.3 fractions分数

```python
from fractions import Fraction

# 创建分数
f1 = Fraction(1, 3)
f2 = Fraction(2, 6)  # 自动约分为1/3

# 运算
f1 + Fraction(1, 6)  # Fraction(1, 2)
f1 * 2               # Fraction(2, 3)

# 从浮点数
Fraction(0.5)        # Fraction(1, 2)
Fraction('0.25')     # Fraction(1, 4)

# 获取分子分母
f = Fraction(3, 4)
f.numerator          # 3
f.denominator        # 4

# 算法题：避免浮点精度问题
def slope(p1, p2):
    """计算斜率，返回约分后的(dy, dx)"""
    dy = p2[1] - p1[1]
    dx = p2[0] - p1[0]
    if dx == 0:
        return (1, 0) if dy > 0 else (-1, 0)
    if dy == 0:
        return (0, 1)
    from math import gcd
    g = gcd(abs(dy), abs(dx))
    dy, dx = dy // g, dx // g
    if dx < 0:
        dy, dx = -dy, -dx
    return (dy, dx)
```

## 总结

### 模块速查

| 模块 | 用途 | 常用函数 |
|------|------|----------|
| random | 随机数 | `randint`, `choice`, `shuffle`, `sample` |
| copy | 复制 | `copy`, `deepcopy` |
| typing | 类型提示 | `List`, `Dict`, `Optional`, `Tuple` |
| dataclasses | 数据类 | `@dataclass`, `field` |
| sys | 系统 | `stdin`, `setrecursionlimit`, `maxsize` |

### 关键记忆

1. `random.randint(a, b)`包含两端，`randrange(a, b)`不包含b
2. 二维列表必须用`deepcopy`或列表推导复制
3. `@dataclass(frozen=True)`使对象可哈希
4. `sys.setrecursionlimit(10000)`提高递归限制
5. 精确小数用`Decimal`，分数用`Fraction`
