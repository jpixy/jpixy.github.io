# 一、动态规划基础

## 1.1 DP核心思想

```python
"""
动态规划的核心要素：

1. 状态定义
   - dp[i] 表示什么含义
   - 状态需要能够完整描述问题

2. 状态转移方程
   - dp[i] = f(dp[i-1], dp[i-2], ...)
   - 如何从已知状态推导出新状态

3. 初始条件
   - dp[0], dp[1] 等边界情况

4. 遍历顺序
   - 确保计算dp[i]时，依赖的状态已经计算过

5. 返回值
   - 通常是dp[n]或max(dp)等
"""

# DP vs 递归+记忆化
# 本质相同，只是实现方式不同

# 方法1: 自顶向下（递归+记忆化）
from functools import lru_cache

def fibonacci_memo(n: int) -> int:
    """
    斐波那契数列（记忆化递归）
    
    时间复杂度: O(n)
    空间复杂度: O(n)
    """
    @lru_cache(maxsize=None)
    def fib(n):
        if n <= 1:
            return n
        return fib(n - 1) + fib(n - 2)
    
    return fib(n)


# 方法2: 自底向上（迭代DP）
def fibonacci_dp(n: int) -> int:
    """
    斐波那契数列（迭代DP）
    
    时间复杂度: O(n)
    空间复杂度: O(n)
    """
    if n <= 1:
        return n
    
    dp = [0] * (n + 1)
    dp[0], dp[1] = 0, 1
    
    for i in range(2, n + 1):
        dp[i] = dp[i - 1] + dp[i - 2]
    
    return dp[n]


# 方法3: 空间优化
def fibonacci_opt(n: int) -> int:
    """
    斐波那契数列（空间优化）
    
    时间复杂度: O(n)
    空间复杂度: O(1)
    """
    if n <= 1:
        return n
    
    prev, curr = 0, 1
    for _ in range(2, n + 1):
        prev, curr = curr, prev + curr
    
    return curr
```

```python
# 1. 爬楼梯
def climb_stairs(n: int) -> int:
    """
    LeetCode 70. Climbing Stairs
    
    每次爬1或2阶，有多少种方法爬到第n阶
    
    状态: dp[i] = 爬到第i阶的方法数
    转移: dp[i] = dp[i-1] + dp[i-2]
    """
    if n <= 2:
        return n
    
    prev, curr = 1, 2
    for _ in range(3, n + 1):
        prev, curr = curr, prev + curr
    
    return curr


# 2. 打家劫舍
def rob(nums: list) -> int:
    """
    LeetCode 198. House Robber
    
    不能偷相邻房屋，求最大金额
    
    状态: dp[i] = 偷前i个房屋的最大金额
    转移: dp[i] = max(dp[i-1], dp[i-2] + nums[i])
    """
    if not nums:
        return 0
    if len(nums) == 1:
        return nums[0]
    
    prev2, prev1 = nums[0], max(nums[0], nums[1])
    
    for i in range(2, len(nums)):
        curr = max(prev1, prev2 + nums[i])
        prev2, prev1 = prev1, curr
    
    return prev1


# 3. 最大子数组和
def max_subarray(nums: list) -> int:
    """
    LeetCode 53. Maximum Subarray
    
    Kadane算法
    
    状态: dp[i] = 以nums[i]结尾的最大子数组和
    转移: dp[i] = max(nums[i], dp[i-1] + nums[i])
    """
    max_sum = curr_sum = nums[0]
    
    for num in nums[1:]:
        curr_sum = max(num, curr_sum + num)
        max_sum = max(max_sum, curr_sum)
    
    return max_sum


# 4. 最长递增子序列
def length_of_lis(nums: list) -> int:
    """
    LeetCode 300. Longest Increasing Subsequence
    
    O(n²)解法
    
    状态: dp[i] = 以nums[i]结尾的LIS长度
    转移: dp[i] = max(dp[j] + 1) for all j < i and nums[j] < nums[i]
    """
    if not nums:
        return 0
    
    n = len(nums)
    dp = [1] * n
    
    for i in range(1, n):
        for j in range(i):
            if nums[j] < nums[i]:
                dp[i] = max(dp[i], dp[j] + 1)
    
    return max(dp)


def length_of_lis_optimized(nums: list) -> int:
    """
    O(n log n)解法（贪心 + 二分）
    
    tails[i] = 长度为i+1的LIS的最小结尾元素
    """
    import bisect
    
    tails = []
    
    for num in nums:
        pos = bisect.bisect_left(tails, num)
        if pos == len(tails):
            tails.append(num)
        else:
            tails[pos] = num
    
    return len(tails)


# 5. 零钱兑换
def coin_change(coins: list, amount: int) -> int:
    """
    LeetCode 322. Coin Change
    
    凑成amount需要的最少硬币数
    
    状态: dp[i] = 凑成金额i的最少硬币数
    转移: dp[i] = min(dp[i - coin] + 1) for coin in coins
    """
    dp = [float('inf')] * (amount + 1)
    dp[0] = 0
    
    for i in range(1, amount + 1):
        for coin in coins:
            if coin <= i:
                dp[i] = min(dp[i], dp[i - coin] + 1)
    
    return dp[amount] if dp[amount] != float('inf') else -1
```

```python
# 1. 不同路径
def unique_paths(m: int, n: int) -> int:
    """
    LeetCode 62. Unique Paths
    
    从左上到右下的路径数（只能向右或向下）
    
    状态: dp[i][j] = 到达(i,j)的路径数

            dp[j] += dp[j-1]
    return dp[n-1]

def min_path_sum(grid: list) -> int:
    """
    LeetCode 64. Minimum Path Sum
    
    从左上到右下的最小路径和
    """
    m, n = len(grid), len(grid[0])
    dp = [[0] * n for _ in range(m)]
    
    dp[0][0] = grid[0][0]
    for j in range(1, n):
        dp[0][j] = dp[0][j-1] + grid[0][j]
    for i in range(1, m):
        dp[i][0] = dp[i-1][0] + grid[i][0]
    
    for i in range(1, m):
        for j in range(1, n):
            dp[i][j] = min(dp[i-1][j], dp[i][j-1]) + grid[i][j]
    
    return dp[m-1][n-1]

def longest_common_subsequence(text1: str, text2: str) -> int:

    LeetCode 1143. Longest Common Subsequence
    
    状态: dp[i][j] = text1[0:i]和text2[0:j]的LCS长度
    转移:
        if text1[i-1] == text2[j-1]:
            dp[i][j] = dp[i-1][j-1] + 1
        else:
            dp[i][j] = max(dp[i-1][j], dp[i][j-1])

            else:

# 4. 编辑距离
def min_distance(word1: str, word2: str) -> int:
    """
    LeetCode 72. Edit Distance
    
    将word1转换成word2需要的最少操作数
    
    状态: dp[i][j] = word1[0:i]转换成word2[0:j]的最少操作
    转移:
        if word1[i-1] == word2[j-1]:
            dp[i][j] = dp[i-1][j-1]
        else:
            dp[i][j] = 1 + min(dp[i-1][j], dp[i][j-1], dp[i-1][j-1])
    """
    m, n = len(word1), len(word2)
    dp = [[0] * (n + 1) for _ in range(m + 1)]
    
    # 初始化
    for i in range(m + 1):
        dp[i][0] = i
    for j in range(n + 1):
        dp[0][j] = j
    
    for i in range(1, m + 1):
        for j in range(1, n + 1):
            if word1[i-1] == word2[j-1]:
                dp[i][j] = dp[i-1][j-1]
            else:
                dp[i][j] = 1 + min(
                    dp[i-1][j],    # 删除
                    dp[i][j-1],    # 插入
                    dp[i-1][j-1]   # 替换
                )
    
    return dp[m][n]
```

## 2.1 0-1背包

```python
def knapsack_01(weights: list, values: list, capacity: int) -> int:
    """
    0-1背包问题
    
    参数:
        weights: List[int] - 物品重量
        values: List[int] - 物品价值
        capacity: int - 背包容量
    返回:
        int - 最大价值
    
    状态: dp[i][w] = 前i个物品、容量为w时的最大价值
    转移: dp[i][w] = max(dp[i-1][w], dp[i-1][w-wi] + vi)
    
    时间复杂度: O(n × capacity)
    空间复杂度: O(n × capacity)
    """
    n = len(weights)
    dp = [[0] * (capacity + 1) for _ in range(n + 1)]
    
    for i in range(1, n + 1):
        w, v = weights[i-1], values[i-1]
        for c in range(capacity + 1):
            dp[i][c] = dp[i-1][c]  # 不选
            if c >= w:
                dp[i][c] = max(dp[i][c], dp[i-1][c-w] + v)  # 选
    
    return dp[n][capacity]


def knapsack_01_optimized(weights: list, values: list, capacity: int) -> int:
    """
    0-1背包（空间优化）
    
    关键: 内层循环逆序，确保每个物品只用一次
    
    空间复杂度: O(capacity)
    """
    dp = [0] * (capacity + 1)
    
    for w, v in zip(weights, values):
        for c in range(capacity, w - 1, -1):  # 逆序！
            dp[c] = max(dp[c], dp[c - w] + v)
    
    return dp[capacity]


# 应用：分割等和子集
def can_partition(nums: list) -> bool:
    """
    LeetCode 416. Partition Equal Subset Sum
    
    判断能否分割成两个和相等的子集
    转化为0-1背包：能否恰好装满容量为sum/2的背包
    """
    total = sum(nums)
    if total % 2 != 0:
        return False
    
    target = total // 2
    dp = [False] * (target + 1)
    dp[0] = True
    
    for num in nums:
        for j in range(target, num - 1, -1):
            dp[j] = dp[j] or dp[j - num]
    
    return dp[target]


# 应用：目标和
def find_target_sum_ways(nums: list, target: int) -> int:
    """
    LeetCode 494. Target Sum
    
    每个数可以取+或-，问有多少种方式使和为target
    
    设正数和为P，负数绝对值和为N
    P - N = target
    P + N = sum
    => P = (target + sum) / 2
    
    转化为0-1背包：选择若干数使和为(target+sum)/2的方案数
    """
    total = sum(nums)
    if (total + target) % 2 != 0 or abs(target) > total:
        return 0
    
    bag = (total + target) // 2
    dp = [0] * (bag + 1)
    dp[0] = 1
    
    for num in nums:
        for j in range(bag, num - 1, -1):
            dp[j] += dp[j - num]
    
    return dp[bag]
```

```python
def knapsack_complete(weights: list, values: list, capacity: int) -> int:
    """
    完全背包：每种物品可以无限次选取
    
    状态: dp[c] = 容量为c时的最大价值
    转移: dp[c] = max(dp[c], dp[c-w] + v)
    
    关键: 内层循环正序（与0-1背包相反）
    """
    dp = [0] * (capacity + 1)
    
    for w, v in zip(weights, values):
        for c in range(w, capacity + 1):  # 正序！
            dp[c] = max(dp[c], dp[c - w] + v)
    
    return dp[capacity]


# 应用：零钱兑换II（方案数）
def change(amount: int, coins: list) -> int:
    """
    LeetCode 518. Coin Change 2
    
    用硬币凑成amount的方案数
    """
    dp = [0] * (amount + 1)
    dp[0] = 1
    
    for coin in coins:  # 先遍历物品
        for j in range(coin, amount + 1):  # 再遍历容量
            dp[j] += dp[j - coin]
    
    return dp[amount]


# 应用：完全平方数
def num_squares(n: int) -> int:
    """
    LeetCode 279. Perfect Squares
    
    n由最少几个完全平方数组成
    """
    dp = [float('inf')] * (n + 1)
    dp[0] = 0
    
    for i in range(1, n + 1):
        j = 1
        while j * j <= i:
            dp[i] = min(dp[i], dp[i - j*j] + 1)
            j += 1
    
    return dp[n]


# 应用：单词拆分
def word_break(s: str, word_dict: list) -> bool:
    """
    LeetCode 139. Word Break
    
    判断s能否由字典中的单词组成
    """
    word_set = set(word_dict)
    n = len(s)
    dp = [False] * (n + 1)
    dp[0] = True
    
    for i in range(1, n + 1):
        for j in range(i):
            if dp[j] and s[j:i] in word_set:
                dp[i] = True
                break
    
    return dp[n]
```

## 2.3 多重背包

```python
def knapsack_multiple(weights: list, values: list, counts: list, capacity: int) -> int:
    """
    多重背包：每种物品有限定数量
    
    参数:
        counts: List[int] - 每种物品的数量
    
    方法: 二进制拆分优化
    将数量为c的物品拆分为1, 2, 4, ..., 2^k, 剩余
    然后转化为0-1背包
    """
    # 二进制拆分
    new_weights = []
    new_values = []
    
    for w, v, c in zip(weights, values, counts):
        k = 1
        while k <= c:
            new_weights.append(w * k)
            new_values.append(v * k)
            c -= k
            k *= 2
        if c > 0:
            new_weights.append(w * c)
            new_values.append(v * c)
    
    # 0-1背包
    dp = [0] * (capacity + 1)
    for w, v in zip(new_weights, new_values):
        for c in range(capacity, w - 1, -1):
            dp[c] = max(dp[c], dp[c - w] + v)
    
    return dp[capacity]
```

```python
def interval_dp_template(arr: list):

# 1. 最长回文子序列
def longest_palindrome_subseq(s: str) -> int:
    """
    LeetCode 516. Longest Palindromic Subsequence
    
    状态: dp[i][j] = s[i:j+1]的最长回文子序列长度
    """
    n = len(s)
    dp = [[0] * n for _ in range(n)]
    
    for i in range(n):
        dp[i][i] = 1
    
    for length in range(2, n + 1):
        for i in range(n - length + 1):
            j = i + length - 1
            if s[i] == s[j]:
                dp[i][j] = dp[i+1][j-1] + 2
            else:
                dp[i][j] = max(dp[i+1][j], dp[i][j-1])
    
    return dp[0][n-1]


# 2. 戳气球
def max_coins(nums: list) -> int:
    """
    LeetCode 312. Burst Balloons
    
    戳破气球获得分数，最大化总分
    nums[i-1] * nums[i] * nums[i+1]
    
    关键: 考虑最后一个戳破的气球
    """
    nums = [1] + nums + [1]
    n = len(nums)
    dp = [[0] * n for _ in range(n)]
    
    for length in range(3, n + 1):
        for i in range(n - length + 1):
            j = i + length - 1
            for k in range(i + 1, j):
                # k是最后一个戳破的
                dp[i][j] = max(
                    dp[i][j],
                    dp[i][k] + dp[k][j] + nums[i] * nums[k] * nums[j]
                )
    
    return dp[0][n-1]


# 3. 合并石子
def merge_stones(stones: list, k: int) -> int:
    """
    LeetCode 1000. Minimum Cost to Merge Stones
    
    每次合并k堆相邻石子，求最小代价
    """
    n = len(stones)
    if (n - 1) % (k - 1) != 0:
        return -1
    
    prefix = [0] * (n + 1)
    for i in range(n):
        prefix[i+1] = prefix[i] + stones[i]
    
    dp = [[0] * n for _ in range(n)]
    
    for length in range(k, n + 1):
        for i in range(n - length + 1):
            j = i + length - 1
            dp[i][j] = float('inf')
            for mid in range(i, j, k - 1):
                dp[i][j] = min(dp[i][j], dp[i][mid] + dp[mid+1][j])
            if (length - 1) % (k - 1) == 0:
                dp[i][j] += prefix[j+1] - prefix[i]
    
    return dp[0][n-1]
```

```python
# 1. 旅行商问题（TSP）
def tsp(dist: list) -> int:
    """
    旅行商问题：访问所有城市的最短路径
    
    状态: dp[mask][i] = 访问了mask表示的城市集合，当前在城市i的最短距离
    转移: dp[mask][i] = min(dp[mask^(1<<i)][j] + dist[j][i])
    
    时间复杂度: O(n² × 2^n)
    空间复杂度: O(n × 2^n)
    """
    n = len(dist)
    INF = float('inf')
    
    # dp[mask][i] = 从0出发，访问mask中的城市，当前在i的最短距离
    dp = [[INF] * n for _ in range(1 << n)]
    dp[1][0] = 0  # 起点在0
    
    for mask in range(1 << n):
        for u in range(n):
            if not (mask & (1 << u)):
                continue
            if dp[mask][u] == INF:
                continue
            
            for v in range(n):
                if mask & (1 << v):
                    continue
                new_mask = mask | (1 << v)
                dp[new_mask][v] = min(dp[new_mask][v], dp[mask][u] + dist[u][v])
    
    # 回到起点
    full_mask = (1 << n) - 1
    return min(dp[full_mask][i] + dist[i][0] for i in range(1, n))


# 2. 最短超级串
def shortest_superstring(words: list) -> str:
    """
    LeetCode 943. Find the Shortest Superstring
    
    找出包含所有单词的最短字符串
    """
    n = len(words)
    
    # 预处理重叠长度
    overlap = [[0] * n for _ in range(n)]
    for i in range(n):
        for j in range(n):
            if i != j:
                for k in range(min(len(words[i]), len(words[j])), 0, -1):
                    if words[i][-k:] == words[j][:k]:
                        overlap[i][j] = k
                        break
    
    # dp[mask][i] = 访问mask中的单词，以i结尾的最大重叠
    dp = [[0] * n for _ in range(1 << n)]
    parent = [[-1] * n for _ in range(1 << n)]
    
    for mask in range(1 << n):
        for i in range(n):
            if not (mask & (1 << i)):
                continue
            prev_mask = mask ^ (1 << i)
            if prev_mask == 0:
                continue
            
            for j in range(n):
                if not (prev_mask & (1 << j)):
                    continue
                val = dp[prev_mask][j] + overlap[j][i]
                if val > dp[mask][i]:
                    dp[mask][i] = val
                    parent[mask][i] = j
    
    # 重建答案
    full_mask = (1 << n) - 1
    last = max(range(n), key=lambda i: dp[full_mask][i])
    
    # 回溯路径
    path = []
    mask = full_mask
    while last != -1:
        path.append(last)
        prev = parent[mask][last]
        mask ^= (1 << last)
        last = prev
    path.reverse()
    
    # 构建结果
    result = [words[path[0]]]
    for i in range(1, len(path)):
        ov = overlap[path[i-1]][path[i]]
        result.append(words[path[i]][ov:])
    
    return ''.join(result)
```

## 4.2 棋盘类问题

```python
# 铺砖问题
def domino_tiling(m: int, n: int) -> int:
    """
    用1×2的多米诺骨牌铺满m×n的棋盘
    
    状态压缩：用二进制表示每行的状态
    """
    if m > n:
        m, n = n, m
    
    # 预处理合法的状态转移
    def is_compatible(prev, curr, m):
        """检查prev状态能否转移到curr状态"""
        i = 0
        while i < m:
            if prev & (1 << i):
                if curr & (1 << i):
                    # 都是1：需要竖放
                    pass
                else:
                    # prev是1，curr是0：非法
                    return False
            else:
                if curr & (1 << i):
                    # prev是0，curr是1：竖放
                    pass
                else:
                    # 都是0：横放
                    if i + 1 >= m or (prev & (1 << (i+1))) or (curr & (1 << (i+1))):
                        return False
                    i += 1
            i += 1
        return True
    
    # 枚举所有合法转移
    transitions = {}
    for prev in range(1 << m):
        transitions[prev] = []
        for curr in range(1 << m):
            if is_compatible(prev, curr, m):
                transitions[prev].append(curr)
    
    # DP
    dp = [0] * (1 << m)
    dp[0] = 1
    
    for _ in range(n):
        new_dp = [0] * (1 << m)
        for prev in range(1 << m):
            if dp[prev] == 0:
                continue
            for curr in transitions[prev]:
                new_dp[curr] += dp[prev]
        dp = new_dp
    
    return dp[0]
```

```python
def tree_dp_template(root):

# 1. 打家劫舍III（树形）
def rob_tree(root) -> int:
    """
    LeetCode 337. House Robber III
    
    二叉树上的打家劫舍
    不能同时偷相邻节点（父子）
    
    返回: (不偷当前节点的最大值, 偷当前节点的最大值)
    """
    def dfs(node):
        if not node:
            return (0, 0)
        
        left = dfs(node.left)
        right = dfs(node.right)
        
        # 不偷当前：子节点可偷可不偷
        not_rob = max(left) + max(right)
        # 偷当前：子节点不能偷
        rob = node.val + left[0] + right[0]
        
        return (not_rob, rob)
    
    return max(dfs(root))


# 2. 二叉树的直径
def diameter_of_binary_tree(root) -> int:
    """
    LeetCode 543. Diameter of Binary Tree
    
    经过某节点的最长路径 = 左子树深度 + 右子树深度
    """
    result = 0
    
    def dfs(node):
        nonlocal result
        if not node:
            return 0
        
        left = dfs(node.left)
        right = dfs(node.right)
        
        result = max(result, left + right)
        return max(left, right) + 1
    
    dfs(root)
    return result


# 3. 二叉树的最大路径和
def max_path_sum(root) -> int:
    """
    LeetCode 124. Binary Tree Maximum Path Sum
    
    路径可以从任意节点开始和结束
    """
    result = float('-inf')
    
    def dfs(node):
        nonlocal result
        if not node:
            return 0
        
        # 只取正贡献
        left = max(0, dfs(node.left))
        right = max(0, dfs(node.right))
        
        # 更新最大路径和
        result = max(result, left + right + node.val)
        
        # 返回能贡献给父节点的最大值
        return max(left, right) + node.val
    
    dfs(root)
    return result


# 4. 树的重心
def find_centroid(n: int, edges: list) -> int:
    """
    找树的重心：删除该节点后，最大子树最小
    """
    from collections import defaultdict
    
    graph = defaultdict(list)
    for u, v in edges:
        graph[u].append(v)
        graph[v].append(u)
    
    result = -1
    min_max_subtree = float('inf')
    
    def dfs(node, parent):
        nonlocal result, min_max_subtree
        
        subtree_size = 1
        max_subtree = 0
        
        for child in graph[node]:
            if child != parent:
                child_size = dfs(child, node)
                subtree_size += child_size
                max_subtree = max(max_subtree, child_size)
        
        # 向上的"子树"大小
        max_subtree = max(max_subtree, n - subtree_size)
        
        if max_subtree < min_max_subtree:
            min_max_subtree = max_subtree
            result = node
        
        return subtree_size
    
    dfs(0, -1)
    return result
```

## 总结

### DP类型速查

| 类型 | 状态定义 | 典型问题 |
|------|----------|----------|
| 线性DP | dp[i] | LIS, 打家劫舍 |
| 二维DP | dp[i][j] | LCS, 编辑距离 |
| 背包DP | dp[i][w] | 0-1背包, 完全背包 |
| 区间DP | dp[l][r] | 戳气球, 合并石子 |
| 状压DP | dp[mask] | TSP, 铺砖 |
| 树形DP | dfs返回值 | 树的直径, 最大路径和 |

### 背包问题速查

| 类型 | 内层循环 | 特点 |
|------|----------|------|
| 0-1背包 | 逆序 | 每个物品最多选1次 |
| 完全背包 | 正序 | 每个物品可选无限次 |
| 多重背包 | 二进制拆分 | 每个物品有数量限制 |

### 关键记忆

1. 0-1背包逆序，完全背包正序
2. 区间DP先枚举长度
3. 状压DP用位运算操作集合
4. 树形DP用后序遍历
5. 空间优化看能否只依赖上一行/上一个状态
