```python
graph = defaultdict(list)

    graph[u].append(v)

    graph[u].append((v, weight))
    graph[v].append((u, weight))  # 无向图

    """
    参数:
        n: int - 节点数量（0到n-1）
        edges: List[Tuple[int, int]] - 边列表 [(u, v), ...]
        directed: bool - 是否有向图，默认False
    返回:
        defaultdict(list) - 邻接表
    """

    for u, v in edges:
        graph[u].append(v)
        if not directed:
            graph[v].append(u)

    参数:

        graph[u].append((v, w))
        if not directed:
            graph[v].append((u, w))
```

# 二、图的遍历

## 2.1 BFS广度优先搜索

```python
from collections import deque

def bfs(graph, start):
    """
    广度优先搜索
    
    参数:
        graph: Dict[int, List[int]] - 邻接表
        start: int - 起始节点
    返回:
        Set[int] - 访问过的节点集合
    
    时间复杂度: O(V + E)
    空间复杂度: O(V)
    """
    visited = {start}
    queue = deque([start])
    
    while queue:
        node = queue.popleft()  # O(1) 这是用deque的原因
        # 处理当前节点
        print(node)
        
        for neighbor in graph[node]:
            if neighbor not in visited:
                visited.add(neighbor)
                queue.append(neighbor)
    
    return visited

# BFS求最短路径（无权图）
def bfs_shortest_path(graph, start, end):
    """
    BFS求最短路径距离（无权图）
    
    参数:
        graph: 邻接表
        start: 起点
        end: 终点
    返回:
        int - 最短距离，不可达返回-1
    
    时间复杂度: O(V + E)
    """
    if start == end:
        return 0
    
    visited = {start}
    queue = deque([(start, 0)])  # (节点, 距离)
    
    while queue:
        node, dist = queue.popleft()
        
        for neighbor in graph[node]:
            if neighbor == end:
                return dist + 1
            if neighbor not in visited:
                visited.add(neighbor)
                queue.append((neighbor, dist + 1))
    
    return -1

# BFS求最短路径（返回路径）
def bfs_path(graph, start, end):
    """
    BFS求最短路径（返回完整路径）
    
    返回:
        List[int] - 路径节点列表，不可达返回空列表
    """
    if start == end:
        return [start]
    
    visited = {start}
    queue = deque([(start, [start])])  # (节点, 路径)
    
    while queue:
        node, path = queue.popleft()
        
        for neighbor in graph[node]:
            if neighbor == end:
                return path + [neighbor]
            if neighbor not in visited:
                visited.add(neighbor)
                queue.append((neighbor, path + [neighbor]))
    
    return []

# 层序BFS（按层处理）
def bfs_by_level(graph, start):
    """
    按层遍历，返回每层的节点
    """
    visited = {start}
    queue = deque([start])
    levels = []
    
    while queue:
        level_size = len(queue)
        current_level = []
        
        for _ in range(level_size):
            node = queue.popleft()
            current_level.append(node)
            
            for neighbor in graph[node]:
                if neighbor not in visited:
                    visited.add(neighbor)
                    queue.append(neighbor)
        
        levels.append(current_level)
    
    return levels
```

```python
# DFS递归版本
def dfs_recursive(graph, start, visited=None):
    """
    深度优先搜索（递归）
    
    参数:
        graph: Dict[int, List[int]] - 邻接表
        start: int - 起始节点
        visited: Set[int] - 已访问节点集合
    返回:
        Set[int] - 访问过的节点
    
    时间复杂度: O(V + E)
    空间复杂度: O(V) 递归栈
    """
    if visited is None:
        visited = set()
    
    visited.add(start)
    print(start)  # 处理当前节点
    
    for neighbor in graph[start]:
        if neighbor not in visited:
            dfs_recursive(graph, neighbor, visited)
    
    return visited

# DFS迭代版本（使用栈）
def dfs_iterative(graph, start):
    """
    深度优先搜索（迭代）
    使用栈模拟递归
    """
    visited = set()
    stack = [start]
    
    while stack:
        node = stack.pop()
        if node in visited:
            continue
        
        visited.add(node)
        print(node)
        
        # 逆序添加邻居，保持与递归相同的遍历顺序
        for neighbor in reversed(graph[node]):
            if neighbor not in visited:
                stack.append(neighbor)
    
    return visited

# DFS求路径
def dfs_path(graph, start, end, path=None, visited=None):
    """
    DFS求从start到end的路径
    
    返回:
        List[int] - 路径，不存在返回None
    """
    if path is None:
        path = []
    if visited is None:
        visited = set()
    
    path = path + [start]
    visited.add(start)
    
    if start == end:
        return path
    
    for neighbor in graph[start]:
        if neighbor not in visited:
            result = dfs_path(graph, neighbor, end, path, visited)
            if result:
                return result
    
    return None

# DFS求所有路径
def dfs_all_paths(graph, start, end):
    """
    DFS求从start到end的所有路径
    """
    all_paths = []
    
    def dfs(node, path):
        if node == end:
            all_paths.append(path[:])
            return
        
        for neighbor in graph[node]:
            if neighbor not in path:
                path.append(neighbor)
                dfs(neighbor, path)
                path.pop()
    
    dfs(start, [start])
    return all_paths
```

## 2.3 双向BFS

```python
from collections import deque

def bidirectional_bfs(graph, start, end):
    """
    双向BFS求最短路径
    
    从起点和终点同时搜索，在中间相遇
    适用于起点终点明确的最短路问题
    
    时间复杂度: O(b^(d/2))，b是分支因子，d是深度
    比单向BFS快很多
    """
    if start == end:
        return 0
    
    # 两个方向的已访问集合
    front = {start}
    back = {end}
    visited = {start, end}
    dist = 0
    
    while front and back:
        dist += 1
        
        # 总是扩展较小的集合（优化）
        if len(front) > len(back):
            front, back = back, front
        
        next_front = set()
        for node in front:
            for neighbor in graph[node]:
                if neighbor in back:
                    return dist  # 相遇
                if neighbor not in visited:
                    visited.add(neighbor)
                    next_front.add(neighbor)
        
        front = next_front
    
    return -1  # 不可达
```

# 三、最短路径算法

## 3.1 Dijkstra算法

```python
import heapq
from collections import defaultdict

def dijkstra(graph, start):
    """
    Dijkstra单源最短路径算法
    
    参数:
        graph: Dict[int, List[Tuple[int, int]]] - 邻接表 {u: [(v, weight), ...]}
        start: int - 起点
    返回:
        Dict[int, int] - {节点: 到起点的最短距离}
    
    适用条件: 边权非负
    时间复杂度: O((V + E) log V)
    空间复杂度: O(V)
    """
    # 距离字典
    dist = {start: 0}
    # 优先队列 (距离, 节点)
    heap = [(0, start)]
    
    while heap:
        d, node = heapq.heappop(heap)
        
        # 跳过过期的条目（懒删除）
        if d > dist.get(node, float('inf')):
            continue
        
        for neighbor, weight in graph[node]:
            new_dist = d + weight
            if new_dist < dist.get(neighbor, float('inf')):
                dist[neighbor] = new_dist
                heapq.heappush(heap, (new_dist, neighbor))
    
    return dist

def dijkstra_with_path(graph, start, end):
    """
    Dijkstra算法 + 路径记录
    
    返回:
        Tuple[int, List[int]] - (最短距离, 路径)
    """
    dist = {start: 0}
    prev = {start: None}  # 记录前驱节点
    heap = [(0, start)]
    
    while heap:
        d, node = heapq.heappop(heap)
        
        if node == end:
            # 重建路径
            path = []
            curr = end
            while curr is not None:
                path.append(curr)
                curr = prev[curr]
            return d, path[::-1]
        
        if d > dist.get(node, float('inf')):
            continue
        
        for neighbor, weight in graph[node]:
            new_dist = d + weight
            if new_dist < dist.get(neighbor, float('inf')):
                dist[neighbor] = new_dist
                prev[neighbor] = node
                heapq.heappush(heap, (new_dist, neighbor))
    
    return float('inf'), []

# 使用示例
graph = defaultdict(list)
edges = [(0, 1, 4), (0, 2, 1), (2, 1, 2), (1, 3, 1), (2, 3, 5)]
for u, v, w in edges:
    graph[u].append((v, w))
    graph[v].append((u, w))  # 无向图

dist = dijkstra(graph, 0)
# dist = {0: 0, 2: 1, 1: 3, 3: 4}
```

## 3.2 Bellman-Ford算法

```python
def bellman_ford(n, edges, start):
    """
    Bellman-Ford单源最短路径算法
    
    参数:
        n: int - 节点数量（0到n-1）
        edges: List[Tuple[int, int, int]] - 边列表 [(u, v, weight), ...]
        start: int - 起点
    返回:
        Tuple[Dict[int, int], bool] - (距离字典, 是否存在负环)
    
    适用条件: 可以处理负权边
    时间复杂度: O(V * E)
    空间复杂度: O(V)
    """
    INF = float('inf')
    dist = {i: INF for i in range(n)}
    dist[start] = 0
    
    # 松弛n-1次
    for _ in range(n - 1):
        updated = False
        for u, v, w in edges:
            if dist[u] != INF and dist[u] + w < dist[v]:
                dist[v] = dist[u] + w
                updated = True
        if not updated:  # 提前退出优化
            break
    
    # 检测负环（第n次还能松弛说明有负环）
    has_negative_cycle = False
    for u, v, w in edges:
        if dist[u] != INF and dist[u] + w < dist[v]:
            has_negative_cycle = True
            break
    
    return dist, has_negative_cycle

# 使用示例
edges = [(0, 1, 4), (0, 2, 1), (2, 1, -2), (1, 3, 1)]
dist, has_cycle = bellman_ford(4, edges, 0)
# dist = {0: 0, 1: -1, 2: 1, 3: 0}, has_cycle = False
```

## 3.3 SPFA算法

```python
from collections import deque, defaultdict

def spfa(graph, n, start):
    """
    SPFA (Shortest Path Faster Algorithm)
    Bellman-Ford的队列优化版本
    
    参数:
        graph: Dict[int, List[Tuple[int, int]]] - 邻接表
        n: int - 节点数
        start: int - 起点
    返回:
        Tuple[Dict[int, int], bool] - (距离, 是否有负环)
    
    时间复杂度: 平均O(kE)，最坏O(VE)
    """
    INF = float('inf')
    dist = {i: INF for i in range(n)}
    dist[start] = 0
    
    in_queue = {start}
    queue = deque([start])
    count = {i: 0 for i in range(n)}  # 入队次数
    count[start] = 1
    
    while queue:
        u = queue.popleft()
        in_queue.discard(u)
        
        for v, w in graph[u]:
            if dist[u] + w < dist[v]:
                dist[v] = dist[u] + w
                if v not in in_queue:
                    queue.append(v)
                    in_queue.add(v)
                    count[v] += 1
                    if count[v] >= n:  # 负环检测
                        return dist, True
    
    return dist, False
```

## 3.4 Floyd-Warshall算法

```python
def floyd_warshall(n, edges):
    """
    Floyd-Warshall全源最短路径算法
    
    参数:
        n: int - 节点数量
        edges: List[Tuple[int, int, int]] - 边列表
    返回:
        List[List[int]] - n×n距离矩阵
    
    适用场景: 求所有点对之间的最短路
    时间复杂度: O(V³)
    空间复杂度: O(V²)
    """
    INF = float('inf')
    
    # 初始化距离矩阵
    dist = [[INF] * n for _ in range(n)]
    for i in range(n):
        dist[i][i] = 0
    for u, v, w in edges:
        dist[u][v] = min(dist[u][v], w)  # 处理重边
    
    # 动态规划
    for k in range(n):      # 中间节点
        for i in range(n):  # 起点
            for j in range(n):  # 终点
                if dist[i][k] != INF and dist[k][j] != INF:
                    dist[i][j] = min(dist[i][j], dist[i][k] + dist[k][j])
    
    return dist

# 检测负环
def floyd_has_negative_cycle(dist):
    """对角线出现负数说明有负环"""
    for i in range(len(dist)):
        if dist[i][i] < 0:
            return True
    return False

# 使用示例
edges = [(0, 1, 3), (0, 2, 8), (1, 2, 2), (2, 0, 5)]
dist = floyd_warshall(3, edges)
# dist[i][j] 表示从i到j的最短距离
```

## 3.5 0-1 BFS

```python
from collections import deque

def bfs_01(graph, start, n):
    """
    0-1 BFS：边权只有0和1时的最短路
    
    参数:
        graph: Dict[int, List[Tuple[int, int]]] - {u: [(v, w), ...]}，w只能是0或1
        start: int - 起点
        n: int - 节点数
    返回:
        List[int] - 到各节点的最短距离
    
    时间复杂度: O(V + E)
    关键: 权重为0加到队首，权重为1加到队尾
    """
    INF = float('inf')
    dist = [INF] * n
    dist[start] = 0
    dq = deque([start])
    
    while dq:
        u = dq.popleft()
        for v, w in graph[u]:
            if dist[u] + w < dist[v]:
                dist[v] = dist[u] + w
                if w == 0:
                    dq.appendleft(v)  # 权重0加到队首
                else:
                    dq.append(v)      # 权重1加到队尾
    
    return dist
```

# 四、最小生成树

## 4.1 Kruskal算法

```python
class UnionFind:
    """
    并查集（用于Kruskal算法）
    
    方法:
        find(x): 查找x的根节点
        union(x, y): 合并x和y所在的集合
    
    时间复杂度: 近乎O(1)（阿克曼函数的反函数）
    """
    def __init__(self, n):
        self.parent = list(range(n))
        self.rank = [0] * n
    
    def find(self, x):
        if self.parent[x] != x:
            self.parent[x] = self.find(self.parent[x])  # 路径压缩
        return self.parent[x]
    
    def union(self, x, y):
        px, py = self.find(x), self.find(y)
        if px == py:
            return False
        # 按秩合并
        if self.rank[px] < self.rank[py]:
            px, py = py, px
        self.parent[py] = px
        if self.rank[px] == self.rank[py]:
            self.rank[px] += 1
        return True

def kruskal(n, edges):
    """
    Kruskal最小生成树算法
    
    参数:
        n: int - 节点数量
        edges: List[Tuple[int, int, int]] - 边列表 [(u, v, weight), ...]
    返回:
        Tuple[int, List[Tuple[int, int, int]]] - (MST总权重, MST的边)
    
    算法思想: 贪心，每次选最小的边，不形成环就加入
    时间复杂度: O(E log E)
    """
    # 按权重排序
    edges = sorted(edges, key=lambda e: e[2])
    
    uf = UnionFind(n)
    mst_weight = 0
    mst_edges = []
    
    for u, v, w in edges:
        if uf.union(u, v):  # 不形成环
            mst_weight += w
            mst_edges.append((u, v, w))
            if len(mst_edges) == n - 1:  # MST有n-1条边
                break
    
    # 检查是否连通
    if len(mst_edges) < n - 1:
        return -1, []  # 图不连通
    
    return mst_weight, mst_edges

# 使用示例
edges = [(0, 1, 4), (0, 2, 3), (1, 2, 1), (1, 3, 2), (2, 3, 4)]
weight, mst = kruskal(4, edges)
# weight = 6, mst = [(1, 2, 1), (1, 3, 2), (0, 2, 3)]
```

## 4.2 Prim算法

```python
import heapq
from collections import defaultdict

def prim(graph, n):
    """
    Prim最小生成树算法
    
    参数:
        graph: Dict[int, List[Tuple[int, int]]] - 邻接表 {u: [(v, w), ...]}
        n: int - 节点数
    返回:
        Tuple[int, List[Tuple[int, int, int]]] - (MST总权重, MST的边)
    
    算法思想: 从一个点开始，每次加入到当前树距离最近的点
    时间复杂度: O(E log V)
    """
    visited = [False] * n
    # 优先队列: (权重, 当前节点, 前驱节点)
    heap = [(0, 0, -1)]
    mst_weight = 0
    mst_edges = []
    
    while heap and len(mst_edges) < n:
        w, u, prev = heapq.heappop(heap)
        
        if visited[u]:
            continue
        
        visited[u] = True
        mst_weight += w
        if prev != -1:
            mst_edges.append((prev, u, w))
        
        for v, weight in graph[u]:
            if not visited[v]:
                heapq.heappush(heap, (weight, v, u))
    
    if len(mst_edges) < n - 1:
        return -1, []  # 图不连通
    
    return mst_weight, mst_edges
```

# 五、拓扑排序

## 5.1 Kahn算法（BFS）

```python
from collections import deque, defaultdict

def topological_sort_kahn(n, edges):
    """
    拓扑排序 - Kahn算法（BFS）
    
    参数:
        n: int - 节点数量（0到n-1）
        edges: List[Tuple[int, int]] - 边列表 [(from, to), ...]
    返回:
        List[int] - 拓扑序列，存在环返回空列表
    
    时间复杂度: O(V + E)
    """
    # 建图并计算入度
    graph = defaultdict(list)
    indegree = [0] * n
    
    for u, v in edges:
        graph[u].append(v)
        indegree[v] += 1
    
    # 入度为0的节点入队
    queue = deque([i for i in range(n) if indegree[i] == 0])
    result = []
    
    while queue:
        node = queue.popleft()
        result.append(node)
        
        for neighbor in graph[node]:
            indegree[neighbor] -= 1
            if indegree[neighbor] == 0:
                queue.append(neighbor)
    
    # 如果结果数量不等于节点数，说明有环
    return result if len(result) == n else []

# 判断是否存在环
def has_cycle(n, edges):
    return len(topological_sort_kahn(n, edges)) == 0
```

## 5.2 DFS拓扑排序

```python
def topological_sort_dfs(n, edges):
    """
    拓扑排序 - DFS版本
    
    利用DFS后序遍历的逆序得到拓扑序
    """
    graph = defaultdict(list)
    for u, v in edges:
        graph[u].append(v)
    
    WHITE, GRAY, BLACK = 0, 1, 2
    color = [WHITE] * n
    result = []
    has_cycle = False
    
    def dfs(node):
        nonlocal has_cycle
        color[node] = GRAY  # 正在访问
        
        for neighbor in graph[node]:
            if color[neighbor] == GRAY:  # 发现环
                has_cycle = True
                return
            if color[neighbor] == WHITE:
                dfs(neighbor)
                if has_cycle:
                    return
        
        color[node] = BLACK  # 访问完成
        result.append(node)
    
    for i in range(n):
        if color[i] == WHITE:
            dfs(i)
            if has_cycle:
                return []
    
    return result[::-1]  # 逆序即为拓扑序
```

## 6.1 二分图判定

```python
from collections import deque

def is_bipartite(graph, n):
    """
    判断图是否为二分图（BFS染色法）
    
    参数:
        graph: Dict[int, List[int]] - 邻接表
        n: int - 节点数
    返回:
        bool - 是否为二分图
    
    二分图定义: 可以将所有节点分成两个集合，使得所有边都连接两个集合中的节点
    """
    color = [-1] * n  # -1表示未染色
    
    for start in range(n):
        if color[start] != -1:
            continue
        
        # BFS染色
        queue = deque([start])
        color[start] = 0
        
        while queue:
            node = queue.popleft()
            for neighbor in graph[node]:
                if color[neighbor] == -1:
                    color[neighbor] = 1 - color[node]  # 染相反颜色
                    queue.append(neighbor)
                elif color[neighbor] == color[node]:
                    return False  # 相邻节点同色，不是二分图
    
    return True
```

```python
def count_components(n, edges):
    """
    计算无向图的连通分量数量
    
    方法1: DFS
    """
    graph = defaultdict(list)
    for u, v in edges:
        graph[u].append(v)
        graph[v].append(u)
    
    visited = [False] * n
    count = 0
    
    def dfs(node):
        visited[node] = True
        for neighbor in graph[node]:
            if not visited[neighbor]:
                dfs(neighbor)
    
    for i in range(n):
        if not visited[i]:
            dfs(i)
            count += 1
    
    return count
```

## 6.3 强连通分量（有向图）

```python
def kosaraju(n, edges):
    """
    Kosaraju算法求强连通分量
    
    参数:
        n: int - 节点数
        edges: List[Tuple[int, int]] - 有向边
    返回:
        List[List[int]] - 各个强连通分量
    
    时间复杂度: O(V + E)
    """
    # 建图和反向图
    graph = defaultdict(list)
    reverse_graph = defaultdict(list)
    for u, v in edges:
        graph[u].append(v)
        reverse_graph[v].append(u)
    
    # 第一次DFS，得到完成时间顺序
    visited = [False] * n
    finish_order = []
    
    def dfs1(node):
        visited[node] = True
        for neighbor in graph[node]:
            if not visited[neighbor]:
                dfs1(neighbor)
        finish_order.append(node)
    
    for i in range(n):
        if not visited[i]:
            dfs1(i)
    
    # 第二次DFS，在反向图上按完成时间逆序访问
    visited = [False] * n
    sccs = []
    
    def dfs2(node, scc):
        visited[node] = True
        scc.append(node)
        for neighbor in reverse_graph[node]:
            if not visited[neighbor]:
                dfs2(neighbor, scc)
    
    for node in reversed(finish_order):
        if not visited[node]:
            scc = []
            dfs2(node, scc)
            sccs.append(scc)
    
    return sccs
```

## 总结

### 算法选择指南

| 问题 | 推荐算法 | 时间复杂度 |
|------|----------|------------|
| 无权图最短路 | BFS | O(V + E) |
| 非负权图最短路 | Dijkstra | O((V + E) log V) |
| 负权图最短路 | Bellman-Ford / SPFA | O(VE) |
| 全源最短路 | Floyd-Warshall | O(V³) |
| 0-1权图最短路 | 0-1 BFS | O(V + E) |
| 最小生成树 | Kruskal / Prim | O(E log E) |
| 拓扑排序 | Kahn (BFS) | O(V + E) |
| 连通分量 | DFS / 并查集 | O(V + E) |

### 关键记忆

1. BFS用deque，Dijkstra用heapq
2. Dijkstra不能处理负权边，用Bellman-Ford
3. Kruskal需要并查集，按边权排序
4. 拓扑排序检测环：结果数量 ≠ 节点数
5. 二分图判定：BFS染色法
