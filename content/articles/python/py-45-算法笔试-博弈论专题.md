# 一、博弈论基础

## 1.1 基本概念

```python
"""
博弈论基本术语:

1. 博弈状态（Game State）
   - 当前游戏的局面

2. 必胜态（N-position, Winning Position）
   - 先手必胜的状态
   - 存在至少一个后继状态是必败态

3. 必败态（P-position, Losing Position）
   - 先手必败的状态
   - 所有后继状态都是必胜态

4. 公平博弈（Impartial Game）
   - 双方可选操作相同
   - 如：Nim游戏

5. 非公平博弈（Partizan Game）
   - 双方可选操作不同
   - 如：国际象棋

判断规则:
- 无法操作的状态是必败态
- 能转移到必败态的状态是必胜态
- 只能转移到必胜态的状态是必败态
"""
```

## 1.2 简单博弈分析

```python
def can_win_nim_simple(n: int) -> bool:
    """
    简单Nim游戏
    
    规则: 一堆n个石子，每次可以取1-3个，取走最后一个获胜
    
    分析:
        n=1,2,3: 先手取完，必胜
        n=4: 无论取多少，后手都能取完，必败
        n=5,6,7: 先手取到剩4个，必胜
        n=8: 必败
        ...
    
    结论: n是4的倍数时必败，否则必胜
    """
    return n % 4 != 0


def can_win_bash(n: int, m: int) -> bool:
    """
    Bash博弈
    
    规则: 一堆n个石子，每次可以取1-m个，取走最后一个获胜
    
    结论: n是(m+1)的倍数时必败，否则必胜
    
    证明:
        - n <= m: 先手取完，必胜
        - n = m+1: 无论取k个(1<=k<=m)，后手取m+1-k个，必败
        - n = 2(m+1): 先手取k，后手取m+1-k，又回到m+1的情况
    """
    return n % (m + 1) != 0


def can_win_take_last_loses(n: int, m: int) -> bool:
    """
    反Bash博弈
    
    规则: 一堆n个石子，每次可以取1-m个，取走最后一个失败
    
    结论: n是(m+1)的倍数时必胜，否则必败
    """
    return n % (m + 1) == 0
```

# 二、Nim游戏

## 2.1 经典Nim游戏

```python
def nim_game(piles: list) -> bool:
    """
    Nim游戏
    
    LeetCode 292 变体
    
    规则: 
        - 有n堆石子，每堆有若干个
        - 两人轮流操作，每次从一堆中取任意正数个
        - 取走最后一个石子的人获胜
    
    必胜条件: 所有堆的异或和不为0
    
    证明（Bouton定理）:
        1. 终态（全0）异或和为0，是必败态
        2. 异或和为0时，任何操作都会使异或和变为非0
        3. 异或和非0时，存在操作使异或和变为0
    
    时间复杂度: O(n)
    """
    xor_sum = 0
    for pile in piles:
        xor_sum ^= pile
    return xor_sum != 0


def nim_game_strategy(piles: list) -> tuple:
    """
    Nim游戏策略
    
    返回应该从哪堆取走多少个
    
    返回:
        (pile_index, take_count) 或 None（必败态）
    """
    xor_sum = 0
    for pile in piles:
        xor_sum ^= pile
    
    if xor_sum == 0:
        return None  # 必败态
    
    # 找一堆使得取走后异或和为0
    for i, pile in enumerate(piles):
        target = pile ^ xor_sum
        if target < pile:
            return (i, pile - target)
    
    return None


# 示例
piles = [3, 4, 5]
# 异或: 3^4^5 = 011^100^101 = 010 = 2 != 0
# 先手必胜
print(nim_game(piles))  # True
print(nim_game_strategy(piles))  # (0, 2) 或其他有效策略
```

## 2.2 Nim游戏变种

```python
def misere_nim(piles: list) -> bool:
    """
    Anti-Nim游戏（取走最后一个的人输）
    
    规则: 取走最后一个石子的人失败
    
    必胜条件:
        1. 所有堆都只有1个时，堆数为偶数
        2. 存在堆>1时，异或和不为0
    """
    xor_sum = 0
    all_ones = True
    
    for pile in piles:
        xor_sum ^= pile
        if pile > 1:
            all_ones = False
    
    if all_ones:
        return len(piles) % 2 == 0
    else:
        return xor_sum != 0


def staircase_nim(piles: list) -> bool:
    """
    阶梯Nim游戏
    
    规则: 
        - 石子排成一排，从1到n编号
        - 每次可以将某堆的若干石子移到前一堆
        - 不能移动第0堆的石子
        - 无法移动的人输
    
    必胜条件: 奇数位置的异或和不为0
    
    原理: 偶数位置的移动可以被对手复制到奇数位置
    """
    xor_sum = 0
    for i in range(0, len(piles), 2):  # 奇数位置（0索引的偶数位）
        xor_sum ^= piles[i]
    return xor_sum != 0


def multi_nim(piles: list, k: int) -> bool:
    """
    多堆Nim（一次可以从最多k堆取）
    
    规则: 每次可以从1到k堆中各取任意正数个
    
    这是更复杂的变种，需要用SG函数分析
    """
    # k=1时是经典Nim
    if k == 1:
        return nim_game(piles)
    
    # k >= n时，只要有石子就必胜
    if k >= len(piles):
        return any(p > 0 for p in piles)
    
    # 一般情况需要更复杂的分析
    # 这里给出k=1的简化处理
    return nim_game(piles)
```

# 三、SG函数

## 3.1 Sprague-Grundy定理

```python
"""
Sprague-Grundy定理:

1. 定义
   SG(x) = mex({SG(y) | y是x的后继状态})
   mex(S) = 不在集合S中的最小非负整数

2. 性质
   - SG(x) = 0 当且仅当 x 是必败态
   - 多个独立游戏的SG值等于各游戏SG值的异或

3. 应用
   将复杂博弈分解为简单子博弈，分别计算SG值后异或
"""


def mex(s: set) -> int:
    """
    计算mex（最小排斥函数）
    
    参数:
        s: set - 整数集合
    返回:
        int - 不在s中的最小非负整数
    """
    i = 0
    while i in s:
        i += 1
    return i


def sg_value(n: int, moves: list, memo: dict = None) -> int:
    """
    计算SG值
    
    参数:
        n: int - 当前状态（如石子数）
        moves: list - 可选的移动（如每次可以取的数量）
        memo: dict - 记忆化
    返回:
        int - SG值
    
    示例:
        moves = [1, 2, 3] 表示每次可以取1、2或3个
    """
    if memo is None:
        memo = {}
    
    if n in memo:
        return memo[n]
    
    if n == 0:
        memo[n] = 0
        return 0
    
    reachable = set()
    for m in moves:
        if m <= n:
            reachable.add(sg_value(n - m, moves, memo))
    
    memo[n] = mex(reachable)
    return memo[n]


# 示例：经典Nim（一堆，每次取1-3个）
moves = [1, 2, 3]
for n in range(10):
    print(f"SG({n}) = {sg_value(n, moves, {})}")
# 输出: SG(0)=0, SG(1)=1, SG(2)=2, SG(3)=3, SG(4)=0, ...
# 周期为4，与 n%4 一致
```

```python
def combined_game_winner(games: list) -> bool:
    """
    组合博弈判断
    
    参数:
        games: list - 每个子游戏的SG值
    返回:
        bool - 先手是否必胜
    
    原理: 多个独立游戏的SG值异或
    """
    xor_sum = 0
    for sg in games:
        xor_sum ^= sg
    return xor_sum != 0


def nim_with_limit(piles: list, limit: int) -> bool:
    """
    限制Nim游戏
    
    规则: 每次最多取limit个石子
    
    分析: 每堆的SG值为 pile % (limit + 1)
    """
    xor_sum = 0
    for pile in piles:
        xor_sum ^= (pile % (limit + 1))
    return xor_sum != 0


def fibonacci_nim(n: int) -> bool:
    """
    斐波那契Nim（Zeckendorf游戏）
    
    规则:
        - 一堆n个石子
        - 第一次可以取1到n-1个
        - 之后每次取的数量不能超过上一次的2倍
        - 取走最后石子的人获胜
    
    必胜条件: n不是斐波那契数
    
    这是Zeckendorf定理的应用
    """
    # 生成斐波那契数列
    fib = [1, 2]
    while fib[-1] < n:
        fib.append(fib[-1] + fib[-2])
    
    return n not in fib
```

## 3.3 SG函数应用

```python
def sg_game_chips(positions: list, n: int) -> bool:
    """
    棋子移动游戏
    
    规则:
        - n个格子，某些格子有棋子
        - 每次选一个棋子向左移动任意格（不能跳过其他棋子）
        - 无法移动的人输
    
    分析:
        每个棋子独立，SG值 = 它左边的空格数
        总SG值 = 各棋子SG值的异或
    """
    # positions是棋子所在位置的列表（已排序）
    positions = sorted(positions)
    
    xor_sum = 0
    for i, pos in enumerate(positions):
        # 左边的空格数 = 当前位置 - 左边棋子数
        empty_left = pos - i
        xor_sum ^= empty_left
    
    return xor_sum != 0


def sg_stone_game(piles: list) -> bool:
    """
    取石子游戏（每次取某堆的任意因子个）
    
    规则: 每次从一堆中取走该堆石子数的某个因子个
    
    需要预计算每个数的SG值
    """
    def get_factors(n):
        """获取n的所有因子（不包括n本身）"""
        factors = []
        i = 1
        while i * i <= n:
            if n % i == 0:
                if i < n:
                    factors.append(i)
                if i != n // i and n // i < n:
                    factors.append(n // i)
            i += 1
        return factors
    
    # 预计算SG值
    max_pile = max(piles) if piles else 0
    sg = [0] * (max_pile + 1)
    
    for n in range(1, max_pile + 1):
        reachable = set()
        for f in get_factors(n):
            reachable.add(sg[n - f])
        sg[n] = mex(reachable)
    
    # 异或所有堆的SG值
    xor_sum = 0
    for pile in piles:
        xor_sum ^= sg[pile]
    
    return xor_sum != 0
```

## 4.1 Minimax基础

```python
def minimax(state, depth: int, is_maximizing: bool, 
            evaluate, get_moves, is_terminal) -> int:
    """
    极大极小算法
    
    参数:
        state: 当前游戏状态
        depth: 搜索深度
        is_maximizing: bool - 当前是否是最大化玩家
        evaluate: 评估函数
        get_moves: 获取所有可能移动的函数
        is_terminal: 判断是否终态的函数
    返回:
        int - 最优评估值
    
    适用于: 井字棋、五子棋、国际象棋等对抗游戏
    """
    if depth == 0 or is_terminal(state):
        return evaluate(state)
    
    moves = get_moves(state)
    
    if is_maximizing:
        max_eval = float('-inf')
        for move in moves:
            new_state = apply_move(state, move)
            eval_score = minimax(new_state, depth - 1, False,
                                evaluate, get_moves, is_terminal)
            max_eval = max(max_eval, eval_score)
        return max_eval
    else:
        min_eval = float('inf')
        for move in moves:
            new_state = apply_move(state, move)
            eval_score = minimax(new_state, depth - 1, True,
                                evaluate, get_moves, is_terminal)
            min_eval = min(min_eval, eval_score)
        return min_eval
```

## 4.2 Alpha-Beta剪枝

```python
def alpha_beta(state, depth: int, alpha: float, beta: float,
               is_maximizing: bool, evaluate, get_moves, is_terminal):
    """
    Alpha-Beta剪枝
    
    在Minimax基础上添加剪枝，大幅提高效率
    
    参数:
        alpha: float - 最大化玩家当前最优值
        beta: float - 最小化玩家当前最优值
    
    剪枝原理:
        - Alpha剪枝: 最小化层发现值 <= alpha，可以剪枝
        - Beta剪枝: 最大化层发现值 >= beta，可以剪枝
    
    时间复杂度: 最优情况O(b^(d/2))，最坏O(b^d)
    """
    if depth == 0 or is_terminal(state):
        return evaluate(state)
    
    moves = get_moves(state)
    
    if is_maximizing:
        max_eval = float('-inf')
        for move in moves:
            new_state = apply_move(state, move)
            eval_score = alpha_beta(new_state, depth - 1, alpha, beta,
                                   False, evaluate, get_moves, is_terminal)
            max_eval = max(max_eval, eval_score)
            alpha = max(alpha, eval_score)
            if beta <= alpha:
                break  # Beta剪枝
        return max_eval
    else:
        min_eval = float('inf')
        for move in moves:
            new_state = apply_move(state, move)
            eval_score = alpha_beta(new_state, depth - 1, alpha, beta,
                                   True, evaluate, get_moves, is_terminal)
            min_eval = min(min_eval, eval_score)
            beta = min(beta, eval_score)
            if beta <= alpha:
                break  # Alpha剪枝
        return min_eval
```

## 4.3 井字棋完整实现

```python
class TicTacToe:
    """
    井字棋（Minimax完整实现）
    
    LeetCode相关: 348, 794
    """
    
    def __init__(self):
        self.board = [[0] * 3 for _ in range(3)]
        # 0: 空, 1: X, -1: O
    
    def get_moves(self) -> list:
        """获取所有可能的移动"""
        moves = []
        for i in range(3):
            for j in range(3):
                if self.board[i][j] == 0:
                    moves.append((i, j))
        return moves
    
    def make_move(self, row: int, col: int, player: int) -> None:
        """执行移动"""
        self.board[row][col] = player
    
    def undo_move(self, row: int, col: int) -> None:
        """撤销移动"""
        self.board[row][col] = 0
    
    def check_winner(self) -> int:
        """
        检查胜者
        
        返回:
            1: X胜
            -1: O胜
            0: 未结束或平局
        """
        # 检查行
        for i in range(3):
            if abs(sum(self.board[i])) == 3:
                return self.board[i][0]
        
        # 检查列
        for j in range(3):
            col_sum = sum(self.board[i][j] for i in range(3))
            if abs(col_sum) == 3:
                return self.board[0][j]
        
        # 检查对角线
        diag1 = sum(self.board[i][i] for i in range(3))
        if abs(diag1) == 3:
            return self.board[0][0]
        
        diag2 = sum(self.board[i][2-i] for i in range(3))
        if abs(diag2) == 3:
            return self.board[0][2]
        
        return 0
    
    def is_terminal(self) -> bool:
        """是否终态"""
        if self.check_winner() != 0:
            return True
        return len(self.get_moves()) == 0
    
    def minimax(self, is_maximizing: bool) -> int:
        """Minimax求解"""
        winner = self.check_winner()
        if winner != 0:
            return winner * 10  # 放大差异
        
        if not self.get_moves():
            return 0  # 平局
        
        if is_maximizing:
            best = float('-inf')
            for row, col in self.get_moves():
                self.make_move(row, col, 1)
                score = self.minimax(False)
                self.undo_move(row, col)
                best = max(best, score)
            return best
        else:
            best = float('inf')
            for row, col in self.get_moves():
                self.make_move(row, col, -1)
                score = self.minimax(True)
                self.undo_move(row, col)
                best = min(best, score)
            return best
    
    def best_move(self, player: int) -> tuple:
        """
        获取最佳移动
        
        参数:
            player: 1(X) 或 -1(O)
        返回:
            (row, col) - 最佳位置
        """
        best_score = float('-inf') if player == 1 else float('inf')
        best_pos = None
        
        for row, col in self.get_moves():
            self.make_move(row, col, player)
            score = self.minimax(player != 1)
            self.undo_move(row, col)
            
            if player == 1:
                if score > best_score:
                    best_score = score
                    best_pos = (row, col)
            else:
                if score < best_score:
                    best_score = score
                    best_pos = (row, col)
        
        return best_pos
```

## 5.1 取石子游戏

```python
def stone_game(piles: list) -> bool:
    """
    LeetCode 877. Stone Game
    
    规则:
        - 石子排成一行，偶数堆
        - 两人轮流从两端取一堆
        - 石子多的人获胜
    
    结论: 先手必胜（偶数堆时可以控制取奇数位或偶数位）
    
    证明: 先手可以计算奇数位和偶数位的总和，选择大的一方
    """
    return True  # 先手必胜


def stone_game_dp(piles: list) -> bool:
    """
    Stone Game - DP解法
    
    dp[i][j] = 从piles[i]到piles[j]，先手比后手多拿的最大石子数
    """
    n = len(piles)
    dp = [[0] * n for _ in range(n)]
    
    # 只有一堆时
    for i in range(n):
        dp[i][i] = piles[i]
    
    # 区间DP
    for length in range(2, n + 1):
        for i in range(n - length + 1):
            j = i + length - 1
            # 先手选左边或右边
            dp[i][j] = max(
                piles[i] - dp[i+1][j],
                piles[j] - dp[i][j-1]
            )
    
    return dp[0][n-1] > 0


def stone_game_ii(piles: list) -> int:
    """
    LeetCode 1140. Stone Game II
    
    规则:
        - 每次可以取前1到2M堆
        - M初始为1，取X堆后M变为max(M, X)
        - 最大化自己获得的石子
    
    返回: 先手能获得的最大石子数
    """
    n = len(piles)
    suffix = [0] * (n + 1)
    for i in range(n - 1, -1, -1):
        suffix[i] = suffix[i + 1] + piles[i]
    
    from functools import lru_cache
    
    @lru_cache(maxsize=None)
    def dp(i: int, m: int) -> int:
        """从位置i开始，M=m时，先手能获得的最大石子数"""
        if i >= n:
            return 0
        
        if i + 2 * m >= n:
            return suffix[i]  # 可以全部取走
        
        min_next = float('inf')
        for x in range(1, 2 * m + 1):
            # 对手获得的最大值
            min_next = min(min_next, dp(i + x, max(m, x)))
        
        return suffix[i] - min_next
    
    return dp(0, 1)


def stone_game_iii(stoneValue: list) -> str:
    """
    LeetCode 1406. Stone Game III
    
    规则:
        - 每次可以取前1、2或3堆
        - 石子有正有负
        - 最大化自己的得分
    
    返回: "Alice", "Bob", 或 "Tie"
    """
    n = len(stoneValue)
    
    from functools import lru_cache
    
    @lru_cache(maxsize=None)
    def dp(i: int) -> int:
        """从位置i开始，先手比后手多的分数"""
        if i >= n:
            return 0
        
        best = float('-inf')
        take = 0
        for k in range(1, 4):
            if i + k - 1 < n:
                take += stoneValue[i + k - 1]
                best = max(best, take - dp(i + k))
        
        return best
    
    diff = dp(0)
    if diff > 0:
        return "Alice"
    elif diff < 0:
        return "Bob"
    else:
        return "Tie"
```

## 5.2 翻转游戏

```python
def can_win_flip_game(s: str) -> bool:
    """
    LeetCode 294. Flip Game II
    
    规则:
        - 字符串只包含'+' 和 '-'
        - 每次将连续的"++"翻转为"--"
        - 无法操作的人输
    
    返回: 先手是否必胜
    """
    from functools import lru_cache
    
    @lru_cache(maxsize=None)
    def can_win(state: str) -> bool:
        for i in range(len(state) - 1):
            if state[i:i+2] == '++':
                new_state = state[:i] + '--' + state[i+2:]
                if not can_win(new_state):
                    return True
        return False
    
    return can_win(s)
```

## 5.3 除数游戏

```python
def divisor_game(n: int) -> bool:
    """
    LeetCode 1025. Divisor Game
    
    规则:
        - 黑板上有数字n
        - 选择0 < x < n且n % x == 0的x
        - 将n替换为n-x
        - 无法操作的人输
    
    结论: n为偶数时先手必胜
    
    证明:
        - n=1: 无法操作，先手输
        - n=2: 取1，对手得到1，先手胜
        - n为偶数: 取1，对手得到奇数
        - n为奇数: 只能取奇数因子，对手得到偶数
    """
    return n % 2 == 0


def divisor_game_dp(n: int) -> bool:
    """DP解法验证"""
    if n == 1:
        return False
    
    dp = [False] * (n + 1)
    # dp[i] = True表示面对i时先手必胜
    
    dp[1] = False
    
    for i in range(2, n + 1):
        for x in range(1, i):
            if i % x == 0:
                if not dp[i - x]:  # 对手面对必败态
                    dp[i] = True
                    break
    
    return dp[n]
```

```python
def nim(piles):
    xor = 0
    for p in piles:
        xor ^= p
    return xor != 0

# 3. SG函数
def sg(n, moves, memo={}):
    if n in memo:
        return memo[n]
    reachable = {sg(n-m, moves, memo) for m in moves if m <= n}
    memo[n] = mex(reachable)
    return memo[n]

# 4. Minimax
def minimax(state, is_max):
    if terminal(state):
        return evaluate(state)
    if is_max:
        return max(minimax(next_state, False) for next_state in moves(state))
    else:
        return min(minimax(next_state, True) for next_state in moves(state))
```
