## 1.1 节点定义

```python
class TreeNode:
    """
    二叉树节点定义
    
    属性:
        val: 节点值
        left: 左子节点
        right: 右子节点
    """
    def __init__(self, val=0, left=None, right=None):
        self.val = val
        self.left = left
        self.right = right


# 从列表构建二叉树（层序）
def build_tree_from_list(values: list) -> TreeNode:
    """
    从层序遍历列表构建二叉树
    
    参数:
        values: List - 层序遍历列表，None表示空节点
    返回:
        TreeNode - 根节点
    
    示例:
        [1, 2, 3, None, 4] 构建:
              1
             / \
            2   3
             \
              4
    """
    if not values:
        return None
    
    from collections import deque
    
    root = TreeNode(values[0])
    queue = deque([root])
    i = 1
    
    while queue and i < len(values):
        node = queue.popleft()
        
        if i < len(values) and values[i] is not None:
            node.left = TreeNode(values[i])
            queue.append(node.left)
        i += 1
        
        if i < len(values) and values[i] is not None:
            node.right = TreeNode(values[i])
            queue.append(node.right)
        i += 1
    
    return root
```

## 1.2 树的遍历

```python
from collections import deque
from typing import List, Optional

# === 前序遍历 ===
def preorder_recursive(root: TreeNode) -> List[int]:
    """前序遍历：根 -> 左 -> 右（递归）"""
    result = []
    
    def dfs(node):
        if not node:
            return
        result.append(node.val)
        dfs(node.left)
        dfs(node.right)
    
    dfs(root)
    return result


def preorder_iterative(root: TreeNode) -> List[int]:
    """
    前序遍历（迭代）
    
    使用栈模拟递归
    """
    if not root:
        return []
    
    result = []
    stack = [root]
    
    while stack:
        node = stack.pop()
        result.append(node.val)
        # 先右后左，因为栈是LIFO
        if node.right:
            stack.append(node.right)
        if node.left:
            stack.append(node.left)
    
    return result


# === 中序遍历 ===
def inorder_recursive(root: TreeNode) -> List[int]:
    """中序遍历：左 -> 根 -> 右（递归）"""
    result = []
    
    def dfs(node):
        if not node:
            return
        dfs(node.left)
        result.append(node.val)
        dfs(node.right)
    
    dfs(root)
    return result


def inorder_iterative(root: TreeNode) -> List[int]:
    """
    中序遍历（迭代）
    
    沿着左子树一直向下，然后访问节点，转向右子树
    """
    result = []
    stack = []
    curr = root
    
    while curr or stack:
        # 一直向左走
        while curr:
            stack.append(curr)
            curr = curr.left
        
        # 访问节点
        curr = stack.pop()
        result.append(curr.val)
        
        # 转向右子树
        curr = curr.right
    
    return result


# === 后序遍历 ===
def postorder_recursive(root: TreeNode) -> List[int]:
    """后序遍历：左 -> 右 -> 根（递归）"""
    result = []
    
    def dfs(node):
        if not node:
            return
        dfs(node.left)
        dfs(node.right)
        result.append(node.val)
    
    dfs(root)
    return result


def postorder_iterative(root: TreeNode) -> List[int]:
    """
    后序遍历（迭代）
    
    前序是 根左右，反过来是 根右左
    将 根右左 反转得到 左右根，即后序
    """
    if not root:
        return []
    
    result = []
    stack = [root]
    
    while stack:
        node = stack.pop()
        result.append(node.val)
        # 先左后右（与前序相反）
        if node.left:
            stack.append(node.left)
        if node.right:
            stack.append(node.right)
    
    return result[::-1]


# === 层序遍历 ===
def level_order(root: TreeNode) -> List[List[int]]:
    """
    LeetCode 102. Binary Tree Level Order Traversal
    
    层序遍历，返回每层的节点值
    """
    if not root:
        return []
    
    result = []
    queue = deque([root])
    
    while queue:
        level_size = len(queue)
        level = []
        
        for _ in range(level_size):
            node = queue.popleft()
            level.append(node.val)
            
            if node.left:
                queue.append(node.left)
            if node.right:
                queue.append(node.right)
        
        result.append(level)
    
    return result


def zigzag_level_order(root: TreeNode) -> List[List[int]]:
    """
    LeetCode 103. Binary Tree Zigzag Level Order Traversal
    
    之字形层序遍历
    """
    if not root:
        return []
    
    result = []
    queue = deque([root])
    left_to_right = True
    
    while queue:
        level_size = len(queue)
        level = deque()
        
        for _ in range(level_size):
            node = queue.popleft()
            
            if left_to_right:
                level.append(node.val)
            else:
                level.appendleft(node.val)
            
            if node.left:
                queue.append(node.left)
            if node.right:
                queue.append(node.right)
        
        result.append(list(level))
        left_to_right = not left_to_right
    
    return result
```

# 二、二叉树构造

## 2.1 从遍历序列构造

```python
def build_tree_pre_in(preorder: List[int], inorder: List[int]) -> TreeNode:
    """
    LeetCode 105. Construct Binary Tree from Preorder and Inorder Traversal
    
    从前序和中序遍历构造二叉树
    
    原理:
    - 前序第一个是根
    - 在中序中找到根，左边是左子树，右边是右子树
    """
    if not preorder or not inorder:
        return None
    
    inorder_map = {val: i for i, val in enumerate(inorder)}
    
    def build(pre_start, pre_end, in_start, in_end):
        if pre_start > pre_end:
            return None
        
        root_val = preorder[pre_start]
        root = TreeNode(root_val)
        
        in_root = inorder_map[root_val]
        left_size = in_root - in_start
        
        root.left = build(pre_start + 1, pre_start + left_size, 
                          in_start, in_root - 1)
        root.right = build(pre_start + left_size + 1, pre_end, 
                           in_root + 1, in_end)
        
        return root
    
    return build(0, len(preorder) - 1, 0, len(inorder) - 1)


def build_tree_in_post(inorder: List[int], postorder: List[int]) -> TreeNode:
    """
    LeetCode 106. Construct Binary Tree from Inorder and Postorder Traversal
    
    从中序和后序遍历构造二叉树
    
    原理:
    - 后序最后一个是根
    - 在中序中找到根，分割左右子树
    """
    if not inorder or not postorder:
        return None
    
    inorder_map = {val: i for i, val in enumerate(inorder)}
    
    def build(in_start, in_end, post_start, post_end):
        if in_start > in_end:
            return None
        
        root_val = postorder[post_end]
        root = TreeNode(root_val)
        
        in_root = inorder_map[root_val]
        left_size = in_root - in_start
        
        root.left = build(in_start, in_root - 1, 
                          post_start, post_start + left_size - 1)
        root.right = build(in_root + 1, in_end, 
                           post_start + left_size, post_end - 1)
        
        return root
    
    return build(0, len(inorder) - 1, 0, len(postorder) - 1)
```

## 2.2 序列化与反序列化

```python
class Codec:
    """
    LeetCode 297. Serialize and Deserialize Binary Tree
    
    二叉树的序列化和反序列化
    """
    
    def serialize(self, root: TreeNode) -> str:
        """
        将二叉树序列化为字符串
        
        使用前序遍历，None用'#'表示
        """
        def dfs(node):
            if not node:
                return ['#']
            return [str(node.val)] + dfs(node.left) + dfs(node.right)
        
        return ','.join(dfs(root))
    
    def deserialize(self, data: str) -> TreeNode:
        """
        将字符串反序列化为二叉树
        """
        values = iter(data.split(','))
        
        def dfs():
            val = next(values)
            if val == '#':
                return None
            node = TreeNode(int(val))
            node.left = dfs()
            node.right = dfs()
            return node
        
        return dfs()


class CodecBFS:
    """
    使用BFS的序列化（更直观）
    """
    
    def serialize(self, root: TreeNode) -> str:
        if not root:
            return ''
        
        result = []
        queue = deque([root])
        
        while queue:
            node = queue.popleft()
            if node:
                result.append(str(node.val))
                queue.append(node.left)
                queue.append(node.right)
            else:
                result.append('#')
        
        # 去除末尾的#
        while result and result[-1] == '#':
            result.pop()
        
        return ','.join(result)
    
    def deserialize(self, data: str) -> TreeNode:
        if not data:
            return None
        
        values = data.split(',')
        root = TreeNode(int(values[0]))
        queue = deque([root])
        i = 1
        
        while queue and i < len(values):
            node = queue.popleft()
            
            if i < len(values) and values[i] != '#':
                node.left = TreeNode(int(values[i]))
                queue.append(node.left)
            i += 1
            
            if i < len(values) and values[i] != '#':
                node.right = TreeNode(int(values[i]))
                queue.append(node.right)
            i += 1
        
        return root
```

```python
def max_depth(root: TreeNode) -> int:
    """
    LeetCode 104. Maximum Depth of Binary Tree
    
    最大深度
    """
    if not root:
        return 0
    return 1 + max(max_depth(root.left), max_depth(root.right))


def min_depth(root: TreeNode) -> int:
    """
    LeetCode 111. Minimum Depth of Binary Tree
    
    最小深度（到叶子节点的最短路径）
    """
    if not root:
        return 0
    
    if not root.left:
        return 1 + min_depth(root.right)
    if not root.right:
        return 1 + min_depth(root.left)
    
    return 1 + min(min_depth(root.left), min_depth(root.right))


def is_balanced(root: TreeNode) -> bool:
    """
    LeetCode 110. Balanced Binary Tree
    
    判断是否平衡二叉树（左右子树高度差不超过1）
    """
    def height(node):
        if not node:
            return 0
        
        left = height(node.left)
        if left == -1:
            return -1
        
        right = height(node.right)
        if right == -1:
            return -1
        
        if abs(left - right) > 1:
            return -1
        
        return 1 + max(left, right)
    
    return height(root) != -1
```

## 3.2 路径问题

```python
def has_path_sum(root: TreeNode, targetSum: int) -> bool:
    """
    LeetCode 112. Path Sum
    
    是否存在根到叶子的路径和等于目标值
    """
    if not root:
        return False
    
    if not root.left and not root.right:
        return root.val == targetSum
    
    remaining = targetSum - root.val
    return (has_path_sum(root.left, remaining) or 
            has_path_sum(root.right, remaining))


def path_sum(root: TreeNode, targetSum: int) -> List[List[int]]:
    """
    LeetCode 113. Path Sum II
    
    返回所有和为目标值的根到叶子路径
    """
    result = []
    
    def dfs(node, remaining, path):
        if not node:
            return
        
        path.append(node.val)
        
        if not node.left and not node.right and node.val == remaining:
            result.append(path.copy())
        else:
            dfs(node.left, remaining - node.val, path)
            dfs(node.right, remaining - node.val, path)
        
        path.pop()
    
    dfs(root, targetSum, [])
    return result


def max_path_sum(root: TreeNode) -> int:
    """
    LeetCode 124. Binary Tree Maximum Path Sum
    
    最大路径和（路径可以从任意节点到任意节点）
    """
    result = float('-inf')
    
    def dfs(node):
        nonlocal result
        if not node:
            return 0
        
        # 只取正贡献
        left = max(0, dfs(node.left))
        right = max(0, dfs(node.right))
        
        # 经过当前节点的最大路径和
        result = max(result, left + right + node.val)
        
        # 返回能贡献给父节点的最大值
        return max(left, right) + node.val
    
    dfs(root)
    return result


def diameter_of_binary_tree(root: TreeNode) -> int:
    """
    LeetCode 543. Diameter of Binary Tree
    
    直径（最长路径的边数）
    """
    result = 0
    
    def depth(node):
        nonlocal result
        if not node:
            return 0
        
        left = depth(node.left)
        right = depth(node.right)
        
        result = max(result, left + right)
        return max(left, right) + 1
    
    depth(root)
    return result
```

## 3.3 最近公共祖先

```python
def lowest_common_ancestor(root: TreeNode, p: TreeNode, q: TreeNode) -> TreeNode:
    """
    LeetCode 236. Lowest Common Ancestor of a Binary Tree
    
    最近公共祖先（LCA）
    """
    if not root or root == p or root == q:
        return root
    
    left = lowest_common_ancestor(root.left, p, q)
    right = lowest_common_ancestor(root.right, p, q)
    
    if left and right:
        return root
    
    return left or right


def lca_bst(root: TreeNode, p: TreeNode, q: TreeNode) -> TreeNode:
    """
    LeetCode 235. Lowest Common Ancestor of a BST
    
    BST的最近公共祖先（利用BST性质）
    """
    while root:
        if p.val < root.val and q.val < root.val:
            root = root.left
        elif p.val > root.val and q.val > root.val:
            root = root.right
        else:
            return root
    
    return None
```

## 3.4 对称与翻转

```python
def is_symmetric(root: TreeNode) -> bool:
    """
    LeetCode 101. Symmetric Tree
    
    判断是否对称
    """
    def is_mirror(left, right):
        if not left and not right:
            return True
        if not left or not right:
            return False
        return (left.val == right.val and 
                is_mirror(left.left, right.right) and 
                is_mirror(left.right, right.left))
    
    return is_mirror(root, root)


def invert_tree(root: TreeNode) -> TreeNode:
    """
    LeetCode 226. Invert Binary Tree
    
    翻转二叉树
    """
    if not root:
        return None
    
    root.left, root.right = invert_tree(root.right), invert_tree(root.left)
    return root


def is_same_tree(p: TreeNode, q: TreeNode) -> bool:
    """
    LeetCode 100. Same Tree
    
    判断两棵树是否相同
    """
    if not p and not q:
        return True
    if not p or not q:
        return False
    return (p.val == q.val and 
            is_same_tree(p.left, q.left) and 
            is_same_tree(p.right, q.right))


def is_subtree(root: TreeNode, subRoot: TreeNode) -> bool:
    """
    LeetCode 572. Subtree of Another Tree
    
    判断subRoot是否是root的子树
    """
    if not root:
        return False
    
    if is_same_tree(root, subRoot):
        return True
    
    return is_subtree(root.left, subRoot) or is_subtree(root.right, subRoot)
```

## 4.1 BST基本操作

```python
def search_bst(root: TreeNode, val: int) -> TreeNode:
    """
    LeetCode 700. Search in a Binary Search Tree
    
    在BST中查找值
    """
    while root and root.val != val:
        root = root.left if val < root.val else root.right
    return root


def insert_into_bst(root: TreeNode, val: int) -> TreeNode:
    """
    LeetCode 701. Insert into a Binary Search Tree
    
    插入值到BST
    """
    if not root:
        return TreeNode(val)
    
    if val < root.val:
        root.left = insert_into_bst(root.left, val)
    else:
        root.right = insert_into_bst(root.right, val)
    
    return root


def delete_node(root: TreeNode, key: int) -> TreeNode:
    """
    LeetCode 450. Delete Node in a BST
    
    从BST中删除值
    """
    if not root:
        return None
    
    if key < root.val:
        root.left = delete_node(root.left, key)
    elif key > root.val:
        root.right = delete_node(root.right, key)
    else:
        # 找到要删除的节点
        if not root.left:
            return root.right
        if not root.right:
            return root.left
        
        # 找右子树的最小值（后继）
        successor = root.right
        while successor.left:
            successor = successor.left
        
        root.val = successor.val
        root.right = delete_node(root.right, successor.val)
    
    return root
```

## 4.2 BST验证与转换

```python
def is_valid_bst(root: TreeNode) -> bool:
    """
    LeetCode 98. Validate Binary Search Tree
    
    验证BST
    """
    def validate(node, min_val, max_val):
        if not node:
            return True
        
        if node.val <= min_val or node.val >= max_val:
            return False
        
        return (validate(node.left, min_val, node.val) and 
                validate(node.right, node.val, max_val))
    
    return validate(root, float('-inf'), float('inf'))


def sorted_array_to_bst(nums: List[int]) -> TreeNode:
    """
    LeetCode 108. Convert Sorted Array to Binary Search Tree
    
    有序数组转平衡BST
    """
    def build(left, right):
        if left > right:
            return None
        
        mid = (left + right) // 2
        node = TreeNode(nums[mid])
        node.left = build(left, mid - 1)
        node.right = build(mid + 1, right)
        
        return node
    
    return build(0, len(nums) - 1)


def kth_smallest(root: TreeNode, k: int) -> int:
    """
    LeetCode 230. Kth Smallest Element in a BST
    
    BST中第k小的元素
    
    中序遍历BST得到有序序列
    """
    stack = []
    curr = root
    
    while curr or stack:
        while curr:
            stack.append(curr)
            curr = curr.left
        
        curr = stack.pop()
        k -= 1
        if k == 0:
            return curr.val
        
        curr = curr.right
    
    return -1
```

## 5.1 二叉树展开为链表

```python
def flatten(root: TreeNode) -> None:
    """
    LeetCode 114. Flatten Binary Tree to Linked List
    
    将二叉树展开为链表（前序）
    """
    curr = root
    
    while curr:
        if curr.left:
            # 找左子树的最右节点
            rightmost = curr.left
            while rightmost.right:
                rightmost = rightmost.right
            
            # 将右子树接到最右节点
            rightmost.right = curr.right
            # 将左子树移到右边
            curr.right = curr.left
            curr.left = None
        
        curr = curr.right
```

## 5.2 二叉树的右视图

```python
def right_side_view(root: TreeNode) -> List[int]:
    """
    LeetCode 199. Binary Tree Right Side View
    
    从右边看二叉树，返回能看到的节点值
    """
    if not root:
        return []
    
    result = []
    queue = deque([root])
    
    while queue:
        level_size = len(queue)
        for i in range(level_size):
            node = queue.popleft()
            if i == level_size - 1:
                result.append(node.val)
            if node.left:
                queue.append(node.left)
            if node.right:
                queue.append(node.right)
    
    return result
```

## 5.3 完全二叉树的节点个数

```python
def count_nodes(root: TreeNode) -> int:
    """
    LeetCode 222. Count Complete Tree Nodes
    
    完全二叉树的节点数
    
    时间复杂度: O(log²n)
    """
    if not root:
        return 0
    
    left_depth = 0
    right_depth = 0
    left = right = root
    
    while left:
        left_depth += 1
        left = left.left
    
    while right:
        right_depth += 1
        right = right.right
    
    # 如果左右深度相等，是满二叉树
    if left_depth == right_depth:
        return (1 << left_depth) - 1
    
    # 否则递归计算
    return 1 + count_nodes(root.left) + count_nodes(root.right)
```

## 总结

### 遍历方式

| 遍历 | 顺序 | 递归 | 迭代 |
|------|------|------|------|
| 前序 | 根左右 | 直接递归 | 栈，先右后左 |
| 中序 | 左根右 | 直接递归 | 栈，沿左向下 |
| 后序 | 左右根 | 直接递归 | 前序反向 |
| 层序 | 逐层 | - | BFS队列 |

### 常见问题模式

| 问题类型 | 解法 |
|----------|------|
| 深度/高度 | 递归，1+max(左,右) |
| 路径问题 | DFS + 回溯 |
| LCA | 递归分治 |
| BST操作 | 利用有序性 |
| 序列化 | 前序/BFS + 标记空节点 |

### 关键记忆

1. **前序迭代**：栈，先压右再压左
2. **中序迭代**：沿左向下，弹出访问，转右
