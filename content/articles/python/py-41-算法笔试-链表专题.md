# 一、链表基础

## 1.1 节点定义

```python
class ListNode:
    """
    单链表节点
    
    属性:
        val: 节点值
        next: 指向下一个节点的指针
    """
    def __init__(self, val=0, next=None):
        self.val = val
        self.next = next


class DoublyListNode:
    """
    双向链表节点
    
    属性:
        val: 节点值
        prev: 指向前一个节点
        next: 指向下一个节点
    """
    def __init__(self, val=0, prev=None, next=None):
        self.val = val
        self.prev = prev
        self.next = next
```

## 1.2 链表创建与辅助函数

```python
def create_linked_list(values: list) -> ListNode:
    """
    从列表创建链表
    
    参数:
        values: List[int] - 节点值列表
    返回:
        ListNode - 头节点
    """
    if not values:
        return None
    
    head = ListNode(values[0])
    curr = head
    
    for val in values[1:]:
        curr.next = ListNode(val)
        curr = curr.next
    
    return head


def linked_list_to_list(head: ListNode) -> list:
    """
    链表转列表
    """
    result = []
    while head:
        result.append(head.val)
        head = head.next
    return result


def print_linked_list(head: ListNode) -> None:
    """
    打印链表
    """
    values = []
    while head:
        values.append(str(head.val))
        head = head.next
    print(" -> ".join(values) + " -> None")


def get_length(head: ListNode) -> int:
    """
    获取链表长度
    """
    length = 0
    while head:
        length += 1
        head = head.next
    return length


def get_node(head: ListNode, index: int) -> ListNode:
    """
    获取第index个节点（0-indexed）
    """
    for _ in range(index):
        if not head:
            return None
        head = head.next
    return head
```

## 1.3 虚拟头节点技巧

```python
"""
虚拟头节点（Dummy Node）技巧

使用场景:
- 可能删除头节点
- 需要在头部插入
- 合并链表
- 统一处理逻辑

好处:
- 不用单独处理头节点
- 代码更简洁
"""

def delete_node_example(head: ListNode, val: int) -> ListNode:
    """
    删除值为val的节点
    """
    dummy = ListNode(0)
    dummy.next = head
    
    curr = dummy
    while curr.next:
        if curr.next.val == val:
            curr.next = curr.next.next
        else:
            curr = curr.next
    
    return dummy.next
```

# 二、链表反转

## 2.1 反转整个链表

```python
def reverse_list_iterative(head: ListNode) -> ListNode:
    """
    LeetCode 206. Reverse Linked List
    
    反转链表（迭代）
    
    时间复杂度: O(n)
    空间复杂度: O(1)
    """
    prev = None
    curr = head
    
    while curr:
        next_temp = curr.next
        curr.next = prev
        prev = curr
        curr = next_temp
    
    return prev


def reverse_list_recursive(head: ListNode) -> ListNode:
    """
    反转链表（递归）
    
    思路: 假设后面的链表已经反转，处理当前节点
    
    空间复杂度: O(n) 递归栈
    """
    if not head or not head.next:
        return head
    
    new_head = reverse_list_recursive(head.next)
    
    # head.next是反转后的尾节点
    head.next.next = head
    head.next = None
    
    return new_head
```

## 2.2 反转部分链表

```python
def reverse_between(head: ListNode, left: int, right: int) -> ListNode:
    """
    LeetCode 92. Reverse Linked List II
    
    反转从left到right的部分
    
    位置从1开始
    """
    if not head or left == right:
        return head
    
    dummy = ListNode(0)
    dummy.next = head
    
    # 找到left的前一个节点
    prev = dummy
    for _ in range(left - 1):
        prev = prev.next
    
    # 反转left到right之间的节点
    curr = prev.next
    for _ in range(right - left):
        next_node = curr.next
        curr.next = next_node.next
        next_node.next = prev.next
        prev.next = next_node
    
    return dummy.next


def reverse_k_group(head: ListNode, k: int) -> ListNode:
    """
    LeetCode 25. Reverse Nodes in k-Group
    
    每k个节点一组反转
    """
    def reverse(start, end):
        """反转start到end之间的节点"""
        prev = None
        curr = start
        while curr != end:
            next_temp = curr.next
            curr.next = prev
            prev = curr
            curr = next_temp
        return prev
    
    dummy = ListNode(0)
    dummy.next = head
    prev_group = dummy
    
    while True:
        # 检查是否有k个节点
        kth = prev_group
        for _ in range(k):
            kth = kth.next
            if not kth:
                return dummy.next
        
        group_start = prev_group.next
        next_group = kth.next
        
        # 反转这组
        reverse(group_start, next_group)
        
        # 连接
        prev_group.next = kth
        group_start.next = next_group
        prev_group = group_start
    
    return dummy.next
```

## 2.3 判断回文链表

```python
def is_palindrome(head: ListNode) -> bool:
    """
    LeetCode 234. Palindrome Linked List
    
    判断链表是否回文
    
    方法: 快慢指针找中点 + 反转后半部分 + 比较
    
    时间复杂度: O(n)
    空间复杂度: O(1)
    """
    if not head or not head.next:
        return True
    
    # 找中点
    slow = fast = head
    while fast.next and fast.next.next:
        slow = slow.next
        fast = fast.next.next
    
    # 反转后半部分
    second_half = reverse_list_iterative(slow.next)
    
    # 比较
    first_half = head
    result = True
    while second_half:
        if first_half.val != second_half.val:
            result = False
            break
        first_half = first_half.next
        second_half = second_half.next
    
    # 恢复链表（可选）
    slow.next = reverse_list_iterative(slow.next)
    
    return result
```

# 三、链表合并

## 3.1 合并两个有序链表

```python
def merge_two_lists_iterative(l1: ListNode, l2: ListNode) -> ListNode:
    """
    LeetCode 21. Merge Two Sorted Lists
    
    合并两个有序链表（迭代）
    """
    dummy = ListNode(0)
    curr = dummy
    
    while l1 and l2:
        if l1.val <= l2.val:
            curr.next = l1
            l1 = l1.next
        else:
            curr.next = l2
            l2 = l2.next
        curr = curr.next
    
    curr.next = l1 or l2
    
    return dummy.next


def merge_two_lists_recursive(l1: ListNode, l2: ListNode) -> ListNode:
    """
    合并两个有序链表（递归）
    """
    if not l1:
        return l2
    if not l2:
        return l1
    
    if l1.val <= l2.val:
        l1.next = merge_two_lists_recursive(l1.next, l2)
        return l1
    else:
        l2.next = merge_two_lists_recursive(l1, l2.next)
        return l2
```

## 3.2 合并K个有序链表

```python
import heapq

def merge_k_lists(lists: list) -> ListNode:
    """
    LeetCode 23. Merge k Sorted Lists
    
    合并K个有序链表
    
    方法1: 最小堆
    时间复杂度: O(N log k)
    """
    # 定义比较函数（Python3需要）
    ListNode.__lt__ = lambda self, other: self.val < other.val
    
    heap = []
    for l in lists:
        if l:
            heapq.heappush(heap, l)
    
    dummy = ListNode(0)
    curr = dummy
    
    while heap:
        node = heapq.heappop(heap)
        curr.next = node
        curr = curr.next
        if node.next:
            heapq.heappush(heap, node.next)
    
    return dummy.next


def merge_k_lists_divide(lists: list) -> ListNode:
    """
    方法2: 分治合并
    """
    if not lists:
        return None
    
    def merge_two(l1, l2):
        dummy = ListNode(0)
        curr = dummy
        while l1 and l2:
            if l1.val <= l2.val:
                curr.next = l1
                l1 = l1.next
            else:
                curr.next = l2
                l2 = l2.next
            curr = curr.next
        curr.next = l1 or l2
        return dummy.next
    
    def merge_range(left, right):
        if left == right:
            return lists[left]
        mid = (left + right) // 2
        l = merge_range(left, mid)
        r = merge_range(mid + 1, right)
        return merge_two(l, r)
    
    return merge_range(0, len(lists) - 1)
```

# 四、环形链表

## 4.1 检测环

```python
def has_cycle(head: ListNode) -> bool:
    """
    LeetCode 141. Linked List Cycle
    
    判断链表是否有环
    
    Floyd判圈算法（快慢指针）
    
    时间复杂度: O(n)
    空间复杂度: O(1)
    """
    if not head or not head.next:
        return False
    
    slow = head
    fast = head
    
    while fast and fast.next:
        slow = slow.next
        fast = fast.next.next
        if slow == fast:
            return True
    
    return False


def detect_cycle(head: ListNode) -> ListNode:
    """
    LeetCode 142. Linked List Cycle II
    
    返回环的入口节点
    
    原理:
    设头到入口距离为a，入口到相遇点距离为b，环长为c
    快指针走的距离: a + b + nc
    慢指针走的距离: a + b
    快指针是慢指针的两倍: 2(a + b) = a + b + nc
    所以: a = nc - b = (n-1)c + (c-b)
    即从头走a步和从相遇点走(c-b)步会在入口相遇
    """
    if not head or not head.next:
        return None
    
    slow = fast = head
    
    # 第一阶段：找相遇点
    while fast and fast.next:
        slow = slow.next
        fast = fast.next.next
        if slow == fast:
            break
    else:
        return None
    
    # 第二阶段：找入口
    slow = head
    while slow != fast:
        slow = slow.next
        fast = fast.next
    
    return slow
```

## 4.2 链表相交

```python
def get_intersection_node(headA: ListNode, headB: ListNode) -> ListNode:
    """
    LeetCode 160. Intersection of Two Linked Lists
    
    找两个链表的交点
    
    原理:
    A走完走B，B走完走A，如果相交，会在交点相遇
    因为 a + c + b = b + c + a
    """
    if not headA or not headB:
        return None
    
    pA, pB = headA, headB
    
    while pA != pB:
        pA = pA.next if pA else headB
        pB = pB.next if pB else headA
    
    return pA
```

## 5.1 找中点

```python
def find_middle(head: ListNode) -> ListNode:
    """
    LeetCode 876. Middle of the Linked List
    
    找链表中点
    
    快指针走两步，慢指针走一步
    偶数个节点时返回第二个中间节点
    """
    slow = fast = head
    
    while fast and fast.next:
        slow = slow.next
        fast = fast.next.next
    
    return slow


def find_middle_first(head: ListNode) -> ListNode:
    """
    找链表中点（偶数个时返回第一个中间节点）
    """
    slow = fast = head
    
    while fast.next and fast.next.next:
        slow = slow.next
        fast = fast.next.next
    
    return slow
```

## 5.2 删除倒数第N个节点

```python
def remove_nth_from_end(head: ListNode, n: int) -> ListNode:
    """
    LeetCode 19. Remove Nth Node From End of List
    
    删除倒数第n个节点
    
    快指针先走n步，然后同步走
    """
    dummy = ListNode(0)
    dummy.next = head
    
    slow = fast = dummy
    
    # 快指针先走n+1步
    for _ in range(n + 1):
        fast = fast.next
    
    # 同步走到末尾
    while fast:
        slow = slow.next
        fast = fast.next
    
    # 删除slow的下一个节点
    slow.next = slow.next.next
    
    return dummy.next
```

## 5.3 重排链表

```python
def reorder_list(head: ListNode) -> None:
    """
    LeetCode 143. Reorder List
    
    将 L0→L1→...→Ln-1→Ln
    变为 L0→Ln→L1→Ln-1→L2→Ln-2→...
    
    步骤:
    1. 找中点
    2. 反转后半部分
    3. 交替合并
    """
    if not head or not head.next:
        return
    
    # 找中点
    slow = fast = head
    while fast.next and fast.next.next:
        slow = slow.next
        fast = fast.next.next
    
    # 反转后半部分
    second = slow.next
    slow.next = None
    prev = None
    while second:
        next_temp = second.next
        second.next = prev
        prev = second
        second = next_temp
    second = prev
    
    # 交替合并
    first = head
    while second:
        next1, next2 = first.next, second.next
        first.next = second
        second.next = next1
        first = next1
        second = next2
```

# 六、其他链表问题

## 6.1 排序链表

```python
def sort_list(head: ListNode) -> ListNode:
    """
    LeetCode 148. Sort List
    
    对链表排序（归并排序）
    
    时间复杂度: O(n log n)
    空间复杂度: O(log n) 递归栈
    """
    if not head or not head.next:
        return head
    
    # 找中点
    slow = head
    fast = head.next
    while fast and fast.next:
        slow = slow.next
        fast = fast.next.next
    
    # 分割
    mid = slow.next
    slow.next = None
    
    # 递归排序
    left = sort_list(head)
    right = sort_list(mid)
    
    # 合并
    dummy = ListNode(0)
    curr = dummy
    while left and right:
        if left.val <= right.val:
            curr.next = left
            left = left.next
        else:
            curr.next = right
            right = right.next
        curr = curr.next
    curr.next = left or right
    
    return dummy.next
```

## 6.2 复制带随机指针的链表

```python
class RandomListNode:
    def __init__(self, val=0, next=None, random=None):
        self.val = val
        self.next = next
        self.random = random


def copy_random_list(head: RandomListNode) -> RandomListNode:
    """
    LeetCode 138. Copy List with Random Pointer
    
    深拷贝带随机指针的链表
    
    方法1: 哈希表
    """
    if not head:
        return None
    
    # 第一遍：创建所有新节点
    old_to_new = {}
    curr = head
    while curr:
        old_to_new[curr] = RandomListNode(curr.val)
        curr = curr.next
    
    # 第二遍：设置next和random
    curr = head
    while curr:
        new_node = old_to_new[curr]
        new_node.next = old_to_new.get(curr.next)
        new_node.random = old_to_new.get(curr.random)
        curr = curr.next
    
    return old_to_new[head]


def copy_random_list_o1(head: RandomListNode) -> RandomListNode:
    """
    方法2: O(1)空间
    
    步骤:
    1. 在每个节点后插入复制节点
    2. 设置random指针
    3. 分离两个链表
    """
    if not head:
        return None
    
    # 1. 插入复制节点
    curr = head
    while curr:
        new_node = RandomListNode(curr.val)
        new_node.next = curr.next
        curr.next = new_node
        curr = new_node.next
    
    # 2. 设置random
    curr = head
    while curr:
        if curr.random:
            curr.next.random = curr.random.next
        curr = curr.next.next
    
    # 3. 分离
    new_head = head.next
    curr = head
    while curr:
        new_node = curr.next
        curr.next = new_node.next
        new_node.next = curr.next.next if curr.next else None
        curr = curr.next
    
    return new_head
```

## 6.3 两数相加

```python
def add_two_numbers(l1: ListNode, l2: ListNode) -> ListNode:
    """
    LeetCode 2. Add Two Numbers
    
    链表表示的两个数相加（逆序存储）
    """
    dummy = ListNode(0)
    curr = dummy
    carry = 0
    
    while l1 or l2 or carry:
        val1 = l1.val if l1 else 0
        val2 = l2.val if l2 else 0
        
        total = val1 + val2 + carry
        carry = total // 10
        curr.next = ListNode(total % 10)
        curr = curr.next
        
        l1 = l1.next if l1 else None
        l2 = l2.next if l2 else None
    
    return dummy.next


def add_two_numbers_ii(l1: ListNode, l2: ListNode) -> ListNode:
    """
    LeetCode 445. Add Two Numbers II
    
    链表表示的两个数相加（正序存储）
    
    使用栈
    """
    stack1, stack2 = [], []
    
    while l1:
        stack1.append(l1.val)
        l1 = l1.next
    while l2:
        stack2.append(l2.val)
        l2 = l2.next
    
    carry = 0
    head = None
    
    while stack1 or stack2 or carry:
        val1 = stack1.pop() if stack1 else 0
        val2 = stack2.pop() if stack2 else 0
        
        total = val1 + val2 + carry
        carry = total // 10
        
        node = ListNode(total % 10)
        node.next = head
        head = node
    
    return head
```

## 总结

### 常用技巧

| 技巧 | 使用场景 |
|------|----------|
| 虚拟头节点 | 可能修改头节点 |
| 快慢指针 | 找中点、检测环 |
| 双指针 | 删除倒数第N个、相交 |
| 递归 | 反转、合并 |
| 栈 | 正序相加 |

### 常见问题

| 问题 | 关键思路 |
|------|----------|
| 反转链表 | prev, curr, next三指针 |
| 检测环 | 快慢指针相遇 |
| 找环入口 | 相遇后从头走 |
| 找中点 | 快走两步慢走一步 |
| 合并链表 | 虚拟头节点 |
| 排序链表 | 归并排序 |

### 关键记忆

1. **反转**: prev = None, curr = head, 迭代交换
2. **环检测**: 快慢指针，快2慢1
3. **环入口**: 相遇后从头同步走
4. **中点**: fast.next and fast.next.next
5. **虚拟头**: dummy.next = head
