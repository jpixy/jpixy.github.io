```python
from functools import cmp_to_key

# 自定义比较函数
# 返回负数：a < b
# 返回0：a == b
# 返回正数：a > b

def compare(a, b):
    # 按长度排序，长度相同按字典序
    if len(a) != len(b):
        return len(a) - len(b)
    return (a > b) - (a < b)  # 等价于cmp(a, b)

words = ['apple', 'pie', 'banana', 'cat']
sorted(words, key=cmp_to_key(compare))
# ['cat', 'pie', 'apple', 'banana']

# 经典问题：最大数
def largest_number(nums):
    """将数字拼接成最大的数"""
    nums = list(map(str, nums))
    
    def compare(a, b):
        if a + b > b + a:
            return -1
        elif a + b < b + a:
            return 1
        return 0
    
    nums.sort(key=cmp_to_key(compare))
    result = ''.join(nums)
    return result if result[0] != '0' else '0'

# 示例
largest_number([3, 30, 34, 5, 9])  # "9534330"
```

```python
def merge_intervals(intervals):
    intervals.sort(key=lambda x: x[0])
    merged = [intervals[0]]

    ends = sorted(i[1] for i in intervals)

    end_ptr = 0

            rooms += 1

def find_overlaps(intervals):
    events = []
    for i, (start, end) in enumerate(intervals):
        events.append((start, 0, i))  # 开始
        events.append((end, 1, i))    # 结束
    
    events.sort()
    active = set()
    overlaps = []
    
    for time, typ, idx in events:
        if typ == 0:  # 开始
            for other in active:
                overlaps.append((other, idx))
            active.add(idx)
        else:  # 结束
            active.discard(idx)
    
    return overlaps
```

```python
def binary_search(arr, target):
```

```python
# 二分答案：最小化最大值 / 最大化最小值

def binary_search_answer(lo, hi, check):
    """
    找到满足check条件的最小值
    check(mid) == True 意味着 mid 可行
    """
    while lo < hi:
        mid = (lo + hi) // 2
        if check(mid):
            hi = mid
        else:
            lo = mid + 1
    return lo

def binary_search_answer_max(lo, hi, check):
    """
    找到满足check条件的最大值
    """
    while lo < hi:
        mid = (lo + hi + 1) // 2  # 注意+1，避免死循环
        if check(mid):
            lo = mid
        else:
            hi = mid - 1
    return lo

# 示例：分割数组的最大值（最小化）
def split_array(nums, k):
    """将数组分成k份，使得各份和的最大值最小"""
    def can_split(max_sum):
        count = 1
        curr_sum = 0
        for num in nums:
            if curr_sum + num > max_sum:
                count += 1
                curr_sum = num
            else:
                curr_sum += num
        return count <= k
    
    lo, hi = max(nums), sum(nums)
    while lo < hi:
        mid = (lo + hi) // 2
        if can_split(mid):
            hi = mid
        else:
            lo = mid + 1
    return lo

# 示例：Koko吃香蕉
def min_eating_speed(piles, h):
    """在h小时内吃完所有香蕉的最小速度"""
    import math
    
    def can_finish(speed):
        hours = sum(math.ceil(p / speed) for p in piles)
        return hours <= h
    
    lo, hi = 1, max(piles)
    while lo < hi:
        mid = (lo + hi) // 2
        if can_finish(mid):
            hi = mid
        else:
            lo = mid + 1
    return lo
```

```python
import bisect

# 1. 搜索旋转排序数组
def search_rotated(nums, target):
    lo, hi = 0, len(nums) - 1
    
    while lo <= hi:
        mid = (lo + hi) // 2
        if nums[mid] == target:
            return mid
        
        # 左半部分有序
        if nums[lo] <= nums[mid]:
            if nums[lo] <= target < nums[mid]:
                hi = mid - 1
            else:
                lo = mid + 1
        # 右半部分有序
        else:
            if nums[mid] < target <= nums[hi]:
                lo = mid + 1
            else:
                hi = mid - 1
    
    return -1

# 2. 寻找峰值
def find_peak(nums):
    lo, hi = 0, len(nums) - 1
    
    while lo < hi:
        mid = (lo + hi) // 2
        if nums[mid] < nums[mid + 1]:
            lo = mid + 1
        else:
            hi = mid
    
    return lo

# 3. 搜索二维矩阵
def search_matrix(matrix, target):
    """每行有序，每行第一个大于上一行最后一个"""
    if not matrix:
        return False
    
    m, n = len(matrix), len(matrix[0])
    lo, hi = 0, m * n - 1
    
    while lo <= hi:
        mid = (lo + hi) // 2
        val = matrix[mid // n][mid % n]
        if val == target:
            return True
        elif val < target:
            lo = mid + 1
        else:
            hi = mid - 1
    
    return False

# 4. 找到K个最接近的元素
def find_closest_elements(arr, k, x):
    # 二分找到最佳起始位置
    lo, hi = 0, len(arr) - k
    
    while lo < hi:
        mid = (lo + hi) // 2
        # 比较左右边界到x的距离
        if x - arr[mid] > arr[mid + k] - x:
            lo = mid + 1
        else:
            hi = mid
    
    return arr[lo:lo + k]

# 5. 统计有序数组中元素出现次数
def count_occurrences(arr, target):
    left = bisect.bisect_left(arr, target)
    right = bisect.bisect_right(arr, target)
    return right - left
```

```python
import heapq

# 1. Top K 问题
def top_k_largest(nums, k):
    """找到第K大的元素"""
    # 维护大小为k的最小堆
    heap = nums[:k]
    heapq.heapify(heap)
    
    for num in nums[k:]:
        if num > heap[0]:
            heapq.heapreplace(heap, num)
    
    return heap[0]  # 第k大

def top_k_frequent(nums, k):
    """找到出现频率最高的k个元素"""
    from collections import Counter
    count = Counter(nums)
    # 使用nlargest
    return heapq.nlargest(k, count.keys(), key=count.get)

# 2. 合并K个有序链表/数组
def merge_k_sorted_lists(lists):
    heap = []
    for i, lst in enumerate(lists):
        if lst:
            heapq.heappush(heap, (lst.val, i, lst))
    
    dummy = ListNode(0)
    curr = dummy
    
    while heap:
        val, i, node = heapq.heappop(heap)
        curr.next = node
        curr = curr.next
        if node.next:
            heapq.heappush(heap, (node.next.val, i, node.next))
    
    return dummy.next

def merge_k_sorted_arrays(arrays):
    """合并k个有序数组"""
    heap = []
    for i, arr in enumerate(arrays):
        if arr:
            heapq.heappush(heap, (arr[0], i, 0))
    
    result = []
    while heap:
        val, arr_idx, elem_idx = heapq.heappop(heap)
        result.append(val)
        if elem_idx + 1 < len(arrays[arr_idx]):
            next_val = arrays[arr_idx][elem_idx + 1]
            heapq.heappush(heap, (next_val, arr_idx, elem_idx + 1))
    
    return result

# 3. 数据流中的中位数
class MedianFinder:
    def __init__(self):
        self.small = []  # 最大堆（存负数）
        self.large = []  # 最小堆
    
    def addNum(self, num: int) -> None:
        # 先加入small
        heapq.heappush(self.small, -num)
        # 将small的最大值移到large
        heapq.heappush(self.large, -heapq.heappop(self.small))
        
        # 保持平衡
        if len(self.large) > len(self.small):
            heapq.heappush(self.small, -heapq.heappop(self.large))
    
    def findMedian(self) -> float:
        if len(self.small) > len(self.large):
            return -self.small[0]
        return (-self.small[0] + self.large[0]) / 2

# 4. 滑动窗口最大值（懒删除堆）
def max_sliding_window(nums, k):
    result = []
    heap = []  # (-value, index)
    
    for i, num in enumerate(nums):
        heapq.heappush(heap, (-num, i))
        
        # 移除超出窗口的元素（懒删除）
        while heap[0][1] <= i - k:
            heapq.heappop(heap)
        
        if i >= k - 1:
            result.append(-heap[0][0])
    
    return result

# 5. 任务调度器
def least_interval(tasks, n):
    """计算完成所有任务需要的最少时间"""
    from collections import Counter
    count = Counter(tasks)
    heap = [-c for c in count.values()]
    heapq.heapify(heap)
    
    time = 0
    while heap:
        temp = []
        for _ in range(n + 1):  # 一个周期
            if heap:
                cnt = heapq.heappop(heap)
                if cnt + 1 < 0:  # 还有剩余
                    temp.append(cnt + 1)
            time += 1
            if not heap and not temp:
                break
        
        for cnt in temp:
            heapq.heappush(heap, cnt)
    
    return time

# 6. Dijkstra最短路
def dijkstra(graph, start):
    """graph: {node: [(neighbor, weight), ...]}"""
    distances = {start: 0}
    heap = [(0, start)]
    
    while heap:
        dist, node = heapq.heappop(heap)
        
        if dist > distances.get(node, float('inf')):
            continue
        
        for neighbor, weight in graph[node]:
            new_dist = dist + weight
            if new_dist < distances.get(neighbor, float('inf')):
                distances[neighbor] = new_dist
                heapq.heappush(heap, (new_dist, neighbor))
    
    return distances
```

```python
# 两数之和变体：找到和最接近target的两个数
def two_sum_closest(nums, target):
    nums.sort()
    left, right = 0, len(nums) - 1
    closest = float('inf')
    result = []
    
    while left < right:
        s = nums[left] + nums[right]
        if abs(s - target) < abs(closest - target):
            closest = s
            result = [nums[left], nums[right]]
        
        if s < target:
            left += 1
        else:
            right -= 1
    
    return result

# 三数之和
def three_sum(nums):
    nums.sort()
    result = []
    
    for i in range(len(nums) - 2):
        if i > 0 and nums[i] == nums[i-1]:
            continue
        
        left, right = i + 1, len(nums) - 1
        while left < right:
            s = nums[i] + nums[left] + nums[right]
            if s == 0:
                result.append([nums[i], nums[left], nums[right]])
                while left < right and nums[left] == nums[left+1]:
                    left += 1
                while left < right and nums[right] == nums[right-1]:
                    right -= 1
                left += 1
                right -= 1
            elif s < 0:
                left += 1
            else:
                right -= 1
    
    return result
```

```python
# 重构字符串（相邻字符不同）
def reorganize_string(s):
    from collections import Counter
    count = Counter(s)
    heap = [(-c, ch) for ch, c in count.items()]
    heapq.heapify(heap)
    
    result = []
    prev = (0, '')  # 上一个字符
    
    while heap:
        cnt, ch = heapq.heappop(heap)
        result.append(ch)
        
        if prev[0] < 0:  # 上一个字符还有剩余
            heapq.heappush(heap, prev)
        
        prev = (cnt + 1, ch)  # 当前字符剩余次数
    
    return ''.join(result) if len(result) == len(s) else ''

# K个最接近原点的点
def k_closest(points, k):
    # 最大堆维护k个最小距离
    heap = []
    for x, y in points:
        dist = x*x + y*y
        if len(heap) < k:
            heapq.heappush(heap, (-dist, x, y))
        elif -dist > heap[0][0]:
            heapq.heapreplace(heap, (-dist, x, y))
    
    return [[x, y] for _, x, y in heap]
```

| 内置 | `sorted(iterable, key, reverse)` | 返回新列表 |
| 内置 | `list.sort(key, reverse)` | 原地排序 |
| bisect | `bisect_left(a, x)` | 左插入点 |
| bisect | `bisect_right(a, x)` | 右插入点 |
| bisect | `insort_left(a, x)` | 插入并保持有序 |
| heapq | `heappush(heap, item)` | 压入堆 |
| heapq | `heappop(heap)` | 弹出最小值 |
| heapq | `heapify(x)` | 列表转堆 |
| heapq | `nsmallest(n, iterable)` | n个最小值 |
| heapq | `nlargest(n, iterable)` | n个最大值 |

### 复杂度

| 操作 | 时间复杂度 |
|------|------------|
| 排序 | O(n log n) |
| 二分查找 | O(log n) |
| 堆push/pop | O(log n) |
| heapify | O(n) |
| nsmallest/nlargest | O(n log k) |

### 关键记忆

1. heapq是最小堆，最大堆用负数
2. bisect_left找第一个>=的位置，bisect_right找第一个>的位置
3. 多关键字排序用元组：`key=lambda x: (x[0], -x[1])`
4. 二分答案：check函数返回是否可行
5. 堆的元素比较用元组：`(priority, counter, data)`
