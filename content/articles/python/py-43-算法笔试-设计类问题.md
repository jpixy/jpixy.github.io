## 0.1 OrderedDict 完整API

```python
from collections import OrderedDict

"""
OrderedDict - 有序字典

继承自dict，保持元素插入顺序。Python 3.7+普通dict也保持顺序，
但OrderedDict提供额外的有序操作方法。

导入: from collections import OrderedDict

设计类问题常用场景:
    - LRU Cache
    - LFU Cache  
    - 按访问顺序维护元素
"""

# ============ 创建 ============
od = OrderedDict()                          # 空有序字典
od = OrderedDict([('a', 1), ('b', 2)])      # 从列表创建
od = OrderedDict(a=1, b=2)                  # 关键字参数创建
od = OrderedDict.fromkeys(['a', 'b'], 0)    # 从键列表创建，值都为0

# ============ 基本操作（继承自dict）============
od['key'] = value       # 设置值，O(1)
value = od['key']       # 获取值，O(1)
value = od.get('key', default)  # 获取值，不存在返回default
del od['key']           # 删除键，O(1)
'key' in od             # 判断键是否存在，O(1)
len(od)                 # 元素个数，O(1)

# ============ OrderedDict特有方法 ============

# 1. move_to_end(key, last=True) - 移动元素到末尾或开头
"""
功能: 将指定key移动到有序字典的末尾或开头
参数:
    key: 要移动的键
    last: True移到末尾(默认)，False移到开头
时间复杂度: O(1)
异常: 如果key不存在，抛出KeyError

应用: LRU Cache中标记元素为"最近使用"
"""
od = OrderedDict([('a', 1), ('b', 2), ('c', 3)])
od.move_to_end('a')          # {'b': 2, 'c': 3, 'a': 1}
od.move_to_end('a', last=False)  # {'a': 1, 'b': 2, 'c': 3}


# 2. popitem(last=True) - 弹出末尾或开头元素
"""
功能: 移除并返回(key, value)元组
参数:
    last: True弹出末尾(默认)，False弹出开头
返回: (key, value) 元组
时间复杂度: O(1)
异常: 如果字典为空，抛出KeyError

应用: 
    - LRU Cache中 popitem(last=False) 淘汰最久未使用的
    - LFU Cache中淘汰最小频率中最久的
"""
od = OrderedDict([('a', 1), ('b', 2), ('c', 3)])
od.popitem()             # 返回 ('c', 3), 剩余 {'a': 1, 'b': 2}
od.popitem(last=False)   # 返回 ('a', 1), 剩余 {'b': 2}


# 3. 迭代顺序
"""
OrderedDict保证迭代按插入顺序进行
"""
od = OrderedDict([('a', 1), ('b', 2), ('c', 3)])
list(od.keys())      # ['a', 'b', 'c']
list(od.values())    # [1, 2, 3]
list(od.items())     # [('a', 1), ('b', 2), ('c', 3)]

# 反向迭代
list(reversed(od))   # ['c', 'b', 'a']


# ============ 常见模式 ============

# 模式1: LRU Cache核心操作
class LRUPattern:
    def __init__(self, capacity):
        self.cache = OrderedDict()
        self.capacity = capacity
    
    def get(self, key):
        if key not in self.cache:
            return -1
        self.cache.move_to_end(key)  # 标记为最近使用
        return self.cache[key]
    
    def put(self, key, value):
        if key in self.cache:
            self.cache.move_to_end(key)
        self.cache[key] = value
        if len(self.cache) > self.capacity:
            self.cache.popitem(last=False)  # 淘汰最久未使用


# 模式2: 按频率分组的有序集合（LFU用）
freq_to_keys = defaultdict(OrderedDict)
freq_to_keys[1]['key1'] = None  # 使用OrderedDict当有序集合
freq_to_keys[1]['key2'] = None
# popitem(last=False) 获取最早插入的key
```

# 一、缓存设计

## 1.1 LRU Cache

```python
from collections import OrderedDict

class LRUCache:
    """
    LeetCode 146. LRU Cache
    
    最近最少使用缓存
    
    功能:
        - get(key): 获取键对应的值，不存在返回-1
        - put(key, value): 设置键值对，超出容量时淘汰最久未使用的
    
    时间复杂度: get和put都是O(1)
    空间复杂度: O(capacity)
    
    实现方式:
        方法1: OrderedDict（推荐，简洁）
        方法2: 哈希表 + 双向链表（面试常考）
    """
    
    def __init__(self, capacity: int):
        """
        初始化LRU缓存
        
        参数:
            capacity: int - 缓存容量
        """
        self.cache = OrderedDict()
        self.capacity = capacity
    
    def get(self, key: int) -> int:
        """
        获取键对应的值
        
        参数:
            key: int - 键
        返回:
            int - 值，不存在返回-1
        
        操作:
            如果key存在，将其移到末尾（表示最近使用）
        """
        if key not in self.cache:
            return -1
        
        # 移到末尾表示最近使用
        self.cache.move_to_end(key)
        return self.cache[key]
    
    def put(self, key: int, value: int) -> None:
        """
        设置键值对
        
        参数:
            key: int - 键
            value: int - 值
        
        操作:
            1. 如果key已存在，更新值并移到末尾
            2. 如果不存在，添加到末尾
            3. 如果超出容量，删除最久未使用的（开头）
        """
        if key in self.cache:
            self.cache.move_to_end(key)
        
        self.cache[key] = value
        
        if len(self.cache) > self.capacity:
            # 删除最久未使用的（第一个）
            self.cache.popitem(last=False)


# 使用示例
cache = LRUCache(2)
cache.put(1, 1)       # cache: {1=1}
cache.put(2, 2)       # cache: {1=1, 2=2}
cache.get(1)          # 返回1, cache: {2=2, 1=1}
cache.put(3, 3)       # 淘汰key=2, cache: {1=1, 3=3}
cache.get(2)          # 返回-1
cache.put(4, 4)       # 淘汰key=1, cache: {3=3, 4=4}
cache.get(1)          # 返回-1
cache.get(3)          # 返回3
cache.get(4)          # 返回4
```

### LRU Cache - 双向链表实现

```python
class DLinkedNode:
    """双向链表节点"""
    def __init__(self, key=0, value=0):
        self.key = key
        self.value = value
        self.prev = None
        self.next = None


class LRUCacheLinkedList:
    """
    LRU Cache - 哈希表 + 双向链表实现
    
    结构:
        - 哈希表: key -> DLinkedNode
        - 双向链表: 头部是最久未使用，尾部是最近使用
    
    这是面试中更常考的实现方式
    """
    
    def __init__(self, capacity: int):
        self.cache = {}  # key -> DLinkedNode
        self.capacity = capacity
        self.size = 0
        
        # 使用虚拟头尾节点简化操作
        self.head = DLinkedNode()
        self.tail = DLinkedNode()
        self.head.next = self.tail
        self.tail.prev = self.head
    
    def get(self, key: int) -> int:
        if key not in self.cache:
            return -1
        
        node = self.cache[key]
        # 移到尾部（最近使用）
        self._move_to_tail(node)
        return node.value
    
    def put(self, key: int, value: int) -> None:
        if key in self.cache:
            node = self.cache[key]
            node.value = value
            self._move_to_tail(node)
        else:
            node = DLinkedNode(key, value)
            self.cache[key] = node
            self._add_to_tail(node)
            self.size += 1
            
            if self.size > self.capacity:
                # 删除头部节点（最久未使用）
                removed = self._remove_head()
                del self.cache[removed.key]
                self.size -= 1
    
    def _add_to_tail(self, node: DLinkedNode) -> None:
        """添加节点到尾部"""
        node.prev = self.tail.prev
        node.next = self.tail
        self.tail.prev.next = node
        self.tail.prev = node
    
    def _remove_node(self, node: DLinkedNode) -> None:
        """从链表中删除节点"""
        node.prev.next = node.next
        node.next.prev = node.prev
    
    def _move_to_tail(self, node: DLinkedNode) -> None:
        """将节点移到尾部"""
        self._remove_node(node)
        self._add_to_tail(node)
    
    def _remove_head(self) -> DLinkedNode:
        """删除并返回头部节点"""
        node = self.head.next
        self._remove_node(node)
        return node
```

## 1.2 LFU Cache

```python
from collections import defaultdict, OrderedDict

class LFUCache:
    """
    LeetCode 460. LFU Cache
    
    最不经常使用缓存
    
    功能:
        - get(key): 获取键对应的值，使用次数+1
        - put(key, value): 设置键值对，使用次数初始为1
        - 淘汰策略: 淘汰使用次数最少的，次数相同淘汰最久未使用的
    
    时间复杂度: get和put都是O(1)
    空间复杂度: O(capacity)
    
    实现:
        - key_to_val: 存储键值
        - key_to_freq: 存储键的使用频率
        - freq_to_keys: 存储每个频率对应的键集合（OrderedDict保持LRU顺序）
        - min_freq: 当前最小频率
    """
    
    def __init__(self, capacity: int):
        """
        初始化LFU缓存
        
        参数:
            capacity: int - 缓存容量
        """
        self.capacity = capacity
        self.key_to_val = {}                    # key -> value
        self.key_to_freq = {}                   # key -> frequency
        self.freq_to_keys = defaultdict(OrderedDict)  # freq -> OrderedDict of keys
        self.min_freq = 0
    
    def get(self, key: int) -> int:
        """
        获取键对应的值
        
        返回:
            int - 值，不存在返回-1
        
        副作用:
            如果key存在，其使用频率+1
        """
        if key not in self.key_to_val:
            return -1
        
        self._increase_freq(key)
        return self.key_to_val[key]
    
    def put(self, key: int, value: int) -> None:
        """
        设置键值对
        
        参数:
            key: int - 键
            value: int - 值
        """
        if self.capacity <= 0:
            return
        
        if key in self.key_to_val:
            self.key_to_val[key] = value
            self._increase_freq(key)
            return
        
        # 容量满了，淘汰
        if len(self.key_to_val) >= self.capacity:
            self._remove_min_freq_key()
        
        # 插入新键
        self.key_to_val[key] = value
        self.key_to_freq[key] = 1
        self.freq_to_keys[1][key] = None  # OrderedDict用于保持LRU顺序
        self.min_freq = 1
    
    def _increase_freq(self, key: int) -> None:
        """增加key的使用频率"""
        freq = self.key_to_freq[key]
        self.key_to_freq[key] = freq + 1
        
        # 从旧频率集合中删除
        del self.freq_to_keys[freq][key]
        
        # 如果旧频率集合为空，可能需要更新min_freq
        if not self.freq_to_keys[freq]:
            del self.freq_to_keys[freq]
            if self.min_freq == freq:
                self.min_freq = freq + 1
        
        # 添加到新频率集合
        self.freq_to_keys[freq + 1][key] = None
    
    def _remove_min_freq_key(self) -> None:
        """删除使用频率最低且最久未使用的key"""
        # 获取最小频率集合中的第一个key（最久未使用）
        key, _ = self.freq_to_keys[self.min_freq].popitem(last=False)
        
        if not self.freq_to_keys[self.min_freq]:
            del self.freq_to_keys[self.min_freq]
        
        del self.key_to_val[key]
        del self.key_to_freq[key]


# 使用示例
cache = LFUCache(2)
cache.put(1, 1)       # freq: {1: [1]}
cache.put(2, 2)       # freq: {1: [1, 2]}
cache.get(1)          # 返回1, freq: {1: [2], 2: [1]}
cache.put(3, 3)       # 淘汰key=2, freq: {1: [3], 2: [1]}
cache.get(2)          # 返回-1
cache.get(3)          # 返回3, freq: {2: [1, 3]}
cache.put(4, 4)       # 淘汰key=1, freq: {1: [4], 2: [3]}
cache.get(1)          # 返回-1
cache.get(3)          # 返回3
cache.get(4)          # 返回4
```

## 1.3 TTL Cache

```python
import time
from collections import OrderedDict

class TTLCache:
    """
    带过期时间的缓存
    
    功能:
        - get(key): 获取值，过期返回None
        - put(key, value, ttl): 设置值和过期时间
    
    应用场景:
        - 会话管理
        - API限流
        - 临时数据存储
    """
    
    def __init__(self, capacity: int):
        """
        参数:
            capacity: int - 缓存容量
        """
        self.capacity = capacity
        self.cache = OrderedDict()  # key -> (value, expire_time)
    
    def get(self, key: str):
        """
        获取值
        
        返回:
            value或None（过期或不存在）
        """
        if key not in self.cache:
            return None
        
        value, expire_time = self.cache[key]
        
        if time.time() > expire_time:
            del self.cache[key]
            return None
        
        # 移到末尾（LRU）
        self.cache.move_to_end(key)
        return value
    
    def put(self, key: str, value, ttl: float) -> None:
        """
        设置值
        
        参数:
            key: str - 键
            value: any - 值
            ttl: float - 过期时间（秒）
        """
        expire_time = time.time() + ttl
        
        if key in self.cache:
            self.cache.move_to_end(key)
        
        self.cache[key] = (value, expire_time)
        
        # 清理过期和超出容量的
        self._cleanup()
    
    def _cleanup(self) -> None:
        """清理过期项和超出容量的项"""
        current_time = time.time()
        
        # 删除过期的
        expired_keys = [
            k for k, (_, exp) in self.cache.items() 
            if current_time > exp
        ]
        for k in expired_keys:
            del self.cache[k]
        
        # 删除超出容量的
        while len(self.cache) > self.capacity:
            self.cache.popitem(last=False)
```

## 2.1 最小栈

```python
class MinStack:
    """
    LeetCode 155. Min Stack
    
    支持O(1)获取最小值的栈
    
    功能:
        - push(val): 压入元素
        - pop(): 弹出栈顶
        - top(): 获取栈顶
        - getMin(): 获取最小值
    
    时间复杂度: 所有操作O(1)
    空间复杂度: O(n)
    
    实现:
        使用辅助栈存储每个状态的最小值
    """
    
    def __init__(self):
        """初始化栈"""
        self.stack = []
        self.min_stack = []  # 存储每个状态的最小值
    
    def push(self, val: int) -> None:
        """
        压入元素
        
        参数:
            val: int - 要压入的值
        """
        self.stack.append(val)
        
        # 更新最小值栈
        if not self.min_stack or val <= self.min_stack[-1]:
            self.min_stack.append(val)
        else:
            self.min_stack.append(self.min_stack[-1])
    
    def pop(self) -> None:
        """弹出栈顶元素"""
        self.stack.pop()
        self.min_stack.pop()
    
    def top(self) -> int:
        """
        获取栈顶元素
        
        返回:
            int - 栈顶元素
        """
        return self.stack[-1]
    
    def getMin(self) -> int:
        """
        获取栈中最小元素
        
        返回:
            int - 最小元素
        """
        return self.min_stack[-1]


class MinStackOptimized:
    """
    最小栈 - 空间优化版
    
    只在新的最小值出现时才压入min_stack
    """
    
    def __init__(self):
        self.stack = []
        self.min_stack = []
    
    def push(self, val: int) -> None:
        self.stack.append(val)
        # 只有新的最小值才压入
        if not self.min_stack or val <= self.min_stack[-1]:
            self.min_stack.append(val)
    
    def pop(self) -> None:
        val = self.stack.pop()
        # 如果弹出的是最小值，也从min_stack弹出
        if val == self.min_stack[-1]:
            self.min_stack.pop()
    
    def top(self) -> int:
        return self.stack[-1]
    
    def getMin(self) -> int:
        return self.min_stack[-1]
```

## 2.2 用栈实现队列

```python
class MyQueue:
    """
    LeetCode 232. Implement Queue using Stacks
    
    用两个栈实现队列
    
    功能:
        - push(x): 入队
        - pop(): 出队
        - peek(): 查看队首
        - empty(): 是否为空
    
    时间复杂度:
        - push: O(1)
        - pop: 均摊O(1)
        - peek: 均摊O(1)
    
    实现:
        - in_stack: 用于入队
        - out_stack: 用于出队
        - 当out_stack为空时，将in_stack全部倒入out_stack
    """
    
    def __init__(self):
        """初始化队列"""
        self.in_stack = []   # 入队栈
        self.out_stack = []  # 出队栈
    
    def push(self, x: int) -> None:
        """
        入队
        
        参数:
            x: int - 要入队的元素
        """
        self.in_stack.append(x)
    
    def pop(self) -> int:
        """
        出队
        
        返回:
            int - 队首元素
        """
        self._transfer()
        return self.out_stack.pop()
    
    def peek(self) -> int:
        """
        查看队首元素
        
        返回:
            int - 队首元素（不移除）
        """
        self._transfer()
        return self.out_stack[-1]
    
    def empty(self) -> bool:
        """
        判断队列是否为空
        
        返回:
            bool - 是否为空
        """
        return not self.in_stack and not self.out_stack
    
    def _transfer(self) -> None:
        """将in_stack转移到out_stack"""
        if not self.out_stack:
            while self.in_stack:
                self.out_stack.append(self.in_stack.pop())
```

## 2.4 最大栈

```python
import heapq
from collections import defaultdict

class MaxStack:
    """
    LeetCode 716. Max Stack
    
    支持O(log n)获取和弹出最大值的栈
    
    功能:
        - push(x): 压栈
        - pop(): 弹栈
        - top(): 查看栈顶
        - peekMax(): 查看最大值
        - popMax(): 弹出最大值
    
    实现:
        - 栈 + 最大堆 + 懒删除
    """
    
    def __init__(self):
        self.stack = []      # 普通栈
        self.heap = []       # 最大堆（存负数）
        self.removed = defaultdict(int)  # 懒删除标记
        self.cnt = 0         # 操作计数器（用于区分相同值）
    
    def push(self, x: int) -> None:
        """压栈"""
        self.stack.append((x, self.cnt))
        heapq.heappush(self.heap, (-x, -self.cnt))
        self.cnt += 1
    
    def pop(self) -> int:
        """弹栈"""
        self._clean_stack()
        val, idx = self.stack.pop()
        self.removed[(val, idx)] += 1
        return val
    
    def top(self) -> int:
        """查看栈顶"""
        self._clean_stack()
        return self.stack[-1][0]
    
    def peekMax(self) -> int:
        """查看最大值"""
        self._clean_heap()
        return -self.heap[0][0]
    
    def popMax(self) -> int:
        """弹出最大值"""
        self._clean_heap()
        val, idx = heapq.heappop(self.heap)
        self.removed[(-val, -idx)] += 1
        return -val
    
    def _clean_stack(self) -> None:
        """清理栈顶已删除的元素"""
        while self.stack and self.removed[self.stack[-1]] > 0:
            self.removed[self.stack[-1]] -= 1
            self.stack.pop()
    
    def _clean_heap(self) -> None:
        """清理堆顶已删除的元素"""
        while self.heap:
            val, idx = self.heap[0]
            if self.removed[(-val, -idx)] > 0:
                self.removed[(-val, -idx)] -= 1
                heapq.heappop(self.heap)
            else:
                break
```

```python
import random

class RandomizedSet:
    """
    LeetCode 380. Insert Delete GetRandom O(1)
    
    支持O(1)插入、删除、随机获取的集合
    
    功能:
        - insert(val): 插入元素，已存在返回False
        - remove(val): 删除元素，不存在返回False
        - getRandom(): 随机返回一个元素
    
    时间复杂度: 所有操作O(1)
    
    实现:
        - 列表存储元素（支持O(1)随机访问）
        - 哈希表存储元素到索引的映射
        - 删除时将要删除的元素与最后一个元素交换
    """
    
    def __init__(self):
        """初始化集合"""
        self.val_to_idx = {}  # 值 -> 索引
        self.vals = []        # 存储所有值
    
    def insert(self, val: int) -> bool:
        """
        插入元素
        
        参数:
            val: int - 要插入的值
        返回:
            bool - True表示插入成功，False表示已存在
        """
        if val in self.val_to_idx:
            return False
        
        self.val_to_idx[val] = len(self.vals)
        self.vals.append(val)
        return True
    
    def remove(self, val: int) -> bool:
        """
        删除元素
        
        参数:
            val: int - 要删除的值
        返回:
            bool - True表示删除成功，False表示不存在
        """
        if val not in self.val_to_idx:
            return False
        
        # 将要删除的元素与最后一个交换
        idx = self.val_to_idx[val]
        last_val = self.vals[-1]
        
        self.vals[idx] = last_val
        self.val_to_idx[last_val] = idx
        
        self.vals.pop()
        del self.val_to_idx[val]
        
        return True
    
    def getRandom(self) -> int:
        """
        随机返回一个元素
        
        返回:
            int - 随机选择的元素
        """
        return random.choice(self.vals)


class RandomizedCollection:
    """
    LeetCode 381. Insert Delete GetRandom O(1) - Duplicates allowed
    
    允许重复元素的随机集合
    """
    
    def __init__(self):
        self.val_to_indices = defaultdict(set)  # 值 -> 索引集合
        self.vals = []
    
    def insert(self, val: int) -> bool:
        """插入元素，返回是否是新元素"""
        is_new = val not in self.val_to_indices or len(self.val_to_indices[val]) == 0
        
        self.val_to_indices[val].add(len(self.vals))
        self.vals.append(val)
        
        return is_new
    
    def remove(self, val: int) -> bool:
        """删除一个val"""
        if val not in self.val_to_indices or len(self.val_to_indices[val]) == 0:
            return False
        
        # 获取要删除的索引
        idx = self.val_to_indices[val].pop()
        last_val = self.vals[-1]
        
        if idx != len(self.vals) - 1:
            # 用最后一个元素替换
            self.vals[idx] = last_val
            self.val_to_indices[last_val].add(idx)
            self.val_to_indices[last_val].discard(len(self.vals) - 1)
        
        self.vals.pop()
        return True
    
    def getRandom(self) -> int:
        return random.choice(self.vals)
```

```python
import bisect
from collections import defaultdict

class TimeMap:
    """
    LeetCode 981. Time Based Key-Value Store
    
    时间戳键值存储
    
    功能:
        - set(key, value, timestamp): 存储键值对和时间戳
        - get(key, timestamp): 获取<=timestamp的最新值
    
    时间复杂度:
        - set: O(1)
        - get: O(log n)
    """
    
    def __init__(self):
        """初始化"""
        self.store = defaultdict(list)  # key -> [(timestamp, value), ...]
    
    def set(self, key: str, value: str, timestamp: int) -> None:
        """
        存储键值对
        
        参数:
            key: str - 键
            value: str - 值
            timestamp: int - 时间戳
        """
        self.store[key].append((timestamp, value))
    
    def get(self, key: str, timestamp: int) -> str:
        """
        获取值
        
        参数:
            key: str - 键
            timestamp: int - 时间戳
        返回:
            str - <=timestamp的最新值，不存在返回空字符串
        """
        if key not in self.store:
            return ""
        
        values = self.store[key]
        # 二分查找
        idx = bisect.bisect_right(values, (timestamp, chr(127)))
        
        if idx == 0:
            return ""
        
        return values[idx - 1][1]
```

## 3.3 全O(1)数据结构

```python
class AllOne:
    """
    LeetCode 432. All O(1) Data Structure
    
    支持O(1)增减计数和获取最大最小key的数据结构
    
    功能:
        - inc(key): key计数+1
        - dec(key): key计数-1
        - getMaxKey(): 返回计数最大的key
        - getMinKey(): 返回计数最小的key
    
    实现:
        - 哈希表 + 双向链表
        - 链表按计数排序，每个节点存储相同计数的所有key
    """
    
    class Node:
        def __init__(self, count=0):
            self.count = count
            self.keys = set()
            self.prev = None
            self.next = None
    
    def __init__(self):
        self.key_to_node = {}  # key -> Node
        # 虚拟头尾
        self.head = self.Node(0)
        self.tail = self.Node(float('inf'))
        self.head.next = self.tail
        self.tail.prev = self.head
    
    def _insert_after(self, node, new_node):
        """在node后插入new_node"""
        new_node.prev = node
        new_node.next = node.next
        node.next.prev = new_node
        node.next = new_node
    
    def _remove_node(self, node):
        """删除节点"""
        node.prev.next = node.next
        node.next.prev = node.prev
    
    def inc(self, key: str) -> None:
        """增加计数"""
        if key in self.key_to_node:
            node = self.key_to_node[key]
            node.keys.remove(key)
            new_count = node.count + 1
            
            # 找或创建新计数节点
            if node.next.count != new_count:
                new_node = self.Node(new_count)
                self._insert_after(node, new_node)
            else:
                new_node = node.next
            
            new_node.keys.add(key)
            self.key_to_node[key] = new_node
            
            if not node.keys:
                self._remove_node(node)
        else:
            # 新key
            if self.head.next.count != 1:
                new_node = self.Node(1)
                self._insert_after(self.head, new_node)
            else:
                new_node = self.head.next
            
            new_node.keys.add(key)
            self.key_to_node[key] = new_node
    
    def dec(self, key: str) -> None:
        """减少计数"""
        if key not in self.key_to_node:
            return
        
        node = self.key_to_node[key]
        node.keys.remove(key)
        new_count = node.count - 1
        
        if new_count == 0:
            del self.key_to_node[key]
        else:
            if node.prev.count != new_count:
                new_node = self.Node(new_count)
                self._insert_after(node.prev, new_node)
            else:
                new_node = node.prev
            
            new_node.keys.add(key)
            self.key_to_node[key] = new_node
        
        if not node.keys:
            self._remove_node(node)
    
    def getMaxKey(self) -> str:
        """获取最大计数的key"""
        if self.tail.prev == self.head:
            return ""
        return next(iter(self.tail.prev.keys))
    
    def getMinKey(self) -> str:
        """获取最小计数的key"""
        if self.head.next == self.tail:
            return ""
        return next(iter(self.head.next.keys))
```

```python
class NestedIterator:
    """
    LeetCode 341. Flatten Nested List Iterator
    
    扁平化嵌套列表的迭代器
    
    功能:
        - next(): 返回下一个整数
        - hasNext(): 是否还有下一个
    
    实现:
        使用栈存储迭代器状态
    """
    
    def __init__(self, nestedList):
        """
        参数:
            nestedList: List[NestedInteger] - 嵌套列表
        """
        self.stack = []
        self._push_list(nestedList)
    
    def _push_list(self, lst):
        """将列表元素逆序压入栈"""
        for i in range(len(lst) - 1, -1, -1):
            self.stack.append(lst[i])
    
    def next(self) -> int:
        """返回下一个整数"""
        self.hasNext()  # 确保栈顶是整数
        return self.stack.pop().getInteger()
    
    def hasNext(self) -> bool:
        """是否还有下一个"""
        while self.stack:
            top = self.stack[-1]
            if top.isInteger():
                return True
            # 展开列表
            self.stack.pop()
            self._push_list(top.getList())
        return False


class NestedIteratorSimple:
    """
    简单实现：预处理展平
    """
    
    def __init__(self, nestedList):
        self.values = []
        self._flatten(nestedList)
        self.idx = 0
    
    def _flatten(self, lst):
        for item in lst:
            if item.isInteger():
                self.values.append(item.getInteger())
            else:
                self._flatten(item.getList())
    
    def next(self) -> int:
        val = self.values[self.idx]
        self.idx += 1
        return val
    
    def hasNext(self) -> bool:
        return self.idx < len(self.values)
```

```python
class PeekingIterator:
    """
    LeetCode 284. Peeking Iterator
    
    支持peek操作的迭代器
    
    功能:
        - peek(): 查看下一个元素但不移动
        - next(): 返回下一个元素并移动
        - hasNext(): 是否还有下一个
    """
    
    def __init__(self, iterator):
        """
        参数:
            iterator: Iterator - 原始迭代器
        """
        self.iterator = iterator
        self.peeked = None
        self.has_peeked = False
    
    def peek(self) -> int:
        """
        查看下一个元素
        
        返回:
            int - 下一个元素（不移除）
        """
        if not self.has_peeked:
            self.peeked = self.iterator.next()
            self.has_peeked = True
        return self.peeked
    
    def next(self) -> int:
        """
        返回并移除下一个元素
        """
        if self.has_peeked:
            self.has_peeked = False
            return self.peeked
        return self.iterator.next()
    
    def hasNext(self) -> bool:
        """是否还有下一个"""
        return self.has_peeked or self.iterator.hasNext()
```

```python
class BSTIterator:
    """
    LeetCode 173. Binary Search Tree Iterator
    
    二叉搜索树的中序遍历迭代器
    
    功能:
        - next(): 返回下一个最小值
        - hasNext(): 是否还有下一个
    
    时间复杂度:
        - next(): 均摊O(1)
        - hasNext(): O(1)
    
    空间复杂度: O(h)，h是树高
    """
    
    def __init__(self, root):
        """
        参数:
            root: TreeNode - BST根节点
        """
        self.stack = []
        self._push_left(root)
    
    def _push_left(self, node):
        """将节点及其所有左子节点压入栈"""
        while node:
            self.stack.append(node)
            node = node.left
    
    def next(self) -> int:
        """返回下一个最小值"""
        node = self.stack.pop()
        self._push_left(node.right)
        return node.val
    
    def hasNext(self) -> bool:
        """是否还有下一个"""
        return len(self.stack) > 0
```

## 5.1 LRU with TTL

```python
import time
from collections import OrderedDict

class LRUCacheWithTTL:
    """
    带过期时间的LRU缓存
    
    结合LRU淘汰策略和TTL过期机制
    """
    
    def __init__(self, capacity: int, default_ttl: float = 300):
        """
        参数:
            capacity: int - 容量
            default_ttl: float - 默认过期时间（秒）
        """
        self.capacity = capacity
        self.default_ttl = default_ttl
        self.cache = OrderedDict()  # key -> (value, expire_time)
    
    def get(self, key: str):
        """获取值"""
        if key not in self.cache:
            return None
        
        value, expire_time = self.cache[key]
        
        if time.time() > expire_time:
            del self.cache[key]
            return None
        
        self.cache.move_to_end(key)
        return value
    
    def put(self, key: str, value, ttl: float = None) -> None:
        """设置值"""
        if ttl is None:
            ttl = self.default_ttl
        
        expire_time = time.time() + ttl
        
        if key in self.cache:
            self.cache.move_to_end(key)
        
        self.cache[key] = (value, expire_time)
        
        self._evict()
    
    def _evict(self) -> None:
        """淘汰过期和超出容量的项"""
        current_time = time.time()
        
        # 删除过期的
        expired = [k for k, (_, exp) in self.cache.items() if current_time > exp]
        for k in expired:
            del self.cache[k]
        
        # 淘汰超出容量的
        while len(self.cache) > self.capacity:
            self.cache.popitem(last=False)
```

## 5.2 Twitter设计

```python
from collections import defaultdict
import heapq

class Twitter:
    """
    LeetCode 355. Design Twitter
    
    简化版Twitter设计
    
    功能:
        - postTweet(userId, tweetId): 发推文
        - getNewsFeed(userId): 获取最新10条推文
        - follow(followerId, followeeId): 关注
        - unfollow(followerId, followeeId): 取消关注
    """
    
    def __init__(self):
        self.tweets = defaultdict(list)    # userId -> [(timestamp, tweetId), ...]
        self.following = defaultdict(set)  # userId -> set of followeeIds
        self.timestamp = 0
    
    def postTweet(self, userId: int, tweetId: int) -> None:
        """发推文"""
        self.tweets[userId].append((self.timestamp, tweetId))
        self.timestamp += 1
    
    def getNewsFeed(self, userId: int) -> list:
        """获取最新10条推文"""
        # 合并自己和关注者的推文
        users = self.following[userId] | {userId}
        
        # 使用堆合并
        heap = []
        for user in users:
            user_tweets = self.tweets[user]
            if user_tweets:
                idx = len(user_tweets) - 1
                time, tweet_id = user_tweets[idx]
                heapq.heappush(heap, (-time, tweet_id, user, idx))
        
        result = []
        while heap and len(result) < 10:
            time, tweet_id, user, idx = heapq.heappop(heap)
            result.append(tweet_id)
            
            if idx > 0:
                idx -= 1
                time, tweet_id = self.tweets[user][idx]
                heapq.heappush(heap, (-time, tweet_id, user, idx))
        
        return result
    
    def follow(self, followerId: int, followeeId: int) -> None:
        """关注"""
        if followerId != followeeId:
            self.following[followerId].add(followeeId)
    
    def unfollow(self, followerId: int, followeeId: int) -> None:
        """取消关注"""
        self.following[followerId].discard(followeeId)
```

```python
import heapq

class MedianFinder:
    """
    LeetCode 295. Find Median from Data Stream
    
    数据流中位数
    
    功能:
        - addNum(num): 添加数字
        - findMedian(): 返回中位数
    
    时间复杂度:
        - addNum: O(log n)
        - findMedian: O(1)
    
    实现:
        - 最大堆存储较小的一半
        - 最小堆存储较大的一半
        - 保持两个堆大小平衡
    """
    
    def __init__(self):
        """初始化"""
        self.small = []  # 最大堆（存负数）
        self.large = []  # 最小堆
    
    def addNum(self, num: int) -> None:
        """
        添加数字
        
        参数:
            num: int - 要添加的数字
        """
        # 先加入small
        heapq.heappush(self.small, -num)
        
        # 将small的最大值移到large
        heapq.heappush(self.large, -heapq.heappop(self.small))
        
        # 保持平衡：small可以比large多一个
        if len(self.large) > len(self.small):
            heapq.heappush(self.small, -heapq.heappop(self.large))
    
    def findMedian(self) -> float:
        """
        返回中位数
        
        返回:
            float - 当前所有数字的中位数
        """
        if len(self.small) > len(self.large):
            return -self.small[0]
        return (-self.small[0] + self.large[0]) / 2
```

## 总结

### 设计问题分类

| 类型 | 问题 | 关键数据结构 |
|------|------|-------------|
| 缓存 | LRU, LFU, TTL | OrderedDict, 双向链表+哈希表 |
| 栈队列 | MinStack, 栈实现队列 | 辅助栈, 双栈 |
| 集合 | RandomizedSet | 数组+哈希表 |
| 迭代器 | BST迭代器, 扁平化 | 栈, 惰性求值 |
| 流处理 | 数据流中位数 | 双堆 |

### 设计要点

| 要求 | 实现技巧 |
|------|----------|
| O(1)查找 | 哈希表 |
| O(1)删除 | 与末尾交换 |
| O(1)最值 | 辅助栈/堆 |
| 有序维护 | OrderedDict, 双向链表 |
| 惰性删除 | 标记+延迟清理 |

### 关键记忆

1. **LRU**: OrderedDict.move_to_end() + popitem(last=False)
2. **LFU**: 三个哈希表 + min_freq
3. **MinStack**: 辅助栈记录每个状态的最小值
4. **RandomizedSet**: 数组+哈希表，删除时与末尾交换
5. **数据流中位数**: 大顶堆(小数) + 小顶堆(大数)
