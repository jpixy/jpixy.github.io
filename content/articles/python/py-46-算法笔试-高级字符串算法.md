# 一、字符串匹配算法

## 1.1 KMP算法

```python
def compute_lps(pattern: str) -> list:
    """
    计算KMP的最长前缀后缀数组（LPS/Failure Function）
    
    参数:
        pattern: str - 模式串
    返回:
        list - lps[i]表示pattern[0:i+1]的最长相等前后缀长度
    
    时间复杂度: O(m)，m是模式串长度
    
    示例:
        pattern = "ABABC"
        lps = [0, 0, 1, 2, 0]
        解释:
            lps[0] = 0  # "A" 无真前后缀
            lps[1] = 0  # "AB" 无相等前后缀
            lps[2] = 1  # "ABA" -> "A" = "A"
            lps[3] = 2  # "ABAB" -> "AB" = "AB"
            lps[4] = 0  # "ABABC" 无相等前后缀
    """
    m = len(pattern)
    lps = [0] * m
    
    length = 0  # 当前最长相等前后缀长度
    i = 1
    
    while i < m:
        if pattern[i] == pattern[length]:
            length += 1
            lps[i] = length
            i += 1
        else:
            if length != 0:
                # 回退到上一个可能的前缀
                length = lps[length - 1]
            else:
                lps[i] = 0
                i += 1
    
    return lps


def kmp_search(text: str, pattern: str) -> list:
    """
    KMP字符串匹配算法
    
    LeetCode 28. Find the Index of the First Occurrence in a String
    
    参数:
        text: str - 文本串
        pattern: str - 模式串
    返回:
        list - 所有匹配位置的起始索引
    
    时间复杂度: O(n + m)
    空间复杂度: O(m)
    
    核心思想:
        利用已匹配的信息，避免从头开始匹配
    """
    if not pattern:
        return [0]
    if not text:
        return []
    
    n, m = len(text), len(pattern)
    lps = compute_lps(pattern)
    
    result = []
    i = 0  # text指针
    j = 0  # pattern指针
    
    while i < n:
        if text[i] == pattern[j]:
            i += 1
            j += 1
            
            if j == m:
                # 找到匹配
                result.append(i - j)
                j = lps[j - 1]  # 继续寻找下一个匹配
        else:
            if j != 0:
                j = lps[j - 1]  # 利用lps回退
            else:
                i += 1
    
    return result


def kmp_first_match(text: str, pattern: str) -> int:
    """
    KMP查找第一个匹配位置
    
    返回:
        int - 第一个匹配的起始位置，不存在返回-1
    """
    matches = kmp_search(text, pattern)
    return matches[0] if matches else -1


# 使用示例
text = "ABABDABACDABABCABAB"
pattern = "ABABC"

print(f"LPS数组: {compute_lps(pattern)}")
print(f"匹配位置: {kmp_search(text, pattern)}")
```

## 1.2 Rabin-Karp算法

```python
def rabin_karp(text: str, pattern: str, base: int = 256, 
               mod: int = 10**9 + 7) -> list:
    """
    Rabin-Karp字符串匹配算法（滚动哈希）
    
    参数:
        text: str - 文本串
        pattern: str - 模式串
        base: int - 哈希基数
        mod: int - 模数（大质数）
    返回:
        list - 所有匹配位置
    
    时间复杂度: 平均O(n+m)，最坏O(nm)
    空间复杂度: O(1)
    
    核心思想:
        使用滚动哈希，O(1)时间更新窗口哈希值
    """
    n, m = len(text), len(pattern)
    if m > n:
        return []
    
    result = []
    
    # 计算base^(m-1) % mod
    h = pow(base, m - 1, mod)
    
    # 计算模式串哈希
    pattern_hash = 0
    for c in pattern:
        pattern_hash = (pattern_hash * base + ord(c)) % mod
    
    # 计算初始窗口哈希
    window_hash = 0
    for i in range(m):
        window_hash = (window_hash * base + ord(text[i])) % mod
    
    # 滑动窗口
    for i in range(n - m + 1):
        # 哈希匹配时验证
        if window_hash == pattern_hash:
            if text[i:i+m] == pattern:  # 防止哈希冲突
                result.append(i)
        
        # 滚动更新哈希（如果还有下一个窗口）
        if i < n - m:
            window_hash = (
                (window_hash - ord(text[i]) * h) * base + 
                ord(text[i + m])
            ) % mod
    
    return result


def rabin_karp_multiple(text: str, patterns: list) -> dict:
    """
    Rabin-Karp多模式匹配
    
    参数:
        text: str - 文本串
        patterns: list - 模式串列表
    返回:
        dict - 每个模式串的匹配位置
    
    适用场景: 多个相同长度的模式串
    """
    if not patterns:
        return {}
    
    base = 256
    mod = 10**9 + 7
    n = len(text)
    
    # 按长度分组
    from collections import defaultdict
    length_groups = defaultdict(list)
    for p in patterns:
        length_groups[len(p)].append(p)
    
    result = {p: [] for p in patterns}
    
    for m, group in length_groups.items():
        if m > n:
            continue
        
        # 计算模式串哈希
        pattern_hashes = {}
        for p in group:
            h = 0
            for c in p:
                h = (h * base + ord(c)) % mod
            if h not in pattern_hashes:
                pattern_hashes[h] = []
            pattern_hashes[h].append(p)
        
        # 滚动哈希匹配
        h_power = pow(base, m - 1, mod)
        window_hash = 0
        
        for i in range(m):
            window_hash = (window_hash * base + ord(text[i])) % mod
        
        for i in range(n - m + 1):
            if window_hash in pattern_hashes:
                window = text[i:i+m]
                for p in pattern_hashes[window_hash]:
                    if window == p:
                        result[p].append(i)
            
            if i < n - m:
                window_hash = (
                    (window_hash - ord(text[i]) * h_power) * base +
                    ord(text[i + m])
                ) % mod
    
    return result
```

## 1.3 字符串哈希

```python
class StringHash:
    """
    字符串哈希（预处理后O(1)查询子串哈希）
    
    应用:
        - 快速比较子串是否相等
        - 最长公共子串
        - 回文判断
    """
    
    def __init__(self, s: str, base: int = 131, mod: int = 10**18 + 9):
        """
        参数:
            s: str - 原字符串
            base: int - 哈希基数（常用131, 1331）
            mod: int - 模数（大质数）
        
        预处理时间: O(n)
        """
        self.n = len(s)
        self.base = base
        self.mod = mod
        
        # 前缀哈希
        self.prefix = [0] * (self.n + 1)
        # base的幂次
        self.power = [1] * (self.n + 1)
        
        for i in range(self.n):
            self.prefix[i + 1] = (self.prefix[i] * base + ord(s[i])) % mod
            self.power[i + 1] = self.power[i] * base % mod
    
    def get_hash(self, left: int, right: int) -> int:
        """
        获取s[left:right+1]的哈希值
        
        参数:
            left, right: int - 子串的起止索引（闭区间）
        返回:
            int - 子串哈希值
        
        时间复杂度: O(1)
        """
        return (self.prefix[right + 1] - 
                self.prefix[left] * self.power[right - left + 1]) % self.mod
    
    def compare(self, l1: int, r1: int, l2: int, r2: int) -> bool:
        """
        比较两个子串是否相等
        
        时间复杂度: O(1)
        """
        if r1 - l1 != r2 - l2:
            return False
        return self.get_hash(l1, r1) == self.get_hash(l2, r2)


class DoubleHash:
    """
    双哈希（减少冲突概率）
    """
    
    def __init__(self, s: str):
        self.h1 = StringHash(s, 131, 10**18 + 9)
        self.h2 = StringHash(s, 1331, 10**18 + 7)
    
    def get_hash(self, left: int, right: int) -> tuple:
        return (self.h1.get_hash(left, right), 
                self.h2.get_hash(left, right))
    
    def compare(self, l1: int, r1: int, l2: int, r2: int) -> bool:
        return (self.h1.compare(l1, r1, l2, r2) and 
                self.h2.compare(l1, r1, l2, r2))


# 使用示例
s = "abcabc"
sh = StringHash(s)

print(sh.get_hash(0, 2))  # "abc"的哈希
print(sh.get_hash(3, 5))  # "abc"的哈希
print(sh.compare(0, 2, 3, 5))  # True
```

# 二、Z函数与扩展KMP

## 2.1 Z函数

```python
def z_function(s: str) -> list:
    """
    Z函数（扩展KMP）
    
    定义:
        z[i] = s[i:]与s的最长公共前缀长度
        z[0] = 0（或等于len(s)，取决于定义）
    
    参数:
        s: str - 输入字符串
    返回:
        list - Z数组
    
    时间复杂度: O(n)
    
    应用:
        - 字符串匹配: pattern + "$" + text
        - 本质不同子串数
        - 最小循环节
    """
    n = len(s)
    if n == 0:
        return []
    
    z = [0] * n
    z[0] = n  # 可选：设为n或0
    
    # [l, r] 是当前最右的匹配区间
    l, r = 0, 0
    
    for i in range(1, n):
        if i < r:
            # 利用已知信息
            z[i] = min(r - i, z[i - l])
        
        # 暴力扩展
        while i + z[i] < n and s[z[i]] == s[i + z[i]]:
            z[i] += 1
        
        # 更新最右区间
        if i + z[i] > r:
            l, r = i, i + z[i]
    
    return z


def z_function_match(text: str, pattern: str) -> list:
    """
    使用Z函数进行字符串匹配
    
    返回:
        list - 所有匹配位置
    """
    if not pattern or not text:
        return []
    
    m = len(pattern)
    # 连接：pattern + "$" + text
    combined = pattern + "$" + text
    z = z_function(combined)
    
    result = []
    for i in range(m + 1, len(combined)):
        if z[i] == m:
            result.append(i - m - 1)
    
    return result


def min_period(s: str) -> int:
    """
    使用Z函数求最小循环节长度
    
    如果存在i使得 i + z[i] == n 且 n % i == 0
    则i是一个可能的周期
    """
    n = len(s)
    z = z_function(s)
    
    for i in range(1, n):
        if i + z[i] == n and n % i == 0:
            return i
    
    return n


# 使用示例
s = "aabaaab"
print(f"Z函数: {z_function(s)}")
# 输出: [7, 1, 0, 2, 3, 1, 0]
```

## 2.2 扩展KMP

```python
def extend_kmp(text: str, pattern: str) -> list:
    """
    扩展KMP
    
    计算text的每个后缀与pattern的最长公共前缀
    
    参数:
        text: str - 文本串
        pattern: str - 模式串
    返回:
        list - extend[i] = lcp(text[i:], pattern)
    
    时间复杂度: O(n + m)
    """
    n, m = len(text), len(pattern)
    if m == 0:
        return [0] * n
    
    # 先计算pattern的Z函数
    z = z_function(pattern)
    
    extend = [0] * n
    
    # 计算初始匹配长度
    while extend[0] < m and extend[0] < n and text[extend[0]] == pattern[extend[0]]:
        extend[0] += 1
    
    l, r = 0, extend[0]
    
    for i in range(1, n):
        if i < r:
            # 利用已知信息
            k = i - l
            if z[k] < r - i:
                extend[i] = z[k]
            else:
                extend[i] = r - i
                # 继续扩展
                while extend[i] < m and i + extend[i] < n and \
                      pattern[extend[i]] == text[i + extend[i]]:
                    extend[i] += 1
                l, r = i, i + extend[i]
        else:
            # 暴力匹配
            extend[i] = 0
            while extend[i] < m and i + extend[i] < n and \
                  pattern[extend[i]] == text[i + extend[i]]:
                extend[i] += 1
            if i + extend[i] > r:
                l, r = i, i + extend[i]
    
    return extend
```

# 三、Manacher算法

## 3.1 最长回文子串

```python
def manacher(s: str) -> list:
    """
    Manacher算法 - O(n)求所有回文半径
    
    LeetCode 5. Longest Palindromic Substring
    
    参数:
        s: str - 输入字符串
    返回:
        list - p[i]表示以i为中心的最长回文半径（在转换后的字符串上）
    
    时间复杂度: O(n)
    空间复杂度: O(n)
    
    核心思想:
        1. 插入分隔符统一奇偶长度
        2. 利用已知回文的对称性
    """
    # 转换字符串: "abc" -> "^#a#b#c#$"
    t = "^#" + "#".join(s) + "#$"
    n = len(t)
    p = [0] * n
    
    center = 0  # 当前最右回文的中心
    right = 0   # 当前最右回文的右边界
    
    for i in range(1, n - 1):
        if i < right:
            # 对称位置
            mirror = 2 * center - i
            p[i] = min(right - i, p[mirror])
        
        # 尝试扩展
        while t[i + p[i] + 1] == t[i - p[i] - 1]:
            p[i] += 1
        
        # 更新最右边界
        if i + p[i] > right:
            center, right = i, i + p[i]
    
    return p


def longest_palindrome(s: str) -> str:
    """
    返回最长回文子串
    
    LeetCode 5. Longest Palindromic Substring
    """
    if not s:
        return ""
    
    p = manacher(s)
    
    # 找最大p值及其位置
    max_len = 0
    max_center = 0
    
    for i in range(1, len(p) - 1):
        if p[i] > max_len:
            max_len = p[i]
            max_center = i
    
    # 转换回原字符串索引
    start = (max_center - max_len) // 2
    return s[start:start + max_len]


def count_palindromic_substrings(s: str) -> int:
    """
    统计回文子串数量
    
    LeetCode 647. Palindromic Substrings
    """
    p = manacher(s)
    
    count = 0
    for i in range(1, len(p) - 1):
        # p[i]个回文（包括空串外的）
        count += (p[i] + 1) // 2
    
    return count


def longest_palindromic_prefix(s: str) -> str:
    """
    最长回文前缀
    
    应用: LeetCode 214. Shortest Palindrome
    """
    p = manacher(s)
    n = len(s)
    
    # 找以第一个字符开始的最长回文
    for i in range(2 * n, 0, -1):
        if i - p[i] == 1:  # 回文从位置1开始（原字符串的位置0）
            return s[:(i + p[i] - 1) // 2]
    
    return s[0] if s else ""


# 使用示例
s = "babad"
print(f"最长回文: {longest_palindrome(s)}")

s = "cbbd"
print(f"最长回文: {longest_palindrome(s)}")
```

## 3.2 回文相关问题

```python
def shortest_palindrome(s: str) -> str:
    """
    最短回文串
    
    LeetCode 214. Shortest Palindrome
    
    在s前面添加最少字符使其成为回文
    
    思路: 找最长回文前缀，将剩余部分反转添加到前面
    """
    if not s:
        return ""
    
    # 方法1: Manacher
    p = manacher(s)
    n = len(s)
    
    # 找最长回文前缀的长度
    longest_prefix_len = 0
    for i in range(1, 2 * n + 2):
        if i - p[i] == 1:  # 从头开始的回文
            palindrome_len = p[i]  # 在转换后字符串中的长度
            original_len = palindrome_len  # 原字符串中的长度
            if original_len > longest_prefix_len:
                longest_prefix_len = original_len
    
    # 需要添加的部分
    suffix = s[longest_prefix_len:]
    return suffix[::-1] + s


def shortest_palindrome_kmp(s: str) -> str:
    """
    最短回文串 - KMP解法
    
    利用KMP的failure function
    """
    if not s:
        return ""
    
    # 构造 s + "#" + reverse(s)
    rev = s[::-1]
    combined = s + "#" + rev
    
    # 计算LPS
    lps = compute_lps(combined)
    
    # 最长回文前缀长度
    longest = lps[-1]
    
    # 需要添加的部分
    return rev[:len(s) - longest] + s


def palindrome_pairs(words: list) -> list:
    """
    回文对
    
    LeetCode 336. Palindrome Pairs
    
    找出所有(i, j)使得words[i] + words[j]是回文
    """
    def is_palindrome(s, left, right):
        while left < right:
            if s[left] != s[right]:
                return False
            left += 1
            right -= 1
        return True
    
    word_dict = {word: i for i, word in enumerate(words)}
    result = []
    
    for i, word in enumerate(words):
        n = len(word)
        
        for j in range(n + 1):
            # 前缀是回文，后缀的反转在字典中
            if is_palindrome(word, 0, j - 1):
                suffix = word[j:][::-1]
                if suffix in word_dict and word_dict[suffix] != i:
                    result.append([word_dict[suffix], i])
            
            # 后缀是回文，前缀的反转在字典中
            if j != n and is_palindrome(word, j, n - 1):
                prefix = word[:j][::-1]
                if prefix in word_dict and word_dict[prefix] != i:
                    result.append([i, word_dict[prefix]])
    
    return result
```

# 四、后缀数组

## 4.1 后缀数组构建

```python
def suffix_array_naive(s: str) -> list:
    """
    后缀数组 - 朴素O(n² log n)实现
    
    定义:
        sa[i] = 排名为i的后缀的起始位置
    
    返回:
        list - 后缀数组
    """
    n = len(s)
    suffixes = [(s[i:], i) for i in range(n)]
    suffixes.sort()
    return [idx for _, idx in suffixes]


def suffix_array_doubling(s: str) -> list:
    """
    后缀数组 - 倍增法 O(n log² n)
    
    核心思想:
        利用长度为k的排名，O(n log n)得到长度为2k的排名
    """
    n = len(s)
    if n == 0:
        return []
    
    # 初始排名（按单个字符）
    rank = [ord(c) for c in s]
    sa = list(range(n))
    k = 1
    
    while k < n:
        # 按(rank[i], rank[i+k])排序
        def key(i):
            return (rank[i], rank[i + k] if i + k < n else -1)
        
        sa.sort(key=key)
        
        # 更新排名
        new_rank = [0] * n
        for i in range(1, n):
            new_rank[sa[i]] = new_rank[sa[i - 1]]
            if key(sa[i]) != key(sa[i - 1]):
                new_rank[sa[i]] += 1
        
        rank = new_rank
        k *= 2
        
        # 优化：如果所有排名都不同，可以提前结束
        if rank[sa[-1]] == n - 1:
            break
    
    return sa


def suffix_array_dc3(s: str) -> list:
    """
    后缀数组 - DC3/Skew算法 O(n)
    
    工业级实现，这里给出简化版
    """
    # 完整DC3实现较复杂，面试中通常使用倍增法
    return suffix_array_doubling(s)
```

## 4.2 LCP数组

```python
def lcp_array(s: str, sa: list) -> list:
    """
    最长公共前缀数组（Kasai算法）
    
    定义:
        lcp[i] = lcp(sa[i], sa[i-1]) 即相邻后缀的最长公共前缀
    
    时间复杂度: O(n)
    """
    n = len(s)
    if n == 0:
        return []
    
    # 排名数组
    rank = [0] * n
    for i, pos in enumerate(sa):
        rank[pos] = i
    
    lcp = [0] * n
    k = 0  # 当前lcp长度
    
    for i in range(n):
        if rank[i] == 0:
            k = 0
            continue
        
        j = sa[rank[i] - 1]  # 前一个后缀的起始位置
        
        # 扩展lcp
        while i + k < n and j + k < n and s[i + k] == s[j + k]:
            k += 1
        
        lcp[rank[i]] = k
        
        # 下一个后缀的lcp至少是k-1
        if k > 0:
            k -= 1
    
    return lcp


def longest_repeated_substring(s: str) -> str:
    """
    最长重复子串
    
    LeetCode 1044. Longest Duplicate Substring
    
    使用后缀数组 + LCP数组
    """
    if len(s) <= 1:
        return ""
    
    sa = suffix_array_doubling(s)
    lcp = lcp_array(s, sa)
    
    max_lcp = 0
    max_pos = 0
    
    for i in range(1, len(s)):
        if lcp[i] > max_lcp:
            max_lcp = lcp[i]
            max_pos = sa[i]
    
    return s[max_pos:max_pos + max_lcp]


def distinct_substrings(s: str) -> int:
    """
    本质不同子串数量
    
    公式: n*(n+1)/2 - sum(lcp)
    """
    n = len(s)
    if n == 0:
        return 0
    
    sa = suffix_array_doubling(s)
    lcp = lcp_array(s, sa)
    
    total = n * (n + 1) // 2
    return total - sum(lcp)
```

## 5.1 AC自动机

```python
from collections import deque, defaultdict

class AhoCorasick:
    """
    AC自动机 - 多模式串匹配
    
    在Trie基础上添加fail指针，实现O(n+m+z)匹配
    n: 文本长度, m: 模式串总长度, z: 匹配数
    
    应用:
        - 敏感词过滤
        - 多模式匹配
        - 病毒特征码检测
    """
    
    def __init__(self):
        """初始化AC自动机"""
        self.goto = [{}]  # 转移函数
        self.fail = [0]    # 失配指针
        self.output = [[]]  # 输出函数（匹配的模式串索引）
        self.patterns = []
    
    def add_pattern(self, pattern: str) -> int:
        """
        添加模式串
        
        返回:
            int - 模式串索引
        """
        idx = len(self.patterns)
        self.patterns.append(pattern)
        
        state = 0
        for c in pattern:
            if c not in self.goto[state]:
                self.goto[state][c] = len(self.goto)
                self.goto.append({})
                self.fail.append(0)
                self.output.append([])
            state = self.goto[state][c]
        
        self.output[state].append(idx)
        return idx
    
    def build(self) -> None:
        """
        构建fail指针（BFS）
        
        时间复杂度: O(m)，m是模式串总长度
        """
        queue = deque()
        
        # 第一层的fail指针都指向根
        for c, next_state in self.goto[0].items():
            self.fail[next_state] = 0
            queue.append(next_state)
        
        # BFS构建
        while queue:
            state = queue.popleft()
            
            for c, next_state in self.goto[state].items():
                queue.append(next_state)
                
                # 找fail指针
                f = self.fail[state]
                while f != 0 and c not in self.goto[f]:
                    f = self.fail[f]
                
                self.fail[next_state] = self.goto[f].get(c, 0)
                
                # 合并输出
                self.output[next_state].extend(
                    self.output[self.fail[next_state]]
                )
    
    def search(self, text: str) -> list:
        """
        在文本中搜索所有模式串
        
        返回:
            list of (position, pattern_index) - 匹配结果
        """
        result = []
        state = 0
        
        for i, c in enumerate(text):
            # 沿fail链找可转移的状态
            while state != 0 and c not in self.goto[state]:
                state = self.fail[state]
            
            state = self.goto[state].get(c, 0)
            
            # 输出匹配
            for pattern_idx in self.output[state]:
                pattern = self.patterns[pattern_idx]
                result.append((i - len(pattern) + 1, pattern_idx))
        
        return result


# 使用示例
ac = AhoCorasick()
patterns = ["he", "she", "his", "hers"]
for p in patterns:
    ac.add_pattern(p)

ac.build()

text = "ushers"
matches = ac.search(text)
print(f"匹配结果: {matches}")
# 输出: [(1, 1), (2, 0), (2, 3)] 表示 "she", "he", "hers"
```

## 5.2 01 Trie

```python
class BinaryTrie:
    """
    01 Trie（二进制字典树）
    
    应用:
        - 最大异或对
        - 异或运算相关问题
    
    LeetCode 421. Maximum XOR of Two Numbers in an Array
    """
    
    def __init__(self, max_bits: int = 30):
        """
        参数:
            max_bits: int - 最大位数
        """
        self.max_bits = max_bits
        self.root = {}
    
    def insert(self, num: int) -> None:
        """插入数字"""
        node = self.root
        for i in range(self.max_bits - 1, -1, -1):
            bit = (num >> i) & 1
            if bit not in node:
                node[bit] = {}
            node = node[bit]
    
    def max_xor(self, num: int) -> int:
        """
        找与num异或最大的数
        
        贪心: 每一位尽量选择与num该位不同的
        """
        node = self.root
        result = 0
        
        for i in range(self.max_bits - 1, -1, -1):
            bit = (num >> i) & 1
            want = 1 - bit  # 想要的位
            
            if want in node:
                result |= (1 << i)
                node = node[want]
            elif bit in node:
                node = node[bit]
            else:
                break
        
        return result


def find_maximum_xor(nums: list) -> int:
    """
    数组中两个数的最大异或值
    
    LeetCode 421
    """
    trie = BinaryTrie()
    
    max_xor = 0
    for num in nums:
        trie.insert(num)
        max_xor = max(max_xor, trie.max_xor(num))
    
    return max_xor
```

## 总结

### 字符串算法分类

| 算法 | 用途 | 时间复杂度 | 空间复杂度 |
|------|------|-----------|-----------|
| KMP | 单模式匹配 | O(n+m) | O(m) |
| Rabin-Karp | 单/多模式匹配 | O(n+m) 平均 | O(1) |
| Z函数 | 前缀匹配 | O(n) | O(n) |
| Manacher | 最长回文 | O(n) | O(n) |
| 后缀数组 | 子串问题 | O(n log n) | O(n) |
| AC自动机 | 多模式匹配 | O(n+m+z) | O(m×σ) |

### 核心记忆

| 算法 | 关键点 |
|------|--------|
| KMP | lps[i]是最长相等前后缀 |
| Rabin-Karp | 滚动哈希，取模防溢出 |
| Z函数 | z[i]是后缀与原串的lcp |
| Manacher | 插入分隔符，利用对称性 |
| 后缀数组 | 倍增法，lcp数组用Kasai |
| AC自动机 | Trie + fail指针（BFS构建） |

### 问题选型

| 问题类型 | 推荐算法 |
|----------|---------|
| 单模式匹配 | KMP |
| 多模式匹配 | AC自动机/Rabin-Karp |
| 最长回文 | Manacher |
| 最长重复子串 | 后缀数组 + LCP |
| 子串哈希比较 | 字符串哈希 |
| 最大异或 | 01 Trie |
