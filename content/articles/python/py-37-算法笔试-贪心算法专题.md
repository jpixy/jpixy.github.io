## 概述

贪心算法是一种在每一步选择中都采取当前状态下最优的选择，期望导出全局最优解的算法策略。本文详细介绍贪心算法的核心思想和经典应用。

# 一、贪心算法基础

## 1.1 核心思想

```python
"""
贪心算法核心思想：

1. 贪心选择性质
   - 每一步都做出局部最优选择
   - 局部最优能导向全局最优

2. 最优子结构
   - 问题的最优解包含子问题的最优解

3. 与动态规划的区别
   - 贪心：每步只考虑当前最优，不回溯
   - DP：考虑所有可能，选择全局最优

4. 贪心的证明方法
   - 交换论证法：假设存在更优解，通过交换证明贪心解不比它差
   - 归纳法：证明每一步贪心选择后，问题规模减小，且仍具有最优子结构
"""

# 贪心算法模板
def greedy_template(problem):
    """
    贪心算法通用模板
    """
    # 1. 排序或预处理（通常需要）
    problem = preprocess(problem)
    
    result = initial_value
    
    # 2. 迭代做贪心选择
    for item in problem:
        if can_select(item):
            result = update(result, item)
    
    return result
```

# 二、区间问题

## 2.1 区间调度（活动选择）

```python
def max_activities(activities: list) -> int:
    """
    活动选择问题：选择最多的互不重叠活动
    
    参数:
        activities: List[Tuple[int, int]] - 活动列表 [(start, end), ...]
    返回:
        int - 最多可选活动数
    
    贪心策略: 按结束时间排序，优先选择结束早的
    
    时间复杂度: O(n log n)
    
    证明: 选择结束最早的活动，给后续活动留出最多空间
    """
    if not activities:
        return 0
    
    # 按结束时间排序
    activities.sort(key=lambda x: x[1])
    
    count = 1
    prev_end = activities[0][1]
    
    for start, end in activities[1:]:
        if start >= prev_end:  # 不重叠
            count += 1
            prev_end = end
    
    return count


def interval_scheduling(intervals: list) -> list:
    """
    返回选中的活动列表
    """
    if not intervals:
        return []
    
    # 保留索引
    indexed = [(start, end, i) for i, (start, end) in enumerate(intervals)]
    indexed.sort(key=lambda x: x[1])
    
    result = [indexed[0][2]]
    prev_end = indexed[0][1]
    
    for start, end, idx in indexed[1:]:
        if start >= prev_end:
            result.append(idx)
            prev_end = end
    
    return result
```

## 2.2 区间覆盖

```python
def min_intervals_to_cover(intervals: list, target: tuple) -> int:
    """
    用最少的区间覆盖目标区间
    
    参数:
        intervals: List[Tuple[int, int]] - 可用区间
        target: Tuple[int, int] - 目标区间 (start, end)
    返回:
        int - 最少需要的区间数，无法覆盖返回-1
    
    贪心策略: 每次选择能覆盖当前点且右端点最远的区间
    """
    start, end = target
    # 按起点排序
    intervals.sort()
    
    count = 0
    i = 0
    n = len(intervals)
    curr = start
    
    while curr < end:
        # 找能覆盖curr且右端点最远的区间
        max_reach = curr
        while i < n and intervals[i][0] <= curr:
            max_reach = max(max_reach, intervals[i][1])
            i += 1
        
        if max_reach == curr:
            return -1  # 无法继续覆盖
        
        curr = max_reach
        count += 1
    
    return count


# LeetCode 45. 跳跃游戏II
def jump(nums: list) -> int:
    """
    最少跳跃次数到达最后位置
    
    贪心策略: 在当前能到达的范围内，选择能跳最远的位置
    """
    n = len(nums)
    if n <= 1:
        return 0
    
    jumps = 0
    curr_end = 0      # 当前跳跃能到达的边界
    farthest = 0      # 下一跳能到达的最远位置
    
    for i in range(n - 1):
        farthest = max(farthest, i + nums[i])
        
        if i == curr_end:
            jumps += 1
            curr_end = farthest
            
            if curr_end >= n - 1:
                break
    
    return jumps
```

## 2.3 无重叠区间

```python
def erase_overlap_intervals(intervals: list) -> int:
    """
    LeetCode 435. Non-overlapping Intervals
    
    移除最少区间使剩余区间不重叠
    
    贪心策略: 按结束时间排序，保留结束早的
    等价于: 总数 - 最多不重叠区间数
    """
    if not intervals:
        return 0
    
    intervals.sort(key=lambda x: x[1])
    
    count = 1  # 保留的区间数
    prev_end = intervals[0][1]
    
    for start, end in intervals[1:]:
        if start >= prev_end:
            count += 1
            prev_end = end
    
    return len(intervals) - count


def find_min_arrow_shots(points: list) -> int:
```

## 2.4 合并区间

```python
def merge(intervals: list) -> list:
    """
    LeetCode 56. Merge Intervals
    
    合并所有重叠区间
    """
    if not intervals:
        return []
    
    intervals.sort(key=lambda x: x[0])
    
    merged = [intervals[0]]
    
    for start, end in intervals[1:]:
        if start <= merged[-1][1]:
            merged[-1][1] = max(merged[-1][1], end)
        else:
            merged.append([start, end])
    
    return merged


def insert(intervals: list, new_interval: list) -> list:
    """
    LeetCode 57. Insert Interval
    
    插入新区间并合并
    """
    result = []
    i = 0
    n = len(intervals)
    
    # 添加所有在新区间之前的
    while i < n and intervals[i][1] < new_interval[0]:
        result.append(intervals[i])
        i += 1
    
    # 合并重叠区间
    while i < n and intervals[i][0] <= new_interval[1]:
        new_interval[0] = min(new_interval[0], intervals[i][0])
        new_interval[1] = max(new_interval[1], intervals[i][1])
        i += 1
    result.append(new_interval)
    
    # 添加剩余区间
    result.extend(intervals[i:])
    
    return result
```

# 三、分配问题

## 3.1 分发糖果

```python
def candy(ratings: list) -> int:
    """
    LeetCode 135. Candy
    
    每个孩子至少1颗糖，评分高的孩子比相邻的获得更多糖
    求最少需要多少糖
    
    贪心策略: 两次遍历
    - 从左到右：保证右边比左边高的多
    - 从右到左：保证左边比右边高的多
    """
    n = len(ratings)
    candies = [1] * n
    
    # 从左到右
    for i in range(1, n):
        if ratings[i] > ratings[i - 1]:
            candies[i] = candies[i - 1] + 1
    
    # 从右到左
    for i in range(n - 2, -1, -1):
        if ratings[i] > ratings[i + 1]:
            candies[i] = max(candies[i], candies[i + 1] + 1)
    
    return sum(candies)
```

## 3.2 分发饼干

```python
def find_content_children(g: list, s: list) -> int:
    """
    LeetCode 455. Assign Cookies
    
    g[i]是孩子i的胃口，s[j]是饼干j的大小
    每个孩子最多一块饼干，求最多能满足多少孩子
    
    贪心策略: 排序后，用最小的饼干满足最小胃口的孩子
    """
    g.sort()
    s.sort()
    
    child = 0
    cookie = 0
    
    while child < len(g) and cookie < len(s):
        if s[cookie] >= g[child]:
            child += 1
        cookie += 1
    
    return child
```

## 3.3 任务调度

```python
def least_interval(tasks: list, n: int) -> int:
    """
    LeetCode 621. Task Scheduler
    
    相同任务之间需要间隔n个单位时间
    求完成所有任务的最少时间
    
    贪心策略: 优先执行剩余次数最多的任务
    """
    from collections import Counter
    
    counts = Counter(tasks)
    max_count = max(counts.values())
    max_count_tasks = sum(1 for c in counts.values() if c == max_count)
    
    # 方法1: 数学公式
    # 框架: (max_count - 1) 个完整周期 + 最后一个不完整周期
    result = (max_count - 1) * (n + 1) + max_count_tasks
    
    # 如果任务种类多，可能不需要等待
    return max(result, len(tasks))


def get_order(tasks: list) -> list:
    """
    LeetCode 1834. Single-Threaded CPU
    
    按处理时间短优先（SJF）调度
    """
    import heapq
    
    # (入队时间, 处理时间, 原始索引)
    indexed = [(t[0], t[1], i) for i, t in enumerate(tasks)]
    indexed.sort()
    
    result = []
    heap = []  # (处理时间, 原始索引)
    time = 0
    i = 0
    n = len(tasks)
    
    while i < n or heap:
        if not heap and i < n and indexed[i][0] > time:
            time = indexed[i][0]
        
        # 将所有可用任务加入堆
        while i < n and indexed[i][0] <= time:
            heapq.heappush(heap, (indexed[i][1], indexed[i][2]))
            i += 1
        
        # 执行处理时间最短的任务
        proc_time, idx = heapq.heappop(heap)
        result.append(idx)
        time += proc_time
    
    return result
```

## 4.1 买卖股票

```python
def max_profit(prices: list) -> int:
    """
    LeetCode 121. Best Time to Buy and Sell Stock
    
    只能买卖一次，求最大利润
    
    贪心: 记录到目前为止的最低价格
    """
    min_price = float('inf')
    max_profit = 0
    
    for price in prices:
        min_price = min(min_price, price)
        max_profit = max(max_profit, price - min_price)
    
    return max_profit


def max_profit_ii(prices: list) -> int:
    """
    LeetCode 122. Best Time to Buy and Sell Stock II
    
    可以多次买卖，求最大利润
    
    贪心: 收集所有上涨的利润
    """
    profit = 0
    for i in range(1, len(prices)):
        if prices[i] > prices[i - 1]:
            profit += prices[i] - prices[i - 1]
    return profit


def max_profit_with_fee(prices: list, fee: int) -> int:
    """
    LeetCode 714. Best Time to Buy and Sell Stock with Transaction Fee
    
    每次交易有手续费
    """
    n = len(prices)
    hold = -prices[0]  # 持有股票的最大收益
    cash = 0           # 不持有股票的最大收益
    
    for i in range(1, n):
        cash = max(cash, hold + prices[i] - fee)
        hold = max(hold, cash - prices[i])
    
    return cash
```

## 4.2 加油站

```python
def can_complete_circuit(gas: list, cost: list) -> int:
    """
    LeetCode 134. Gas Station
    
    能否绕一圈，返回起始加油站索引
    
    贪心: 如果从i出发到j失败，则从i到j之间任意点出发都会失败
    """
    n = len(gas)
    total_tank = 0
    curr_tank = 0
    start = 0
    
    for i in range(n):
        diff = gas[i] - cost[i]
        total_tank += diff
        curr_tank += diff
        
        if curr_tank < 0:
            # 从i+1重新开始
            start = i + 1
            curr_tank = 0
    
    return start if total_tank >= 0 else -1
```

## 4.3 跳跃游戏

```python
def can_jump(nums: list) -> bool:
    """
    LeetCode 55. Jump Game
    
    能否到达最后位置
    
    贪心: 维护能到达的最远位置
    """
    max_reach = 0
    
    for i, num in enumerate(nums):
        if i > max_reach:
            return False
        max_reach = max(max_reach, i + num)
        if max_reach >= len(nums) - 1:
            return True
    
    return True
```

## 5.1 划分字母区间

```python
def partition_labels(s: str) -> list:
    """
    LeetCode 763. Partition Labels
    
    将字符串划分成最多的片段，使每个字母只出现在一个片段中
    
    贪心: 找每个字母最后出现的位置，扩展当前片段的边界
    """
    last = {c: i for i, c in enumerate(s)}
    
    result = []
    start = 0
    end = 0
    
    for i, c in enumerate(s):
        end = max(end, last[c])
        if i == end:
            result.append(end - start + 1)
            start = end + 1
    
    return result
```

## 5.2 移除K位数字

```python
def remove_k_digits(num: str, k: int) -> str:
    """
    LeetCode 402. Remove K Digits
    
    移除k位数字使剩余数字最小
    
    贪心: 使用单调栈，保持数字递增
    """
    stack = []
    
    for digit in num:
        while k > 0 and stack and stack[-1] > digit:
            stack.pop()
            k -= 1
        stack.append(digit)
    
    # 如果还需要删除
    stack = stack[:-k] if k else stack
    
    # 去除前导零
    result = ''.join(stack).lstrip('0')
    
    return result if result else '0'
```

## 5.3 重构字符串

```python
def reorganize_string(s: str) -> str:
    """
    LeetCode 767. Reorganize String
    
    重排字符串使相邻字符不同
    
    贪心: 优先放置出现次数最多的字符
    """
    from collections import Counter
    import heapq
    
    count = Counter(s)
    max_count = max(count.values())
    
    # 如果最多的字符超过一半，无解
    if max_count > (len(s) + 1) // 2:
        return ""
    
    # 最大堆
    heap = [(-c, ch) for ch, c in count.items()]
    heapq.heapify(heap)
    
    result = []
    prev = (0, '')
    
    while heap:
        cnt, ch = heapq.heappop(heap)
        result.append(ch)
        
        if prev[0] < 0:
            heapq.heappush(heap, prev)
        
        prev = (cnt + 1, ch)
    
    return ''.join(result)
```

# 六、数学问题

## 6.1 分数背包

```python
def fractional_knapsack(weights: list, values: list, capacity: int) -> float:
    """
    分数背包：物品可以分割
    
    贪心策略: 按单位价值（value/weight）排序，优先选性价比高的
    
    时间复杂度: O(n log n)
    """
    n = len(weights)
    # (单位价值, 重量, 价值)
    items = [(values[i] / weights[i], weights[i], values[i]) for i in range(n)]
    items.sort(reverse=True)
    
    total_value = 0
    remaining = capacity
    
    for ratio, w, v in items:
        if remaining >= w:
            total_value += v
            remaining -= w
        else:
            total_value += ratio * remaining
            break
    
    return total_value
```

## 6.2 霍夫曼编码

```python
import heapq

def huffman_encoding(frequencies: dict) -> dict:
    """
    霍夫曼编码
    
    参数:
        frequencies: Dict[str, int] - 字符频率
    返回:
        Dict[str, str] - 字符到编码的映射
    
    贪心策略: 每次合并频率最小的两个节点
    """
    # 构建最小堆
    heap = [[freq, [char, ""]] for char, freq in frequencies.items()]
    heapq.heapify(heap)
    
    if len(heap) == 1:
        return {heap[0][1][0]: "0"}
    
    while len(heap) > 1:
        lo = heapq.heappop(heap)
        hi = heapq.heappop(heap)
        
        # 给左子树加0，右子树加1
        for pair in lo[1:]:
            pair[1] = '0' + pair[1]
        for pair in hi[1:]:
            pair[1] = '1' + pair[1]
        
        # 合并
        heapq.heappush(heap, [lo[0] + hi[0]] + lo[1:] + hi[1:])
    
    return {char: code for char, code in heap[0][1:]}


# 使用示例
freq = {'a': 5, 'b': 9, 'c': 12, 'd': 13, 'e': 16, 'f': 45}
codes = huffman_encoding(freq)
# 输出类似: {'f': '0', 'c': '100', 'd': '101', 'a': '1100', 'b': '1101', 'e': '111'}
```

## 6.3 找零钱问题

```python
def min_coins_greedy(coins: list, amount: int) -> int:
    """
    找零钱问题（贪心版，仅适用于特殊面额）
    
    注意: 贪心只在特定面额（如1,5,10,25）下正确
          一般情况需要用DP
    """
    coins.sort(reverse=True)
    count = 0
    
    for coin in coins:
        if amount >= coin:
            count += amount // coin
            amount %= coin
    
    return count if amount == 0 else -1


# 通用版本（DP）
def min_coins_dp(coins: list, amount: int) -> int:
    dp = [float('inf')] * (amount + 1)
    dp[0] = 0
    
    for i in range(1, amount + 1):
        for coin in coins:
            if coin <= i:
                dp[i] = min(dp[i], dp[i - coin] + 1)
    
    return dp[amount] if dp[amount] != float('inf') else -1
```

## 总结

### 贪心问题分类

| 问题类型 | 贪心策略 | 典型问题 |
|----------|----------|----------|
| 区间调度 | 按结束时间排序 | 活动选择, 无重叠区间 |
| 区间覆盖 | 选右端点最远的 | 跳跃游戏, 最少区间覆盖 |
| 分配问题 | 排序后匹配 | 分发糖果, 分发饼干 |
| 数组问题 | 维护最优值 | 股票买卖, 加油站 |
| 字符串问题 | 单调栈/贪心选择 | 移除K位, 重构字符串 |

### 贪心 vs DP

| 特点 | 贪心 | DP |
|------|------|-----|
| 思路 | 局部最优 | 全局最优 |
| 复杂度 | 通常更低 | 通常更高 |
| 正确性 | 需要证明 | 保证正确 |
| 适用范围 | 有限 | 广泛 |

### 关键记忆

1. **区间问题**：按结束时间排序是关键
2. **证明方法**：交换论证法最常用
3. **贪心可能错**：找零钱问题不能用贪心
4. **排序是基础**：大多数贪心需要先排序
5. **单调栈常用**：字符串贪心常配合单调栈
