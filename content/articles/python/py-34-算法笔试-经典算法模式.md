```python
def fixed_window_template(nums: list, k: int):
    """
    固定窗口大小模板
    
    参数:
        nums: List[int] - 输入数组
        k: int - 窗口大小
    返回:
        根据具体问题而定
    
    时间复杂度: O(n)
    空间复杂度: O(1) 或 O(k)
    
    适用场景:
        - 求窗口内的和/平均值/最值
        - 窗口大小固定
    """
    n = len(nums)
    if n < k:
        return None
    
    # 初始化第一个窗口
    window_sum = sum(nums[:k])
    result = window_sum  # 或其他初始化
    
    # 滑动窗口
    for i in range(k, n):
        # 加入新元素，移出旧元素
        window_sum += nums[i] - nums[i - k]
        result = max(result, window_sum)  # 或其他操作
    
    return result

def max_sum_subarray(nums: list, k: int) -> int:
    """
    求长度为k的子数组的最大和
    """
    if len(nums) < k:
        return 0
    
    window_sum = sum(nums[:k])
    max_sum = window_sum
    
    for i in range(k, len(nums)):
        window_sum += nums[i] - nums[i - k]
        max_sum = max(max_sum, window_sum)
    
    return max_sum


# 示例2: 滑动窗口平均值
class MovingAverage:
    """
    LeetCode 346. Moving Average from Data Stream
    
    计算最近size个数的平均值
    """
    def __init__(self, size: int):
        """
        参数:
            size: int - 窗口大小
        """
        from collections import deque
        self.size = size
        self.window = deque()
        self.total = 0
    
    def next(self, val: int) -> float:
        """
        添加新值并返回当前平均值
        
        时间复杂度: O(1)
        """
        self.window.append(val)
        self.total += val
        
        if len(self.window) > self.size:
            self.total -= self.window.popleft()
        
        return self.total / len(self.window)


# 示例3: 定长子串中元音的最大数目
def max_vowels(s: str, k: int) -> int:
    """
    LeetCode 1456. Maximum Number of Vowels in a Substring of Given Length
    """
    vowels = set('aeiou')
    count = sum(1 for c in s[:k] if c in vowels)
    max_count = count
    
    for i in range(k, len(s)):
        if s[i] in vowels:
            count += 1
        if s[i - k] in vowels:
            count -= 1
        max_count = max(max_count, count)
    
    return max_count
```

```python
# 示例1: 无重复字符的最长子串
def length_of_longest_substring(s: str) -> int:
    """
    LeetCode 3. Longest Substring Without Repeating Characters
    
    返回不含重复字符的最长子串的长度
    
    时间复杂度: O(n)
    空间复杂度: O(min(n, 字符集大小))
    """
    char_index = {}  # 字符最后出现的位置
    left = 0
    max_len = 0
    
    for right, char in enumerate(s):
        # 如果字符已存在且在当前窗口内
        if char in char_index and char_index[char] >= left:
            left = char_index[char] + 1
        
        char_index[char] = right
        max_len = max(max_len, right - left + 1)
    
    return max_len


# 示例2: 长度最小的子数组
def min_sub_array_len(target: int, nums: list) -> int:
    """
    LeetCode 209. Minimum Size Subarray Sum
    
    找出和 >= target 的最短子数组长度
    
    时间复杂度: O(n)
    """
    left = 0
    window_sum = 0
    min_len = float('inf')
    
    for right in range(len(nums)):
        window_sum += nums[right]
        
        # 满足条件时尝试收缩
        while window_sum >= target:
            min_len = min(min_len, right - left + 1)
            window_sum -= nums[left]
            left += 1
    
    return min_len if min_len != float('inf') else 0


# 示例3: 最多K个不同字符的最长子串
def longest_substring_k_distinct(s: str, k: int) -> int:
    """
    LeetCode 340. Longest Substring with At Most K Distinct Characters
    """
    from collections import defaultdict
    
    char_count = defaultdict(int)
    left = 0
    max_len = 0
    
    for right, char in enumerate(s):
        char_count[char] += 1
        
        # 不同字符数超过k时收缩
        while len(char_count) > k:
            left_char = s[left]
            char_count[left_char] -= 1
            if char_count[left_char] == 0:
                del char_count[left_char]
            left += 1
        
        max_len = max(max_len, right - left + 1)
    
    return max_len
```

```python
from collections import Counter

# 最小覆盖子串
def min_window(s: str, t: str) -> str:
    """
    LeetCode 76. Minimum Window Substring
    
    找出s中包含t所有字符的最小子串
    
    时间复杂度: O(m + n)
    空间复杂度: O(字符集大小)
    """
    if not s or not t:
        return ""
    
    # 需要的字符计数
    need = Counter(t)
    missing = len(t)  # 还缺少的字符数
    
    left = 0
    result = ""
    min_len = float('inf')
    
    for right, char in enumerate(s):
        # 扩展窗口
        if need[char] > 0:
            missing -= 1
        need[char] -= 1
        
        # 收缩窗口
        while missing == 0:
            # 更新结果
            if right - left + 1 < min_len:
                min_len = right - left + 1
                result = s[left:right + 1]
            
            # 移出左边元素
            need[s[left]] += 1
            if need[s[left]] > 0:
                missing += 1
            left += 1
    
    return result


# 找到字符串中所有字母异位词
def find_anagrams(s: str, p: str) -> list:
    """
    LeetCode 438. Find All Anagrams in a String
    
    返回s中所有p的异位词的起始索引
    """
    if len(s) < len(p):
        return []
    
    p_count = Counter(p)
    s_count = Counter()
    result = []
    
    for i, char in enumerate(s):
        s_count[char] += 1
        
        # 移除窗口外的字符
        if i >= len(p):
            left_char = s[i - len(p)]
            s_count[left_char] -= 1
            if s_count[left_char] == 0:
                del s_count[left_char]
        
        # 比较计数器
        if s_count == p_count:
            result.append(i - len(p) + 1)
    
    return result


# 替换后的最长重复字符
def character_replacement(s: str, k: int) -> int:
    """
    LeetCode 424. Longest Repeating Character Replacement
    
    最多替换k个字符，求最长的相同字符子串
    """
    count = {}
    left = 0
    max_count = 0  # 窗口内出现最多的字符次数
    max_len = 0
    
    for right, char in enumerate(s):
        count[char] = count.get(char, 0) + 1
        max_count = max(max_count, count[char])
        
        # 需要替换的字符数 = 窗口大小 - 最多的字符数
        # 如果需要替换的超过k，收缩窗口
        while right - left + 1 - max_count > k:
            count[s[left]] -= 1
            left += 1
        
        max_len = max(max_len, right - left + 1)
    
    return max_len
```

```python
# 示例2: 三数之和
def three_sum(nums: list) -> list:
    """
    LeetCode 15. 3Sum
    
    找出所有和为0的三元组（不重复）
    
    时间复杂度: O(n²)
    """
    nums.sort()
    result = []
    n = len(nums)
    
    for i in range(n - 2):
        # 跳过重复
        if i > 0 and nums[i] == nums[i - 1]:
            continue
        
        # 剪枝
        if nums[i] > 0:
            break
        
        left, right = i + 1, n - 1
        while left < right:
            total = nums[i] + nums[left] + nums[right]
            if total == 0:
                result.append([nums[i], nums[left], nums[right]])
                # 跳过重复
                while left < right and nums[left] == nums[left + 1]:
                    left += 1
                while left < right and nums[right] == nums[right - 1]:
                    right -= 1
                left += 1
                right -= 1
            elif total < 0:
                left += 1
            else:
                right -= 1
    
    return result


# 示例3: 盛最多水的容器
def max_area(height: list) -> int:
    """
    LeetCode 11. Container With Most Water
    
    找出能盛最多水的两条线
    """
    left, right = 0, len(height) - 1
    max_water = 0
    
    while left < right:
        # 面积 = 宽度 × 较短边
        area = (right - left) * min(height[left], height[right])
        max_water = max(max_water, area)
        
        # 移动较短的边（可能找到更高的）
        if height[left] < height[right]:
            left += 1
        else:
            right -= 1
    
    return max_water


# 示例4: 验证回文串
def is_palindrome(s: str) -> bool:
    """
    LeetCode 125. Valid Palindrome
    
    忽略大小写和非字母数字字符
    """
    left, right = 0, len(s) - 1
    
    while left < right:
        # 跳过非字母数字
        while left < right and not s[left].isalnum():
            left += 1
        while left < right and not s[right].isalnum():
            right -= 1
        
        if s[left].lower() != s[right].lower():
            return False
        
        left += 1
        right -= 1
    
    return True
```

```python
# 示例1: 判断链表是否有环
def has_cycle(head) -> bool:
    """
    LeetCode 141. Linked List Cycle
    
    Floyd判圈算法（龟兔赛跑）
    
    时间复杂度: O(n)
    空间复杂度: O(1)
    """
    if not head or not head.next:
        return False
    
    slow = head
    fast = head.next
    
    while fast and fast.next:
        if slow == fast:
            return True
        slow = slow.next
        fast = fast.next.next
    
    return False


# 示例2: 找到环的入口
def detect_cycle(head):
    """
    LeetCode 142. Linked List Cycle II
    
    找到环的起始节点
    """
    if not head or not head.next:
        return None
    
    slow = fast = head
    
    # 第一阶段：找到相遇点
    while fast and fast.next:
        slow = slow.next
        fast = fast.next.next
        if slow == fast:
            break
    else:
        return None  # 无环
    
    # 第二阶段：找到环入口
    slow = head
    while slow != fast:
        slow = slow.next
        fast = fast.next
    
    return slow
```

```python
    从后往前合并，避免覆盖

            i -= 1
        else:
            nums1[k] = nums2[j]
            j -= 1
        k -= 1
```

```python
def build_prefix_sum(nums: list) -> list:
    """
    构建前缀和数组
    
    参数:
        nums: List[int] - 原始数组
    返回:
        List[int] - 前缀和数组，prefix[i] = sum(nums[0:i])
    
    时间复杂度: O(n)
    空间复杂度: O(n)
    
    性质:
        - prefix[0] = 0
        - prefix[i] = nums[0] + nums[1] + ... + nums[i-1]
        - 区间和 sum(nums[l:r+1]) = prefix[r+1] - prefix[l]
    """
    n = len(nums)
    prefix = [0] * (n + 1)
    for i in range(n):
        prefix[i + 1] = prefix[i] + nums[i]
    return prefix


def range_sum(prefix: list, l: int, r: int) -> int:
    """
    查询区间和 [l, r]
    
    时间复杂度: O(1)
    """
    return prefix[r + 1] - prefix[l]

def subarray_sum(nums: list, k: int) -> int:
    """
    LeetCode 560. Subarray Sum Equals K
    
    统计和为k的连续子数组个数
    
    关键思想:
        如果 prefix[j] - prefix[i] = k
        则 sum(nums[i:j]) = k
    """
    from collections import defaultdict
    
    count = 0
    prefix_sum = 0
    prefix_count = defaultdict(int)
    prefix_count[0] = 1  # 空前缀
    
    for num in nums:
        prefix_sum += num
        # 找有多少个前缀和等于 prefix_sum - k
        count += prefix_count[prefix_sum - k]
        prefix_count[prefix_sum] += 1
    
    return count

def find_max_length(nums: list) -> int:

    """
```

```python
def build_prefix_sum_2d(matrix: list) -> list:
    """
    构建二维前缀和
    
    参数:
        matrix: List[List[int]] - m×n矩阵
    返回:
        List[List[int]] - (m+1)×(n+1)前缀和矩阵
    
    性质:
        prefix[i][j] = sum(matrix[0:i][0:j])
    """
    if not matrix or not matrix[0]:
        return [[]]
    
    m, n = len(matrix), len(matrix[0])
    prefix = [[0] * (n + 1) for _ in range(m + 1)]
    
    for i in range(1, m + 1):
        for j in range(1, n + 1):
            prefix[i][j] = (prefix[i-1][j] + prefix[i][j-1] 
                           - prefix[i-1][j-1] + matrix[i-1][j-1])
    
    return prefix


def region_sum(prefix: list, r1: int, c1: int, r2: int, c2: int) -> int:
    """
    查询区域和 [(r1,c1), (r2,c2)]
    
    时间复杂度: O(1)
    """
    return (prefix[r2+1][c2+1] - prefix[r1][c2+1] 
            - prefix[r2+1][c1] + prefix[r1][c1])

class NumMatrix:

    LeetCode 304. Range Sum Query 2D - Immutable
    """
    def __init__(self, matrix: list):

            self.prefix = [[]]
            return
        
        m, n = len(matrix), len(matrix[0])
        self.prefix = [[0] * (n + 1) for _ in range(m + 1)]

                    self.prefix[i-1][j] + self.prefix[i][j-1]
                    - self.prefix[i-1][j-1] + matrix[i-1][j-1]
                )
    
    def sumRegion(self, row1: int, col1: int, row2: int, col2: int) -> int:
        return (
            self.prefix[row2+1][col2+1] - self.prefix[row1][col2+1]
            - self.prefix[row2+1][col1] + self.prefix[row1][col1]
        )
```

## 4.1 一维差分

```python
def build_diff_array(nums: list) -> list:
    """
    构建差分数组
    
    参数:
        nums: List[int] - 原始数组
    返回:
        List[int] - 差分数组
    
    性质:
        diff[0] = nums[0]
        diff[i] = nums[i] - nums[i-1] (i > 0)
        
    用途:
        O(1) 区间加操作
    """
    n = len(nums)
    diff = [0] * n
    diff[0] = nums[0]
    for i in range(1, n):
        diff[i] = nums[i] - nums[i - 1]
    return diff


def range_add(diff: list, l: int, r: int, val: int) -> None:
    """
    区间加操作：给 [l, r] 每个元素加val
    
    时间复杂度: O(1)
    """
    diff[l] += val
    if r + 1 < len(diff):
        diff[r + 1] -= val


def restore_from_diff(diff: list) -> list:
    """
    从差分数组还原原数组
    
    时间复杂度: O(n)
    """
    nums = [0] * len(diff)
    nums[0] = diff[0]
    for i in range(1, len(diff)):
        nums[i] = nums[i - 1] + diff[i]
    return nums


# 使用accumulate还原
from itertools import accumulate

def restore_from_diff_v2(diff: list) -> list:
    return list(accumulate(diff))


# 示例1: 航班预订统计
def corp_flight_bookings(bookings: list, n: int) -> list:
    """
    LeetCode 1109. Corporate Flight Bookings
    
    bookings[i] = [first, last, seats]
    表示从航班first到航班last预订了seats个座位
    返回每个航班的座位总数
    """
    diff = [0] * (n + 1)
    
    for first, last, seats in bookings:
        diff[first - 1] += seats
        diff[last] -= seats
    
    result = [0] * n
    result[0] = diff[0]
    for i in range(1, n):
        result[i] = result[i - 1] + diff[i]
    
    return result


# 示例2: 拼车
def car_pooling(trips: list, capacity: int) -> bool:
    """
    LeetCode 1094. Car Pooling
    
    trips[i] = [numPassengers, from, to]
    判断能否完成所有行程
    """
    # 找出最大位置
    max_pos = max(to for _, _, to in trips)
    diff = [0] * (max_pos + 1)
    
    for num, start, end in trips:
        diff[start] += num
        diff[end] -= num
    
    passengers = 0
    for d in diff:
        passengers += d
        if passengers > capacity:
            return False
    
    return True
```

## 4.2 二维差分

```python
def build_diff_2d(m: int, n: int) -> list:
    """
    构建二维差分数组
    """
    return [[0] * (n + 1) for _ in range(m + 1)]


def range_add_2d(diff: list, r1: int, c1: int, r2: int, c2: int, val: int) -> None:
    """
    二维区间加操作
    
    给区域 [(r1,c1), (r2,c2)] 每个元素加val
    
    时间复杂度: O(1)
    """
    diff[r1][c1] += val
    diff[r2 + 1][c1] -= val
    diff[r1][c2 + 1] -= val
    diff[r2 + 1][c2 + 1] += val


def restore_from_diff_2d(diff: list) -> list:
    """
    从二维差分数组还原原数组
    
    时间复杂度: O(m × n)
    """
    m, n = len(diff), len(diff[0])
    result = [[0] * n for _ in range(m)]
    
    for i in range(m):
        for j in range(n):
            result[i][j] = diff[i][j]
            if i > 0:
                result[i][j] += result[i-1][j]
            if j > 0:
                result[i][j] += result[i][j-1]
            if i > 0 and j > 0:
                result[i][j] -= result[i-1][j-1]
    
    return result
```

# 五、区间问题

## 5.1 区间合并

```python
def merge_intervals(intervals: list) -> list:
    """
    LeetCode 56. Merge Intervals
    
    合并所有重叠的区间
    
    时间复杂度: O(n log n)
    """
    if not intervals:
        return []
    
    # 按起点排序
    intervals.sort(key=lambda x: x[0])
    
    merged = [intervals[0]]
    
    for start, end in intervals[1:]:
        if start <= merged[-1][1]:
            # 重叠，扩展终点
            merged[-1][1] = max(merged[-1][1], end)
        else:
            merged.append([start, end])
    
    return merged


# 插入区间
def insert_interval(intervals: list, new_interval: list) -> list:
    """
    LeetCode 57. Insert Interval
    
    插入新区间并合并
    """
    result = []
    i = 0
    n = len(intervals)
    
    # 添加所有在新区间之前的
    while i < n and intervals[i][1] < new_interval[0]:
        result.append(intervals[i])
        i += 1
    
    # 合并重叠区间
    while i < n and intervals[i][0] <= new_interval[1]:
        new_interval[0] = min(new_interval[0], intervals[i][0])
        new_interval[1] = max(new_interval[1], intervals[i][1])
        i += 1
    result.append(new_interval)
    
    # 添加剩余区间
    result.extend(intervals[i:])
    
    return result
```

## 5.2 区间调度

```python
def erase_overlap_intervals(intervals: list) -> int:
    """
    LeetCode 435. Non-overlapping Intervals
    
    最少删除多少个区间可以使剩余区间不重叠
    
    贪心：按终点排序，优先保留结束早的
    """
    if not intervals:
        return 0
    
    # 按终点排序
    intervals.sort(key=lambda x: x[1])
    
    count = 0
    prev_end = float('-inf')
    
    for start, end in intervals:
        if start >= prev_end:
            # 不重叠，保留
            prev_end = end
        else:
            # 重叠，删除当前区间
            count += 1
    
    return count


# 用最少数量的箭引爆气球
def find_min_arrow_shots(points: list) -> int:
    """
    LeetCode 452. Minimum Number of Arrows to Burst Balloons
    
    每个气球是一个区间，一支箭可以射穿所有经过该点的气球
    求最少需要多少支箭
    """
    if not points:
        return 0
    
    points.sort(key=lambda x: x[1])
    
    arrows = 1
    prev_end = points[0][1]
    
    for start, end in points[1:]:
        if start > prev_end:
            arrows += 1
            prev_end = end
    
    return arrows

def max_meetings(intervals: list) -> int:

    最多能参加多少个不重叠的会议
    """

    count = 0
    prev_end = float('-inf')
    
    for start, end in intervals:
        if start >= prev_end:
            count += 1
            prev_end = end

# 会议室问题：需要多少会议室
def min_meeting_rooms(intervals: list) -> int:
    """
    LeetCode 253. Meeting Rooms II
    
    同时进行的会议最多有多少个
    """
    events = []
    for start, end in intervals:
        events.append((start, 1))   # 会议开始
        events.append((end, -1))    # 会议结束
    
    events.sort()
    
    rooms = 0
    max_rooms = 0
    
    for _, delta in events:
        rooms += delta
        max_rooms = max(max_rooms, rooms)
    
    return max_rooms
```

## 总结

### 模式选择指南

| 问题特征 | 推荐模式 |
|----------|----------|
| 连续子数组/子串 | 滑动窗口 |
| 有序数组操作 | 双指针 |
| 多次区间和查询 | 前缀和 |
| 多次区间加操作 | 差分数组 |
| 区间重叠问题 | 排序 + 贪心 |

### 时间复杂度

| 模式 | 时间复杂度 |
|------|------------|
| 固定窗口滑动 | O(n) |
| 可变窗口滑动 | O(n) |
| 双指针 | O(n) |
| 前缀和构建 | O(n) |
| 前缀和查询 | O(1) |
| 差分区间加 | O(1) |
| 差分还原 | O(n) |

### 关键记忆

1. **滑动窗口**: 右指针扩展，左指针收缩
2. **双指针**: 对撞（首尾）、快慢（链表）、同向（原地修改）
3. **前缀和**: prefix[r+1] - prefix[l] = 区间和
4. **差分**: diff[l] += val, diff[r+1] -= val
5. **区间调度**: 按终点排序，贪心选择
