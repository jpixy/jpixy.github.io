# 一、递归基础

## 1.1 递归三要素

```python
"""
递归三要素：

1. 递归终止条件（Base Case）
   - 最简单的情况，直接返回结果
   - 防止无限递归

2. 递归关系（Recursive Case）
   - 将大问题分解为小问题
   - 调用自身解决子问题

3. 返回值
   - 如何从子问题的解构建当前问题的解
"""

def factorial(n: int) -> int:
    """
    阶乘：n! = n × (n-1)!
    
    终止条件: n <= 1 时返回 1
    递归关系: f(n) = n × f(n-1)
    """
    # 终止条件
    if n <= 1:
        return 1
    # 递归关系
    return n * factorial(n - 1)


def fibonacci(n: int) -> int:
    """
    斐波那契数列：F(n) = F(n-1) + F(n-2)
    
    终止条件: n <= 1
    递归关系: f(n) = f(n-1) + f(n-2)
    """
    if n <= 1:
        return n
    return fibonacci(n - 1) + fibonacci(n - 2)


def power(x: float, n: int) -> float:
    """
    快速幂：x^n
    
    LeetCode 50. Pow(x, n)
    
    递归关系:
        x^n = (x^(n/2))^2      if n是偶数
        x^n = x × (x^(n/2))^2  if n是奇数
    """
    if n == 0:
        return 1
    if n < 0:
        return 1 / power(x, -n)
    
    half = power(x, n // 2)
    if n % 2 == 0:
        return half * half
    else:
        return x * half * half
```

## 1.2 递归思维模式

```python
"""
递归思维的核心：

1. 相信递归函数能正确工作
   - 不要试图跟踪每一层调用
   - 假设子问题已经解决

2. 找出问题的递归结构
   - 当前问题如何分解
   - 子问题的解如何组合

3. 明确函数的定义
   - 输入是什么
   - 输出是什么
"""

# 例1: 反转链表
def reverse_list(head):
    """
    LeetCode 206. Reverse Linked List
    
    函数定义: 反转以head为头的链表，返回新的头节点
    """
    # 终止条件
    if not head or not head.next:
        return head
    
    # 递归：反转后面的链表
    new_head = reverse_list(head.next)
    
    # 处理当前节点
    head.next.next = head
    head.next = None
    
    return new_head


# 例2: 合并两个有序链表
def merge_two_lists(l1, l2):
    """
    LeetCode 21. Merge Two Sorted Lists
    
    函数定义: 合并l1和l2，返回合并后的头节点
    """
    if not l1:
        return l2
    if not l2:
        return l1
    
    if l1.val < l2.val:
        l1.next = merge_two_lists(l1.next, l2)
        return l1
    else:
        l2.next = merge_two_lists(l1, l2.next)
        return l2


# 例3: 二叉树的最大深度
def max_depth(root) -> int:
    """
    LeetCode 104. Maximum Depth of Binary Tree
    
    函数定义: 返回以root为根的树的最大深度
    """
    if not root:
        return 0
    
    left_depth = max_depth(root.left)
    right_depth = max_depth(root.right)
    
    return max(left_depth, right_depth) + 1


# 例4: 验证二叉搜索树
def is_valid_bst(root) -> bool:
    """
    LeetCode 98. Validate Binary Search Tree
    
    使用辅助函数传递范围
    """
    def validate(node, min_val, max_val):
        if not node:
            return True
        
        if node.val <= min_val or node.val >= max_val:
            return False
        
        return (validate(node.left, min_val, node.val) and
                validate(node.right, node.val, max_val))
    
    return validate(root, float('-inf'), float('inf'))
```

```python
# 例1: 二叉树前序遍历（递归）
def preorder_recursive(root):
    result = []
    def dfs(node):
        if node:
            result.append(node.val)
            dfs(node.left)
            dfs(node.right)
    dfs(root)
    return result


# 例1: 二叉树前序遍历（迭代）
def preorder_iterative(root):
    if not root:
        return []
    
    result = []
    stack = [root]
    
    while stack:
        node = stack.pop()
        result.append(node.val)
        # 先右后左（栈是LIFO）
        if node.right:
            stack.append(node.right)
        if node.left:
            stack.append(node.left)
    
    return result


# 例2: 中序遍历（迭代）
def inorder_iterative(root):
    result = []
    stack = []
    curr = root
    
    while curr or stack:
        # 一直向左走
        while curr:
            stack.append(curr)
            curr = curr.left
        
        curr = stack.pop()
        result.append(curr.val)
        curr = curr.right
    
    return result


# 例3: 后序遍历（迭代）
def postorder_iterative(root):
    if not root:
        return []
    
    result = []
    stack = [root]
    
    while stack:
        node = stack.pop()
        result.append(node.val)
        if node.left:
            stack.append(node.left)
        if node.right:
            stack.append(node.right)
    
    return result[::-1]  # 反转
```

# 二、回溯算法

## 2.1 回溯模板

```python
"""
回溯算法模板：

def backtrack(路径, 选择列表):
    if 满足结束条件:
        结果.add(路径)
        return
    
    for 选择 in 选择列表:
        做选择（将选择加入路径）
        backtrack(路径, 选择列表)
        撤销选择（将选择从路径移除）

关键点：
1. 路径：已经做出的选择
2. 选择列表：当前可以做的选择
3. 结束条件：到达决策树底层
"""

def backtrack_template(nums):
    result = []
    
    def backtrack(path, choices):
        # 结束条件
        if 满足条件:
            result.append(path.copy())  # 注意复制！
            return
        
        for choice in choices:
            # 剪枝（可选）
            if 不满足约束:
                continue
            
            # 做选择
            path.append(choice)
            
            # 递归
            backtrack(path, new_choices)
            
            # 撤销选择
            path.pop()
    
    backtrack([], nums)
    return result
```

```python
# 1. 子集（无重复元素）
def subsets(nums: list) -> list:
    """
    LeetCode 78. Subsets
    
    返回所有子集
    
    时间复杂度: O(n × 2^n)
    空间复杂度: O(n)
    """
    result = []
    
    def backtrack(start, path):
        result.append(path.copy())
        
        for i in range(start, len(nums)):
            path.append(nums[i])
            backtrack(i + 1, path)
            path.pop()
    
    backtrack(0, [])
    return result


# 2. 子集II（有重复元素）
def subsets_with_dup(nums: list) -> list:
    """
    LeetCode 90. Subsets II
    
    包含重复元素，返回不重复的子集
    """
    result = []
    nums.sort()  # 排序以便去重
    
    def backtrack(start, path):
        result.append(path.copy())
        
        for i in range(start, len(nums)):
            # 跳过同层重复元素
            if i > start and nums[i] == nums[i - 1]:
                continue
            
            path.append(nums[i])
            backtrack(i + 1, path)
            path.pop()
    
    backtrack(0, [])
    return result
```

## 2.3 组合问题

```python
# 1. 组合
def combine(n: int, k: int) -> list:
    """
    LeetCode 77. Combinations
    
    从1到n选k个数的所有组合
    """
    result = []
    
    def backtrack(start, path):
        if len(path) == k:
            result.append(path.copy())
            return
        
        # 剪枝：剩余元素不够
        if len(path) + (n - start + 1) < k:
            return
        
        for i in range(start, n + 1):
            path.append(i)
            backtrack(i + 1, path)
            path.pop()
    
    backtrack(1, [])
    return result


# 2. 组合总和（可重复使用）
def combination_sum(candidates: list, target: int) -> list:
    """
    LeetCode 39. Combination Sum
    
    元素可以无限次使用
    """
    result = []
    candidates.sort()
    
    def backtrack(start, path, remaining):
        if remaining == 0:
            result.append(path.copy())
            return
        
        for i in range(start, len(candidates)):
            if candidates[i] > remaining:
                break  # 剪枝
            
            path.append(candidates[i])
            backtrack(i, path, remaining - candidates[i])  # 不是i+1
            path.pop()
    
    backtrack(0, [], target)
    return result


# 3. 组合总和II（不可重复使用）
def combination_sum2(candidates: list, target: int) -> list:
    """
    LeetCode 40. Combination Sum II
    
    每个元素只能使用一次，但数组中可能有重复
    """
    result = []
    candidates.sort()
    
    def backtrack(start, path, remaining):
        if remaining == 0:
            result.append(path.copy())
            return
        
        for i in range(start, len(candidates)):
            if candidates[i] > remaining:
                break
            # 同层去重
            if i > start and candidates[i] == candidates[i - 1]:
                continue
            
            path.append(candidates[i])
            backtrack(i + 1, path, remaining - candidates[i])
            path.pop()
    
    backtrack(0, [], target)
    return result
```

## 2.4 排列问题

```python
# 1. 全排列（无重复）
def permute(nums: list) -> list:
    """
    LeetCode 46. Permutations
    
    返回所有排列
    
    时间复杂度: O(n × n!)
    """
    result = []
    
    def backtrack(path, used):
        if len(path) == len(nums):
            result.append(path.copy())
            return
        
        for i in range(len(nums)):
            if used[i]:
                continue
            
            used[i] = True
            path.append(nums[i])
            backtrack(path, used)
            path.pop()
            used[i] = False
    
    backtrack([], [False] * len(nums))
    return result


# 使用交换的方法（更高效）
def permute_swap(nums: list) -> list:
    result = []
    
    def backtrack(first):
        if first == len(nums):
            result.append(nums.copy())
            return
        
        for i in range(first, len(nums)):
            nums[first], nums[i] = nums[i], nums[first]
            backtrack(first + 1)
            nums[first], nums[i] = nums[i], nums[first]
    
    backtrack(0)
    return result


# 2. 全排列II（有重复）
def permute_unique(nums: list) -> list:
    """
    LeetCode 47. Permutations II
    
    包含重复元素，返回不重复的排列
    """
    result = []
    nums.sort()
    
    def backtrack(path, used):
        if len(path) == len(nums):
            result.append(path.copy())
            return
        
        for i in range(len(nums)):
            if used[i]:
                continue
            # 同层去重：前一个相同元素没被使用
            if i > 0 and nums[i] == nums[i - 1] and not used[i - 1]:
                continue
            
            used[i] = True
            path.append(nums[i])
            backtrack(path, used)
            path.pop()
            used[i] = False
    
    backtrack([], [False] * len(nums))
    return result
```

```python
# 1. N皇后
def solve_n_queens(n: int) -> list:
    """
    LeetCode 51. N-Queens
    
    在n×n棋盘上放置n个皇后，使其不能互相攻击
    """
    result = []
    
    def backtrack(row, cols, diag1, diag2, board):
        if row == n:
            result.append([''.join(r) for r in board])
            return
        
        for col in range(n):
            if col in cols or (row - col) in diag1 or (row + col) in diag2:
                continue
            
            board[row][col] = 'Q'
            cols.add(col)
            diag1.add(row - col)
            diag2.add(row + col)
            
            backtrack(row + 1, cols, diag1, diag2, board)
            
            board[row][col] = '.'
            cols.remove(col)
            diag1.remove(row - col)
            diag2.remove(row + col)
    
    board = [['.' for _ in range(n)] for _ in range(n)]
    backtrack(0, set(), set(), set(), board)
    return result


# 2. 解数独
def solve_sudoku(board: list) -> None:
    """
    LeetCode 37. Sudoku Solver
    """
    def is_valid(row, col, num):
        # 检查行
        if num in board[row]:
            return False
        # 检查列
        if num in [board[i][col] for i in range(9)]:
            return False
        # 检查3x3宫格
        box_row, box_col = 3 * (row // 3), 3 * (col // 3)
        for i in range(box_row, box_row + 3):
            for j in range(box_col, box_col + 3):
                if board[i][j] == num:
                    return False
        return True
    
    def backtrack():
        for i in range(9):
            for j in range(9):
                if board[i][j] == '.':
                    for num in '123456789':
                        if is_valid(i, j, num):
                            board[i][j] = num
                            if backtrack():
                                return True
                            board[i][j] = '.'
                    return False
        return True
    
    backtrack()


# 3. 单词搜索
def exist(board: list, word: str) -> bool:
    """
    LeetCode 79. Word Search
    
    在二维网格中搜索单词
    """
    m, n = len(board), len(board[0])
    
    def backtrack(i, j, k):
        if k == len(word):
            return True
        
        if i < 0 or i >= m or j < 0 or j >= n:
            return False
        if board[i][j] != word[k]:
            return False
        
        # 标记已访问
        temp = board[i][j]
        board[i][j] = '#'
        
        # 四个方向搜索
        found = (backtrack(i + 1, j, k + 1) or
                 backtrack(i - 1, j, k + 1) or
                 backtrack(i, j + 1, k + 1) or
                 backtrack(i, j - 1, k + 1))
        
        # 恢复
        board[i][j] = temp
        
        return found
    
    for i in range(m):
        for j in range(n):
            if backtrack(i, j, 0):
                return True
    return False


# 4. 分割回文串
def partition(s: str) -> list:
    """
    LeetCode 131. Palindrome Partitioning
    
    将s分割成若干回文子串
    """
    result = []
    
    def is_palindrome(start, end):
        while start < end:
            if s[start] != s[end]:
                return False
            start += 1
            end -= 1
        return True
    
    def backtrack(start, path):
        if start == len(s):
            result.append(path.copy())
            return
        
        for end in range(start, len(s)):
            if is_palindrome(start, end):
                path.append(s[start:end + 1])
                backtrack(end + 1, path)
                path.pop()
    
    backtrack(0, [])
    return result
```

# 三、剪枝技巧

## 3.1 常见剪枝策略

```python
"""
剪枝策略：

1. 可行性剪枝
   - 当前状态不可能达到目标，提前终止

2. 最优性剪枝
   - 当前状态不可能比已知最优解更好

3. 对称性剪枝
   - 利用对称性避免重复搜索

4. 排序剪枝
   - 排序后可以利用单调性剪枝

5. 记忆化剪枝
   - 记录已搜索的状态，避免重复
"""

# 例1: 组合总和的多重剪枝
def combination_sum_with_pruning(candidates: list, target: int) -> list:
    result = []
    candidates.sort()  # 排序剪枝的基础
    
    def backtrack(start, path, remaining):
        if remaining == 0:
            result.append(path.copy())
            return
        
        for i in range(start, len(candidates)):
            # 可行性剪枝：当前元素已经超过剩余目标
            if candidates[i] > remaining:
                break  # 后面的更大，直接终止
            
            # 同层去重剪枝
            if i > start and candidates[i] == candidates[i - 1]:
                continue
            
            path.append(candidates[i])
            backtrack(i + 1, path, remaining - candidates[i])
            path.pop()
    
    backtrack(0, [], target)
    return result


# 例2: 火柴拼正方形
def makesquare(matchsticks: list) -> bool:
    """
    LeetCode 473. Matchsticks to Square
    
    将火柴分成4组，每组和相等
    """
    total = sum(matchsticks)
    if total % 4 != 0:
        return False
    
    side = total // 4
    matchsticks.sort(reverse=True)  # 从大到小排序
    sides = [0] * 4
    
    def backtrack(i):
        if i == len(matchsticks):
            return all(s == side for s in sides)
        
        for j in range(4):
            # 可行性剪枝
            if sides[j] + matchsticks[i] > side:
                continue
            
            # 对称性剪枝：如果当前边和之前某边相同，跳过
            if j > 0 and sides[j] == sides[j - 1]:
                continue
            
            sides[j] += matchsticks[i]
            if backtrack(i + 1):
                return True
            sides[j] -= matchsticks[i]
            
            # 最优性剪枝：如果当前边为空还放不下，后面也不行
            if sides[j] == 0:
                break
        
        return False
    
    return backtrack(0)
```

## 3.2 最优性剪枝

```python
# 例: 划分为K个相等的子集
def can_partition_k_subsets(nums: list, k: int) -> bool:
    """
    LeetCode 698. Partition to K Equal Sum Subsets
    """
    total = sum(nums)
    if total % k != 0:
        return False
    
    target = total // k
    nums.sort(reverse=True)
    
    if nums[0] > target:
        return False
    
    buckets = [0] * k
    
    def backtrack(i):
        if i == len(nums):
            return True
        
        seen = set()  # 记录已尝试的桶大小
        for j in range(k):
            if buckets[j] + nums[i] > target:
                continue
            
            # 对称性剪枝
            if buckets[j] in seen:
                continue
            seen.add(buckets[j])
            
            buckets[j] += nums[i]
            if backtrack(i + 1):
                return True
            buckets[j] -= nums[i]
            
            # 最优性剪枝
            if buckets[j] == 0:
                break
        
        return False
    
    return backtrack(0)
```

```python
# 当参数不可哈希时，使用字典

def longest_increasing_path(matrix: list) -> int:
    """
    LeetCode 329. Longest Increasing Path in a Matrix
    """
    if not matrix or not matrix[0]:
        return 0
    
    m, n = len(matrix), len(matrix[0])
    memo = {}  # (i, j) -> 最长路径长度
    
    def dfs(i, j):
        if (i, j) in memo:
            return memo[(i, j)]
        
        max_len = 1
        for di, dj in [(0, 1), (0, -1), (1, 0), (-1, 0)]:
            ni, nj = i + di, j + dj
            if 0 <= ni < m and 0 <= nj < n and matrix[ni][nj] > matrix[i][j]:
                max_len = max(max_len, 1 + dfs(ni, nj))
        
        memo[(i, j)] = max_len
        return max_len
    
    return max(dfs(i, j) for i in range(m) for j in range(n))


# 将列表转换为元组以便哈希
def word_break_memo(s: str, word_dict: list) -> bool:
    """
    LeetCode 139. Word Break
    """
    word_set = frozenset(word_dict)  # 不可变集合
    
    @lru_cache(maxsize=None)
    def dfs(start):
        if start == len(s):
            return True
        
        for end in range(start + 1, len(s) + 1):
            if s[start:end] in word_set and dfs(end):
                return True
        
        return False
    
    return dfs(0)
```

```python
# 例: 最长公共子序列

# 方法1: 记忆化递归（自顶向下）
def lcs_memo(text1: str, text2: str) -> int:
    @lru_cache(maxsize=None)
    def dp(i, j):
        if i == len(text1) or j == len(text2):
            return 0
        
        if text1[i] == text2[j]:
            return 1 + dp(i + 1, j + 1)
        else:
            return max(dp(i + 1, j), dp(i, j + 1))
    
    return dp(0, 0)


# 方法2: 动态规划（自底向上）
def lcs_dp(text1: str, text2: str) -> int:
    m, n = len(text1), len(text2)
    dp = [[0] * (n + 1) for _ in range(m + 1)]
    
    for i in range(m - 1, -1, -1):
        for j in range(n - 1, -1, -1):
            if text1[i] == text2[j]:
                dp[i][j] = 1 + dp[i + 1][j + 1]
            else:
                dp[i][j] = max(dp[i + 1][j], dp[i][j + 1])
    
    return dp[0][0]


# 例: 不同路径（带障碍物）
def unique_paths_with_obstacles(obstacleGrid: list) -> int:
    """
    LeetCode 63. Unique Paths II
    """
    m, n = len(obstacleGrid), len(obstacleGrid[0])
    
    @lru_cache(maxsize=None)
    def dp(i, j):
        if i >= m or j >= n or obstacleGrid[i][j] == 1:
            return 0
        if i == m - 1 and j == n - 1:
            return 1
        return dp(i + 1, j) + dp(i, j + 1)
    
    return dp(0, 0)
```

## 4.4 复杂状态的记忆化

```python
# 使用元组表示状态

def stickers_to_spell_word(stickers: list, target: str) -> int:
    """
    LeetCode 691. Stickers to Spell Word
    
    用贴纸拼出目标词的最少贴纸数
    """
    from collections import Counter
    
    sticker_counts = [Counter(s) for s in stickers]
    
    @lru_cache(maxsize=None)
    def dp(remaining):
        if not remaining:
            return 0
        
        remaining_count = Counter(remaining)
        result = float('inf')
        
        for sticker in sticker_counts:
            # 必须包含remaining的第一个字符（避免死循环）
            if sticker[remaining[0]] == 0:
                continue
            
            # 使用这个贴纸后的剩余字符
            new_remaining = []
            for c, count in remaining_count.items():
                new_remaining.append(c * max(0, count - sticker[c]))
            
            new_remaining = ''.join(new_remaining)
            if new_remaining != remaining:
                result = min(result, 1 + dp(new_remaining))
        
        return result
    
    ans = dp(target)
    return ans if ans != float('inf') else -1


# 使用frozenset作为状态
def shortest_path_all_keys(grid: list) -> int:
    """
    LeetCode 864. Shortest Path to Get All Keys
    
    收集所有钥匙的最短路径
    """
    from collections import deque
    
    m, n = len(grid), len(grid[0])
    start = None
    all_keys = 0
    
    for i in range(m):
        for j in range(n):
            if grid[i][j] == '@':
                start = (i, j)
            elif grid[i][j].islower():
                all_keys |= (1 << (ord(grid[i][j]) - ord('a')))
    
    # BFS: 状态 = (位置, 拥有的钥匙)
    queue = deque([(start[0], start[1], 0, 0)])  # (i, j, keys, steps)
    visited = {(start[0], start[1], 0)}
    
    while queue:
        i, j, keys, steps = queue.popleft()
        
        if keys == all_keys:
            return steps
        
        for di, dj in [(0, 1), (0, -1), (1, 0), (-1, 0)]:
            ni, nj = i + di, j + dj
            if 0 <= ni < m and 0 <= nj < n and grid[ni][nj] != '#':
                cell = grid[ni][nj]
                
                # 遇到锁但没有钥匙
                if cell.isupper() and not (keys & (1 << (ord(cell.lower()) - ord('a')))):
                    continue
                
                new_keys = keys
                if cell.islower():
                    new_keys |= (1 << (ord(cell) - ord('a')))
                
                if (ni, nj, new_keys) not in visited:
                    visited.add((ni, nj, new_keys))
                    queue.append((ni, nj, new_keys, steps + 1))
    
    return -1
```

## 总结

### 回溯问题分类

| 问题类型 | 特点 | 关键处理 |
|----------|------|----------|
| 子集 | 元素可选可不选 | start控制起点 |
| 组合 | 选择顺序无关 | start控制起点，长度限制 |
| 排列 | 选择顺序有关 | used数组标记 |
| 分割 | 切割位置选择 | 判断合法性 |
| 棋盘 | 二维放置 | 多维约束检查 |

### 去重方法

| 场景 | 方法 |
|------|------|
| 同层去重 | 排序 + i > start && nums[i] == nums[i-1] |
| 路径去重 | used数组 |
| 结果去重 | set存储 |

### 剪枝策略

| 类型 | 说明 |
|------|------|
| 可行性剪枝 | 不满足约束，直接跳过 |
| 最优性剪枝 | 不可能更优，提前终止 |
| 排序剪枝 | 排序后利用单调性 |
| 对称性剪枝 | 避免等价搜索 |

### 关键记忆

1. 回溯 = 递归 + 撤销选择
2. 子集/组合用start，排列用used
3. 有重复元素先排序再去重
4. 记忆化用@lru_cache或dict
5. 参数不可哈希时转换为tuple
