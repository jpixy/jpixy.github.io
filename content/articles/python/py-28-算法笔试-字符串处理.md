# 一、字符串基础

## 1.1 字符串创建与索引


## 1.2 常用字符串方法

```python
s = "  Hello World  "

"hello".count('l')      # 2 计数

"hello".endswith('lo')       # True
"hello".startswith(('he', 'ha'))  # True 元组参数
```

## 1.4 ASCII与字符转换

```python
# 字符转ASCII码
ord('A')               # 65
ord('a')               # 97
ord('0')               # 48

# ASCII码转字符
chr(65)                # 'A'
chr(97)                # 'a'

# 常用ASCII范围
# 'A'-'Z': 65-90
# 'a'-'z': 97-122
# '0'-'9': 48-57

# 判断字符类型
def is_upper(c):
    return 'A' <= c <= 'Z'

def is_lower(c):
    return 'a' <= c <= 'z'

def is_digit(c):
    return '0' <= c <= '9'

def is_alpha(c):
    return is_upper(c) or is_lower(c)

# 大小写转换
def to_lower(c):
    if is_upper(c):
        return chr(ord(c) + 32)
    return c

def to_upper(c):
    if is_lower(c):
        return chr(ord(c) - 32)
    return c
```

# 二、字符串算法技巧

## 2.1 常用模式

```python
# 1. 反转字符串
s = "hello"
s[::-1]                # "olleh"
''.join(reversed(s))   # "olleh"

# 2. 判断回文
def is_palindrome(s):
    return s == s[::-1]

def is_palindrome_v2(s):
    left, right = 0, len(s) - 1
    while left < right:
        if s[left] != s[right]:
            return False
        left += 1
        right -= 1
    return True

# 3. 统计字符频率
from collections import Counter
Counter("hello")       # Counter({'l': 2, 'h': 1, 'e': 1, 'o': 1})

# 4. 判断异位词
def is_anagram(s1, s2):
    return Counter(s1) == Counter(s2)

def is_anagram_v2(s1, s2):
    return sorted(s1) == sorted(s2)

# 5. 字符串转数字
def atoi(s):
    s = s.strip()
    if not s:
        return 0
    
    sign = 1
    i = 0
    if s[0] in ['+', '-']:
        sign = -1 if s[0] == '-' else 1
        i = 1
    
    result = 0
    while i < len(s) and s[i].isdigit():
        result = result * 10 + int(s[i])
        i += 1
    
    result *= sign
    return max(-2**31, min(2**31 - 1, result))

# 6. 压缩字符串
def compress(s):
    if not s:
        return ""
    
    result = []
    count = 1
    for i in range(1, len(s)):
        if s[i] == s[i-1]:
            count += 1
        else:
            result.append(s[i-1] + str(count))
            count = 1
    result.append(s[-1] + str(count))
    
    return ''.join(result)
```

## 2.2 子串问题

```python
# 1. 最长公共前缀
def longest_common_prefix(strs):
    if not strs:
        return ""
    
    prefix = strs[0]
    for s in strs[1:]:
        while not s.startswith(prefix):
            prefix = prefix[:-1]
            if not prefix:
                return ""
    return prefix

# 2. 最长回文子串（中心扩展）
def longest_palindrome(s):
    def expand(left, right):
        while left >= 0 and right < len(s) and s[left] == s[right]:
            left -= 1
            right += 1
        return s[left+1:right]
    
    result = ""
    for i in range(len(s)):
        # 奇数长度回文
        odd = expand(i, i)
        if len(odd) > len(result):
            result = odd
        # 偶数长度回文
        even = expand(i, i+1)
        if len(even) > len(result):
            result = even
    
    return result

# 3. 无重复字符的最长子串（滑动窗口）
def length_of_longest_substring(s):
    seen = {}
    left = 0
    max_len = 0
    
    for right, char in enumerate(s):
        if char in seen and seen[char] >= left:
            left = seen[char] + 1
        seen[char] = right
        max_len = max(max_len, right - left + 1)
    
    return max_len

# 4. 最小覆盖子串
def min_window(s, t):
    from collections import Counter
    
    need = Counter(t)
    missing = len(t)
    left = 0
    result = ""
    
    for right, char in enumerate(s):
        if need[char] > 0:
            missing -= 1
        need[char] -= 1
        
        while missing == 0:
            if not result or right - left + 1 < len(result):
                result = s[left:right+1]
            
            need[s[left]] += 1
            if need[s[left]] > 0:
                missing += 1
            left += 1
    
    return result
```

## 2.3 字符串匹配

```python
# 1. 朴素匹配
def naive_search(text, pattern):
    n, m = len(text), len(pattern)
    for i in range(n - m + 1):
        if text[i:i+m] == pattern:
            return i
    return -1

# 2. KMP算法
def kmp_search(text, pattern):
    def build_lps(pattern):
        lps = [0] * len(pattern)
        length = 0
        i = 1
        while i < len(pattern):
            if pattern[i] == pattern[length]:
                length += 1
                lps[i] = length
                i += 1
            elif length != 0:
                length = lps[length - 1]
            else:
                lps[i] = 0
                i += 1
        return lps
    
    if not pattern:
        return 0
    
    lps = build_lps(pattern)
    i = j = 0
    
    while i < len(text):
        if pattern[j] == text[i]:
            i += 1
            j += 1
        
        if j == len(pattern):
            return i - j
        elif i < len(text) and pattern[j] != text[i]:
            if j != 0:
                j = lps[j - 1]
            else:
                i += 1
    
    return -1

# 3. Rabin-Karp（滚动哈希）
def rabin_karp(text, pattern):
    n, m = len(text), len(pattern)
    if m > n:
        return -1
    
    base = 26
    mod = 10**9 + 7
    
    # 计算pattern的哈希值
    pattern_hash = 0
    text_hash = 0
    power = 1
    
    for i in range(m):
        pattern_hash = (pattern_hash * base + ord(pattern[i])) % mod
        text_hash = (text_hash * base + ord(text[i])) % mod
        if i < m - 1:
            power = power * base % mod
    
    # 滚动匹配
    for i in range(n - m + 1):
        if pattern_hash == text_hash:
            if text[i:i+m] == pattern:
                return i
        
        if i < n - m:
            text_hash = (text_hash - ord(text[i]) * power) % mod
            text_hash = (text_hash * base + ord(text[i+m])) % mod
            text_hash = (text_hash + mod) % mod
    
    return -1
```

# 三、正则表达式re模块

## 3.1 基础用法

```python
import re

# 查找第一个匹配
match = re.search(r'\d+', 'hello 123 world 456')
if match:
    print(match.group())  # '123'
    print(match.start())  # 6 起始位置
    print(match.end())    # 9 结束位置
    print(match.span())   # (6, 9)

# 从开头匹配
match = re.match(r'\d+', '123 hello')
print(match.group())      # '123'

match = re.match(r'\d+', 'hello 123')
print(match)              # None

# 完全匹配
match = re.fullmatch(r'\d+', '123')
print(match.group())      # '123'

match = re.fullmatch(r'\d+', '123 hello')
print(match)              # None

# 查找所有匹配
re.findall(r'\d+', 'a1b2c3')  # ['1', '2', '3']

# 迭代查找
for match in re.finditer(r'\d+', 'a1b22c333'):
    print(match.group(), match.span())
# '1' (1, 2)
# '22' (3, 5)
# '333' (6, 9)
```

## 3.2 替换与分割

```python
import re

# 替换
re.sub(r'\d+', 'X', 'a1b2c3')           # 'aXbXcX'
re.sub(r'\d+', 'X', 'a1b2c3', count=1)  # 'aXb2c3'

# 使用函数替换
def double(match):
    return str(int(match.group()) * 2)

re.sub(r'\d+', double, 'a1b2c3')  # 'a2b4c6'

# 替换并返回次数
re.subn(r'\d+', 'X', 'a1b2c3')  # ('aXbXcX', 3)

# 分割
re.split(r'\d+', 'a1b2c3')      # ['a', 'b', 'c', '']
re.split(r'\d+', 'a1b2c3', maxsplit=1)  # ['a', 'b2c3']

# 保留分隔符
re.split(r'(\d+)', 'a1b2c3')    # ['a', '1', 'b', '2', 'c', '3', '']
```

## 3.3 常用模式

```python
import re

# 字符类
r'\d'       # 数字 [0-9]
r'\D'       # 非数字
r'\w'       # 单词字符 [a-zA-Z0-9_]
r'\W'       # 非单词字符
r'\s'       # 空白字符
r'\S'       # 非空白字符
r'.'        # 任意字符（除换行）
r'[abc]'    # a或b或c
r'[^abc]'   # 非a、b、c
r'[a-z]'    # a到z

# 量词
r'a*'       # 0个或多个a
r'a+'       # 1个或多个a
r'a?'       # 0个或1个a
r'a{3}'     # 恰好3个a
r'a{2,4}'   # 2到4个a
r'a{2,}'    # 至少2个a

# 非贪婪
r'a*?'      # 最少匹配
r'a+?'      # 最少匹配
r'a??'      # 最少匹配

# 边界
r'^'        # 字符串开头
r'$'        # 字符串结尾
r'\b'       # 单词边界
r'\B'       # 非单词边界

# 分组
r'(abc)'    # 捕获组
r'(?:abc)'  # 非捕获组
r'(?P<name>abc)'  # 命名组

# 引用
r'\1'       # 引用第1组
r'(?P=name)'  # 引用命名组

# 前后断言
r'(?=abc)'  # 正向肯定断言
r'(?!abc)'  # 正向否定断言
r'(?<=abc)' # 反向肯定断言
r'(?<!abc)' # 反向否定断言
```

## 3.4 编译与标志

```python
import re

# 编译正则（重复使用时更高效）
pattern = re.compile(r'\d+')
pattern.findall('a1b2c3')  # ['1', '2', '3']

# 标志
re.IGNORECASE  # re.I 忽略大小写
re.MULTILINE   # re.M 多行模式（^$匹配每行）
re.DOTALL      # re.S .匹配换行符
re.VERBOSE     # re.X 允许注释

# 使用标志
re.findall(r'hello', 'Hello World', re.I)  # ['Hello']

# 多行模式
text = "line1\nline2\nline3"
re.findall(r'^line\d', text, re.M)  # ['line1', 'line2', 'line3']

# 详细模式（便于阅读）
pattern = re.compile(r'''
    \d{3}    # 区号
```

## 3.5 分组与捕获

```python
import re

# 基本分组
match = re.search(r'(\d+)-(\d+)', '123-456')
match.group(0)    # '123-456' 完整匹配
match.group(1)    # '123' 第1组
match.group(2)    # '456' 第2组
match.groups()    # ('123', '456')

# 命名分组
match = re.search(r'(?P<area>\d+)-(?P<number>\d+)', '123-456')
match.group('area')    # '123'
match.group('number')  # '456'
match.groupdict()      # {'area': '123', 'number': '456'}

# 引用分组
# 匹配重复的单词
re.search(r'(\w+)\s+\1', 'hello hello world')  # 匹配 'hello hello'

# findall与分组
re.findall(r'(\d+)-(\d+)', 'a123-456b789-012c')
# [('123', '456'), ('789', '012')]

# 非捕获组
re.findall(r'(?:\d+)-(\d+)', 'a123-456b789-012c')
# ['456', '012']
```

## 3.6 常用正则示例

```python
import re

# 1. 验证邮箱
email_pattern = r'^[\w\.-]+@[\w\.-]+\.\w+$'
re.match(email_pattern, 'test@example.com')  # 匹配

# 2. 验证手机号（中国）
phone_pattern = r'^1[3-9]\d{9}$'
re.match(phone_pattern, '13812345678')  # 匹配

# 3. 验证IP地址
ip_pattern = r'^(\d{1,3}\.){3}\d{1,3}$'
# 更严格的版本
ip_pattern = r'^((25[0-5]|2[0-4]\d|[01]?\d\d?)\.){3}(25[0-5]|2[0-4]\d|[01]?\d\d?)$'

# 4. 提取URL
url_pattern = r'https?://[\w\.-]+(/[\w\.-]*)*'
re.findall(url_pattern, 'Visit https://example.com/path for more info')

# 5. 提取HTML标签内容
html = '<div>Hello</div><span>World</span>'
re.findall(r'<(\w+)>([^<]+)</\1>', html)
# [('div', 'Hello'), ('span', 'World')]

# 6. 驼峰转下划线
def camel_to_snake(s):
    return re.sub(r'([a-z])([A-Z])', r'\1_\2', s).lower()

camel_to_snake('camelCase')  # 'camel_case'

# 7. 下划线转驼峰
def snake_to_camel(s):
    parts = s.split('_')
    return parts[0] + ''.join(p.capitalize() for p in parts[1:])

snake_to_camel('snake_case')  # 'snakeCase'

# 8. 删除多余空格
def normalize_spaces(s):
    return re.sub(r'\s+', ' ', s).strip()

normalize_spaces('  hello   world  ')  # 'hello world'

# 9. 提取数字（包括小数和负数）
number_pattern = r'-?\d+\.?\d*'
re.findall(number_pattern, 'a-1.5b2c3.14d-100')
# ['-1.5', '2', '3.14', '-100']

# 10. 密码验证（至少8位，包含大小写字母和数字）
def validate_password(password):
    if len(password) < 8:
        return False
    if not re.search(r'[a-z]', password):
        return False
    if not re.search(r'[A-Z]', password):
        return False
    if not re.search(r'\d', password):
        return False
    return True
```

## 总结

### 字符串方法速查

| 类别 | 方法 | 说明 |
|------|------|------|
| 大小写 | `lower()`, `upper()`, `title()` | 转换大小写 |
| 去空白 | `strip()`, `lstrip()`, `rstrip()` | 去除空白 |
| 分割 | `split()`, `splitlines()` | 分割字符串 |
| 连接 | `join()` | 连接字符串 |
| 查找 | `find()`, `index()`, `count()` | 查找子串 |
| 替换 | `replace()` | 替换子串 |
| 判断 | `startswith()`, `endswith()`, `isdigit()` | 判断类型 |

### 正则表达式速查

| 模式 | 说明 |
|------|------|
| `\d`, `\D` | 数字/非数字 |
| `\w`, `\W` | 单词字符/非单词字符 |
| `\s`, `\S` | 空白/非空白 |
| `*`, `+`, `?` | 0+, 1+, 0或1 |
| `^`, `$` | 开头/结尾 |
| `()` | 捕获组 |
| `(?:)` | 非捕获组 |
| `(?P<name>)` | 命名组 |

### 关键记忆

1. 字符串是不可变的，修改返回新字符串
2. `split()`不带参数按任意空白分割
3. `join()`的调用者是分隔符
4. 正则用原始字符串`r'...'`
5. `findall`有分组时返回分组内容
