# 一、数值操作

## 1.1 基础数值函数

```python
# === abs() ===
# 返回绝对值
abs(-5)           # 5
abs(-3.14)        # 3.14
abs(3 + 4j)       # 5.0 (复数的模)

# === max() / min() ===
# 返回最大/最小值
max(1, 2, 3)      # 3
min([1, 2, 3])    # 1
max([])           # ValueError!
max([], default=0)  # 0 (设置默认值)

# 使用key参数
max(['a', 'bb', 'ccc'], key=len)  # 'ccc'
min([-5, 3, -2], key=abs)         # -2

# 多个可迭代对象
max([1, 2], [3, 4])  # [3, 4] (按字典序比较)

# === sum() ===
# 求和
sum([1, 2, 3])        # 6
sum([1, 2, 3], 10)    # 16 (从10开始累加)
sum([[1, 2], [3]], [])  # [1, 2, 3] (展平列表)

# === pow() ===
# 幂运算
pow(2, 10)            # 1024
pow(2, 10, 1000)      # 24 (2^10 % 1000，快速幂)
2 ** 10               # 1024

# === divmod() ===
# 同时返回商和余数
divmod(17, 5)         # (3, 2)
q, r = divmod(17, 5)

# === round() ===
# 四舍五入（银行家舍入法）
round(2.5)            # 2 (注意！)
round(3.5)            # 4
round(2.555, 2)       # 2.55 (浮点精度问题)
round(2.567, 2)       # 2.57
```

## 1.2 类型转换

```python
# === int() ===
int(3.7)              # 3 (截断)
int(-3.7)             # -3 (向零取整)
int('42')             # 42
int('1010', 2)        # 10 (二进制转十进制)
int('ff', 16)         # 255 (十六进制转十进制)

# === float() ===
float('3.14')         # 3.14
float('inf')          # inf
float('-inf')         # -inf
float('nan')          # nan

# === bool() ===
bool(0)               # False
bool('')              # False
bool([])              # False
bool(None)            # False
bool(1)               # True
bool([0])             # True (非空列表)

# === str() ===
str(42)               # '42'
str([1, 2, 3])        # '[1, 2, 3]'

# === bin() / oct() / hex() ===
bin(10)               # '0b1010'
oct(10)               # '0o12'
hex(255)              # '0xff'

# === chr() / ord() ===
chr(65)               # 'A'
chr(97)               # 'a'
ord('A')              # 65
ord('a')              # 97
```

# 二、序列操作

## 2.1 遍历与索引

```python
# === enumerate() ===
# 返回索引和元素
for i, val in enumerate(['a', 'b', 'c']):
    print(i, val)  # 0 a, 1 b, 2 c

for i, val in enumerate(['a', 'b', 'c'], start=1):
    print(i, val)  # 1 a, 2 b, 3 c

# 转换为列表
list(enumerate('abc'))  # [(0, 'a'), (1, 'b'), (2, 'c')]

# === zip() ===
# 并行迭代多个可迭代对象
list(zip([1, 2, 3], ['a', 'b', 'c']))
# [(1, 'a'), (2, 'b'), (3, 'c')]

# 长度不同时取最短
list(zip([1, 2], ['a', 'b', 'c']))
# [(1, 'a'), (2, 'b')]

# 使用zip_longest填充
from itertools import zip_longest
list(zip_longest([1, 2], ['a', 'b', 'c'], fillvalue=0))
# [(1, 'a'), (2, 'b'), (0, 'c')]

# 解压缩
pairs = [(1, 'a'), (2, 'b'), (3, 'c')]
nums, chars = zip(*pairs)
# nums = (1, 2, 3), chars = ('a', 'b', 'c')

# 矩阵转置
matrix = [[1, 2, 3], [4, 5, 6]]
transposed = list(zip(*matrix))
# [(1, 4), (2, 5), (3, 6)]

# === reversed() ===
# 反向迭代
list(reversed([1, 2, 3]))   # [3, 2, 1]
list(reversed('abc'))       # ['c', 'b', 'a']
'abc'[::-1]                 # 'cba' (字符串反转更快)

# === sorted() ===
# 返回排序后的新列表
sorted([3, 1, 2])               # [1, 2, 3]
sorted([3, 1, 2], reverse=True) # [3, 2, 1]
sorted('bca')                   # ['a', 'b', 'c']

# 使用key
sorted(['apple', 'pie', 'banana'], key=len)
# ['pie', 'apple', 'banana']

sorted([(1, 'b'), (2, 'a')], key=lambda x: x[1])
# [(2, 'a'), (1, 'b')]

# 多关键字排序
sorted([(1, 2), (3, 1), (1, 1)], key=lambda x: (x[0], x[1]))
# [(1, 1), (1, 2), (3, 1)]
```

## 2.2 函数式编程

```python
# === map() ===
# 对每个元素应用函数
list(map(str, [1, 2, 3]))       # ['1', '2', '3']
list(map(int, '123'))           # [1, 2, 3]
list(map(lambda x: x*2, [1, 2, 3]))  # [2, 4, 6]

# 多个可迭代对象
list(map(lambda x, y: x + y, [1, 2], [3, 4]))  # [4, 6]

# 更Pythonic的方式：列表推导
[x * 2 for x in [1, 2, 3]]      # [2, 4, 6]

# === filter() ===
# 过滤元素
list(filter(lambda x: x > 0, [-1, 0, 1, 2]))  # [1, 2]
list(filter(None, [0, 1, '', 'a', [], [1]]))  # [1, 'a', [1]]

# 更Pythonic的方式：列表推导
[x for x in [-1, 0, 1, 2] if x > 0]  # [1, 2]

# === reduce() ===
# 从functools导入
from functools import reduce

reduce(lambda x, y: x + y, [1, 2, 3, 4])  # 10
reduce(lambda x, y: x * y, [1, 2, 3, 4])  # 24
reduce(lambda x, y: x + y, [1, 2, 3], 10) # 16 (初始值10)

# 使用operator模块
import operator
reduce(operator.add, [1, 2, 3, 4])  # 10
reduce(operator.mul, [1, 2, 3, 4])  # 24
```

## 2.3 判断函数

```python
# === all() ===
# 所有元素为True返回True
all([True, True, True])    # True
all([True, False, True])   # False
all([1, 2, 3])             # True
all([1, 0, 3])             # False
all([])                    # True (空序列)

# 常用：检查条件
all(x > 0 for x in [1, 2, 3])   # True
all(c.isdigit() for c in '123') # True

# === any() ===
# 任一元素为True返回True
any([False, True, False])  # True
any([False, False])        # False
any([0, 0, 1])             # True
any([])                    # False (空序列)

# 常用：检查是否存在
any(x > 10 for x in [1, 2, 3])  # False
any(c.isalpha() for c in '12a') # True

# === isinstance() ===
# 检查类型
isinstance(42, int)              # True
isinstance(42, (int, float))     # True (多个类型)
isinstance([1, 2], list)         # True

# === len() ===
# 返回长度
len([1, 2, 3])             # 3
len('hello')               # 5
len({1, 2, 3})             # 3
len({'a': 1, 'b': 2})      # 2
```

# 三、创建与转换

## 3.1 数据结构创建

```python
# === range() ===
# 生成整数序列
list(range(5))           # [0, 1, 2, 3, 4]
list(range(1, 6))        # [1, 2, 3, 4, 5]
list(range(0, 10, 2))    # [0, 2, 4, 6, 8]
list(range(5, 0, -1))    # [5, 4, 3, 2, 1]

# 常用场景
for i in range(n):       # 0到n-1
for i in range(1, n+1):  # 1到n
for i in range(n-1, -1, -1):  # n-1到0（逆序）

# === list() / tuple() / set() / dict() ===
list('abc')              # ['a', 'b', 'c']
tuple([1, 2, 3])         # (1, 2, 3)
set([1, 2, 2, 3])        # {1, 2, 3}
dict([('a', 1), ('b', 2)])  # {'a': 1, 'b': 2}

# === frozenset() ===
# 不可变集合（可作为字典键）
fs = frozenset([1, 2, 3])
d = {fs: 'value'}

# === bytearray() / bytes() ===
bytes([65, 66, 67])      # b'ABC'
bytearray([65, 66, 67])  # bytearray(b'ABC')
```

## 3.2 对象操作

```python
# === id() ===
# 返回对象的唯一标识
a = [1, 2, 3]
b = a
id(a) == id(b)           # True (同一对象)

# === type() ===
# 返回对象类型
type(42)                 # <class 'int'>
type([])                 # <class 'list'>
type(42).__name__        # 'int'

# === hash() ===
# 返回对象的哈希值
hash(42)                 # 42
hash('hello')            # 一个整数
hash((1, 2, 3))          # 可哈希
# hash([1, 2, 3])        # TypeError: 列表不可哈希

# === callable() ===
# 检查对象是否可调用
callable(len)            # True
callable(42)             # False
callable(lambda x: x)    # True

# === getattr() / setattr() / hasattr() ===
class Obj:
    x = 1

obj = Obj()
getattr(obj, 'x')        # 1
getattr(obj, 'y', 0)     # 0 (默认值)
hasattr(obj, 'x')        # True
setattr(obj, 'y', 2)     # obj.y = 2
```

# 四、输入输出

## 4.1 输入

```python
# === input() ===
# 读取一行输入
s = input()              # 读取字符串
n = int(input())         # 读取整数

# 读取多个整数
a, b = map(int, input().split())
nums = list(map(int, input().split()))

# 读取n行
n = int(input())
lines = [input() for _ in range(n)]

# 快速输入（大数据量）
import sys
input = sys.stdin.readline
n = int(input())
```

## 4.2 输出

```python
# === print() ===
print('hello')                    # hello
print('a', 'b', 'c')              # a b c
print('a', 'b', sep=',')          # a,b
print('hello', end=' ')           # 不换行
print(*[1, 2, 3])                 # 1 2 3
print(' '.join(map(str, [1, 2, 3])))  # 1 2 3

# 格式化输出
print(f'{3.14159:.2f}')           # 3.14
print(f'{42:05d}')                # 00042
print(f'{42:>10}')                # '        42'
print(f'{1000000:,}')             # 1,000,000

# 快速输出（大数据量）
import sys
print = sys.stdout.write
print(str(n) + '\n')

# 收集后一次输出
result = []
result.append(str(x))
print('\n'.join(result))
```

# 五、常用技巧速查

## 5.1 列表操作

```python
# 创建
arr = [0] * n                     # n个0
matrix = [[0] * m for _ in range(n)]  # n×m矩阵
arr = list(range(n))              # 0到n-1

# 切片
arr[::2]                          # 偶数索引
arr[1::2]                         # 奇数索引
arr[::-1]                         # 反转
arr[-3:]                          # 最后3个

# 常用方法
arr.append(x)                     # 尾部添加
arr.pop()                         # 尾部弹出
arr.pop(0)                        # 头部弹出（慢！用deque）
arr.insert(i, x)                  # 插入
arr.remove(x)                     # 删除第一个x
arr.index(x)                      # 查找索引
arr.count(x)                      # 计数
arr.sort()                        # 原地排序
arr.reverse()                     # 原地反转
arr.copy()                        # 浅拷贝
```

## 5.3 字典操作


## 总结

### 函数速查表

| 类别 | 函数 | 用途 |
|------|------|------|
| 数值 | `abs`, `max`, `min`, `sum`, `pow`, `divmod` | 数值计算 |
| 转换 | `int`, `float`, `str`, `bool`, `bin`, `hex`, `ord`, `chr` | 类型转换 |
| 遍历 | `enumerate`, `zip`, `reversed`, `sorted`, `range` | 序列遍历 |
| 函数式 | `map`, `filter`, `reduce`, `all`, `any` | 函数式编程 |
| 创建 | `list`, `tuple`, `set`, `dict`, `frozenset` | 数据结构 |
| 对象 | `len`, `type`, `isinstance`, `hash`, `id` | 对象操作 |

### 关键记忆

1. `enumerate(arr, start=1)` 从1开始编号
2. `zip(*matrix)` 矩阵转置
3. `sorted(arr, key=lambda x: (-x[0], x[1]))` 多关键字排序
4. `all()/any()` 配合生成器表达式
5. `input = sys.stdin.readline` 快速输入
