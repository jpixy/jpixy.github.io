# 二、O(n²)排序算法

## 2.1 冒泡排序

```python
def bubble_sort(arr: list) -> list:
    """
    冒泡排序
    
    原理: 相邻元素两两比较，大的向后移动
    
    时间复杂度: O(n²)
    空间复杂度: O(1)
    稳定性: 稳定
    """
    arr = arr.copy()
    n = len(arr)
    
    for i in range(n - 1):
        swapped = False
        for j in range(n - 1 - i):
            if arr[j] > arr[j + 1]:
                arr[j], arr[j + 1] = arr[j + 1], arr[j]
                swapped = True
        
        # 优化：如果没有交换，说明已经有序
        if not swapped:
            break
    
    return arr


def bubble_sort_bidirectional(arr: list) -> list:
    """
    双向冒泡排序（鸡尾酒排序）
    
    先从左到右，再从右到左
    """
    arr = arr.copy()
    left, right = 0, len(arr) - 1
    
    while left < right:
        # 从左到右
        for i in range(left, right):
            if arr[i] > arr[i + 1]:
                arr[i], arr[i + 1] = arr[i + 1], arr[i]
        right -= 1
        
        # 从右到左
        for i in range(right, left, -1):
            if arr[i] < arr[i - 1]:
                arr[i], arr[i - 1] = arr[i - 1], arr[i]
        left += 1
    
    return arr
```

## 2.2 选择排序

```python
def selection_sort(arr: list) -> list:
    """
    选择排序
    
    原理: 每次选择最小的元素放到已排序序列末尾
    
    时间复杂度: O(n²)
    空间复杂度: O(1)
    稳定性: 不稳定（交换会破坏稳定性）
    """
    arr = arr.copy()
    n = len(arr)
    
    for i in range(n - 1):
        min_idx = i
        for j in range(i + 1, n):
            if arr[j] < arr[min_idx]:
                min_idx = j
        
        if min_idx != i:
            arr[i], arr[min_idx] = arr[min_idx], arr[i]
    
    return arr
```

## 2.3 插入排序

```python
def insertion_sort(arr: list) -> list:
    """
    插入排序
    
    原理: 将每个元素插入到已排序序列的正确位置
    
    时间复杂度: O(n²)，近乎有序时O(n)
    空间复杂度: O(1)
    稳定性: 稳定
    
    适用场景: 小规模数据、基本有序数据
    """
    arr = arr.copy()
    n = len(arr)
    
    for i in range(1, n):
        key = arr[i]
        j = i - 1
        
        while j >= 0 and arr[j] > key:
            arr[j + 1] = arr[j]
            j -= 1
        
        arr[j + 1] = key
    
    return arr


def binary_insertion_sort(arr: list) -> list:
    """
    二分插入排序
    
    使用二分查找找插入位置
    减少比较次数，但移动次数不变
    """
    import bisect
    
    arr = arr.copy()
    
    for i in range(1, len(arr)):
        key = arr[i]
        pos = bisect.bisect_left(arr, key, 0, i)
        arr[pos + 1:i + 1] = arr[pos:i]
        arr[pos] = key
    
    return arr
```

## 2.4 希尔排序

```python
def shell_sort(arr: list) -> list:
    """
    希尔排序（缩小增量排序）
    
    原理: 分组插入排序，逐步减小增量
    
    时间复杂度: O(n^1.3) ~ O(n²)，取决于增量序列
    空间复杂度: O(1)
    稳定性: 不稳定
    """
    arr = arr.copy()
    n = len(arr)
    gap = n // 2
    
    while gap > 0:
        for i in range(gap, n):
            key = arr[i]
            j = i
            
            while j >= gap and arr[j - gap] > key:
                arr[j] = arr[j - gap]
                j -= gap
            
            arr[j] = key
        
        gap //= 2
    
    return arr


def shell_sort_hibbard(arr: list) -> list:
    """
    使用Hibbard增量序列: 1, 3, 7, 15, ...（2^k - 1）
    
    时间复杂度: O(n^1.5)
    """
    arr = arr.copy()
    n = len(arr)
    
    # 计算增量序列
    gaps = []
    k = 1
    while (1 << k) - 1 < n:
        gaps.append((1 << k) - 1)
        k += 1
    
    for gap in reversed(gaps):
        for i in range(gap, n):
            key = arr[i]
            j = i
            while j >= gap and arr[j - gap] > key:
                arr[j] = arr[j - gap]
                j -= gap
            arr[j] = key
    
    return arr
```

# 三、O(n log n)排序算法

## 3.1 归并排序

```python
def merge_sort(arr: list) -> list:
    """
    归并排序（递归）
    
    原理: 分治，先递归排序两半，再合并
    
    时间复杂度: O(n log n)
    空间复杂度: O(n)
    稳定性: 稳定
    
    适用场景: 链表排序、外部排序
    """
    if len(arr) <= 1:
        return arr
    
    mid = len(arr) // 2
    left = merge_sort(arr[:mid])
    right = merge_sort(arr[mid:])
    
    return merge(left, right)


def merge(left: list, right: list) -> list:
    """合并两个有序数组"""
    result = []
    i = j = 0
    
    while i < len(left) and j < len(right):
        if left[i] <= right[j]:
            result.append(left[i])
            i += 1
        else:
            result.append(right[j])
            j += 1
    
    result.extend(left[i:])
    result.extend(right[j:])
    
    return result


def merge_sort_inplace(arr: list) -> list:
    """
    归并排序（原地，减少空间复制）
    """
    arr = arr.copy()
    
    def sort(left, right):
        if left >= right:
            return
        
        mid = (left + right) // 2
        sort(left, mid)
        sort(mid + 1, right)
        merge_inplace(left, mid, right)
    
    def merge_inplace(left, mid, right):
        temp = arr[left:right + 1]
        i, j = 0, mid - left + 1
        k = left
        
        while i <= mid - left and j <= right - left:
            if temp[i] <= temp[j]:
                arr[k] = temp[i]
                i += 1
            else:
                arr[k] = temp[j]
                j += 1
            k += 1
        
        while i <= mid - left:
            arr[k] = temp[i]
            i += 1
            k += 1
        
        while j <= right - left:
            arr[k] = temp[j]
            j += 1
            k += 1
    
    sort(0, len(arr) - 1)
    return arr


def merge_sort_iterative(arr: list) -> list:
    """
    归并排序（迭代/自底向上）
    
    避免递归栈空间
    """
    arr = arr.copy()
    n = len(arr)
    
    size = 1
    while size < n:
        left = 0
        while left < n:
            mid = min(left + size - 1, n - 1)
            right = min(left + 2 * size - 1, n - 1)
            
            if mid < right:
                # 合并 [left, mid] 和 [mid+1, right]
                temp = []
                i, j = left, mid + 1
                
                while i <= mid and j <= right:
                    if arr[i] <= arr[j]:
                        temp.append(arr[i])
                        i += 1
                    else:
                        temp.append(arr[j])
                        j += 1
                
                while i <= mid:
                    temp.append(arr[i])
                    i += 1
                while j <= right:
                    temp.append(arr[j])
                    j += 1
                
                arr[left:right + 1] = temp
            
            left += 2 * size
        
        size *= 2
    
    return arr
```

## 3.2 快速排序

```python
import random

def quick_sort(arr: list) -> list:
    """
    快速排序（简洁版）
    
    时间复杂度: 平均O(n log n)，最坏O(n²)
    空间复杂度: O(log n) ~ O(n)
    稳定性: 不稳定
    """
    if len(arr) <= 1:
        return arr
    
    pivot = random.choice(arr)
    left = [x for x in arr if x < pivot]
    mid = [x for x in arr if x == pivot]
    right = [x for x in arr if x > pivot]
    
    return quick_sort(left) + mid + quick_sort(right)


def quick_sort_inplace(arr: list) -> list:
    """
    快速排序（原地版）
    """
    arr = arr.copy()
    
    def sort(left, right):
        if left >= right:
            return
        
        pivot_idx = partition(left, right)
        sort(left, pivot_idx - 1)
        sort(pivot_idx + 1, right)
    
    def partition(left, right):
        # 随机选择pivot
        rand_idx = random.randint(left, right)
        arr[rand_idx], arr[right] = arr[right], arr[rand_idx]
        
        pivot = arr[right]
        i = left
        
        for j in range(left, right):
            if arr[j] < pivot:
                arr[i], arr[j] = arr[j], arr[i]
                i += 1
        
        arr[i], arr[right] = arr[right], arr[i]
        return i
    
    sort(0, len(arr) - 1)
    return arr


def quick_sort_3way(arr: list) -> list:
    """
    三路快速排序
    
    适合有大量重复元素的情况
    """
    arr = arr.copy()
    
    def sort(left, right):
        if left >= right:
            return
        
        # 随机选择pivot
        rand_idx = random.randint(left, right)
        arr[left], arr[rand_idx] = arr[rand_idx], arr[left]
        pivot = arr[left]
        
        lt = left      # arr[left+1..lt] < pivot
        gt = right     # arr[gt..right] > pivot
        i = left + 1   # arr[lt+1..i) == pivot
        
        while i <= gt:
            if arr[i] < pivot:
                arr[lt + 1], arr[i] = arr[i], arr[lt + 1]
                lt += 1
                i += 1
            elif arr[i] > pivot:
                arr[gt], arr[i] = arr[i], arr[gt]
                gt -= 1
            else:
                i += 1
        
        arr[left], arr[lt] = arr[lt], arr[left]
        
        sort(left, lt - 1)
        sort(gt + 1, right)
    
    sort(0, len(arr) - 1)
    return arr
```

## 3.3 堆排序

```python
def heap_sort(arr: list) -> list:
    """
    堆排序
    
    原理: 构建最大堆，依次取出堆顶
    
    时间复杂度: O(n log n)
    空间复杂度: O(1)
    稳定性: 不稳定
    """
    arr = arr.copy()
    n = len(arr)
    
    def heapify(n, i):
        """调整以i为根的子树为最大堆"""
        largest = i
        left = 2 * i + 1
        right = 2 * i + 2
        
        if left < n and arr[left] > arr[largest]:
            largest = left
        if right < n and arr[right] > arr[largest]:
            largest = right
        
        if largest != i:
            arr[i], arr[largest] = arr[largest], arr[i]
            heapify(n, largest)
    
    # 构建最大堆
    for i in range(n // 2 - 1, -1, -1):
        heapify(n, i)
    
    # 依次取出堆顶
    for i in range(n - 1, 0, -1):
        arr[0], arr[i] = arr[i], arr[0]
        heapify(i, 0)
    
    return arr


def heap_sort_pythonic(arr: list) -> list:
    """
    使用heapq模块
    
    注意: heapq是最小堆
    """
    import heapq
    
    arr = arr.copy()
    heapq.heapify(arr)
    return [heapq.heappop(arr) for _ in range(len(arr))]
```

# 四、非比较排序

## 4.1 计数排序

```python
def counting_sort(arr: list) -> list:
    """
    计数排序
    
    原理: 统计每个值出现的次数
    
    时间复杂度: O(n + k)，k是值域大小
    空间复杂度: O(k)
    稳定性: 稳定
    
    适用场景: 值域较小的整数
    """
    if not arr:
        return []
    
    min_val, max_val = min(arr), max(arr)
    range_val = max_val - min_val + 1
    
    # 计数
    count = [0] * range_val
    for num in arr:
        count[num - min_val] += 1
    
    # 累积计数（用于稳定排序）
    for i in range(1, range_val):
        count[i] += count[i - 1]
    
    # 输出结果
    result = [0] * len(arr)
    for num in reversed(arr):  # 逆序保持稳定性
        count[num - min_val] -= 1
        result[count[num - min_val]] = num
    
    return result


def counting_sort_simple(arr: list) -> list:
    """
    计数排序（简化版，不稳定）
    """
    if not arr:
        return []
    
    min_val, max_val = min(arr), max(arr)
    count = [0] * (max_val - min_val + 1)
    
    for num in arr:
        count[num - min_val] += 1
    
    result = []
    for i, c in enumerate(count):
        result.extend([i + min_val] * c)
    
    return result
```

## 4.2 桶排序

```python
def bucket_sort(arr: list, bucket_size: int = 5) -> list:
    """
    桶排序
    
    原理: 将元素分到多个桶中，桶内排序后合并
    
    时间复杂度: O(n + k)，k是桶数
    空间复杂度: O(n + k)
    稳定性: 取决于桶内排序算法
    
    适用场景: 均匀分布的数据
    """
    if not arr:
        return []
    
    min_val, max_val = min(arr), max(arr)
    bucket_count = (max_val - min_val) // bucket_size + 1
    
    # 分桶
    buckets = [[] for _ in range(bucket_count)]
    for num in arr:
        bucket_idx = (num - min_val) // bucket_size
        buckets[bucket_idx].append(num)
    
    # 桶内排序并合并
    result = []
    for bucket in buckets:
        bucket.sort()  # 可以用其他排序算法
        result.extend(bucket)
    
    return result


def bucket_sort_float(arr: list, bucket_count: int = 10) -> list:
    """
    桶排序（浮点数，范围[0, 1)）
    """
    if not arr:
        return []
    
    buckets = [[] for _ in range(bucket_count)]
    
    for num in arr:
        bucket_idx = int(num * bucket_count)
        if bucket_idx == bucket_count:
            bucket_idx -= 1
        buckets[bucket_idx].append(num)
    
    result = []
    for bucket in buckets:
        bucket.sort()
        result.extend(bucket)
    
    return result
```

## 4.3 基数排序

```python
def radix_sort(arr: list) -> list:
    """
    基数排序（LSD，从低位到高位）
    
    原理: 按位数排序，每位使用稳定排序
    
    时间复杂度: O(d × (n + k))，d是位数，k是基数
    空间复杂度: O(n + k)
    稳定性: 稳定
    
    适用场景: 整数、定长字符串
    """
    if not arr:
        return []
    
    # 处理负数
    min_val = min(arr)
    if min_val < 0:
        arr = [x - min_val for x in arr]
    
    max_val = max(arr)
    exp = 1  # 当前位
    
    while max_val // exp > 0:
        arr = counting_sort_by_digit(arr, exp)
        exp *= 10
    
    if min_val < 0:
        arr = [x + min_val for x in arr]
    
    return arr


def counting_sort_by_digit(arr: list, exp: int) -> list:
    """按某一位进行计数排序"""
    n = len(arr)
    output = [0] * n
    count = [0] * 10
    
    # 计数
    for num in arr:
        digit = (num // exp) % 10
        count[digit] += 1
    
    # 累积
    for i in range(1, 10):
        count[i] += count[i - 1]
    
    # 输出（逆序保持稳定性）
    for num in reversed(arr):
        digit = (num // exp) % 10
        count[digit] -= 1
        output[count[digit]] = num
    
    return output


def radix_sort_string(arr: list) -> list:
    """
    基数排序（字符串，MSD从高位到低位）
    """
    if not arr:
        return []
    
    max_len = max(len(s) for s in arr)
    
    for i in range(max_len - 1, -1, -1):
        # 按第i位字符排序
        buckets = [[] for _ in range(27)]  # 26字母 + 空字符
        
        for s in arr:
            if i < len(s):
                idx = ord(s[i]) - ord('a') + 1
            else:
                idx = 0
            buckets[idx].append(s)
        
        arr = []
        for bucket in buckets:
            arr.extend(bucket)
    
    return arr
```

## 5.2 topK问题

```python
import heapq

def top_k_smallest(arr: list, k: int) -> list:
    """
    找最小的k个数
    
    方法1: 堆 O(n log k)
    """
    if k >= len(arr):
        return sorted(arr)
    
    # 维护大小为k的最大堆
    heap = [-x for x in arr[:k]]
    heapq.heapify(heap)
    
    for x in arr[k:]:
        if -x > heap[0]:
            heapq.heapreplace(heap, -x)
    
    return sorted([-x for x in heap])


def top_k_largest(arr: list, k: int) -> list:
    """
    找最大的k个数
    
    使用nlargest
    """
    return heapq.nlargest(k, arr)


def quick_select(arr: list, k: int) -> int:
    """
    找第k小的元素
    
    时间复杂度: 平均O(n)
    """
    arr = arr.copy()
    
    def partition(left, right):
        pivot_idx = random.randint(left, right)
        arr[pivot_idx], arr[right] = arr[right], arr[pivot_idx]
        pivot = arr[right]
        i = left
        
        for j in range(left, right):
            if arr[j] < pivot:
                arr[i], arr[j] = arr[j], arr[i]
                i += 1
        
        arr[i], arr[right] = arr[right], arr[i]
        return i
    
    left, right = 0, len(arr) - 1
    while left <= right:
        pivot_idx = partition(left, right)
        if pivot_idx == k - 1:
            return arr[pivot_idx]
        elif pivot_idx < k - 1:
            left = pivot_idx + 1
        else:
            right = pivot_idx - 1
    
    return -1
```

## 总结

### 排序算法选择

| 场景 | 推荐算法 |
|------|----------|
| 通用排序 | 快速排序、归并排序 |
| 小规模数据 | 插入排序 |
| 基本有序 | 插入排序 |
| 链表排序 | 归并排序 |
| 稳定排序 | 归并排序、插入排序 |
| 内存受限 | 堆排序 |
| 整数排序 | 计数排序、基数排序 |
| 外部排序 | 归并排序 |

### 关键记忆

1. **快排**: 随机pivot避免最坏情况
2. **归并**: 稳定，适合链表
3. **堆排**: O(1)空间，不稳定
4. **计数**: 值域小的整数
5. **Python**: list.sort()是TimSort
