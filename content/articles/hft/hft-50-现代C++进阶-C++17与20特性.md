### 1.2 if constexpr

编译期条件分支，避免 SFINAE 复杂性：

```cpp
template<typename T>
auto process(T value) {
    if constexpr (std::is_integral_v<T>) {
        // 只对整数类型编译
        return value * 2;
    } else if constexpr (std::is_floating_point_v<T>) {
        // 只对浮点类型编译
        return value * 2.5;
    } else if constexpr (std::is_same_v<T, std::string>) {
        // 只对 string 编译
        return value + value;
    } else {
        static_assert(always_false_v<T>, "Unsupported type");
    }
}

// HFT 应用：消息处理
template<typename Msg>
void handle_message(const Msg& msg) {
    if constexpr (std::is_same_v<Msg, QuoteMessage>) {
        update_quote(msg);
    } else if constexpr (std::is_same_v<Msg, TradeMessage>) {
        process_trade(msg);
    } else if constexpr (std::is_same_v<Msg, OrderMessage>) {
        process_order(msg);
    }
}

// 编译期分发 vs 运行时分发
template<typename... Handlers>
class MessageDispatcher {
public:
    template<typename Msg>
    void dispatch(const Msg& msg) {
        if constexpr ((std::is_invocable_v<Handlers, Msg> || ...)) {
            // 找到可以处理此消息的 handler
            dispatch_impl<Msg, Handlers...>(msg);
        }
    }
    
private:
    template<typename Msg, typename H, typename... Rest>
    void dispatch_impl(const Msg& msg) {
        if constexpr (std::is_invocable_v<H, Msg>) {
            std::get<H>(handlers_)(msg);
        } else if constexpr (sizeof...(Rest) > 0) {
            dispatch_impl<Msg, Rest...>(msg);
        }
    }
    
    std::tuple<Handlers...> handlers_;
};
```

```cpp
auto result = get_price("AAPL")
    .and_then(apply_spread)
    .and_then(create_order);

// C++17 手动实现
template<typename T, typename F>
auto and_then(std::optional<T> opt, F&& f) 
    -> decltype(f(*opt)) {
    if (opt) {
        return f(*opt);
    }
    return std::nullopt;
}
```

### 1.4 std::variant

```cpp
#include <variant>

// 类型安全的 union
using Message = std::variant<QuoteMsg, TradeMsg, OrderMsg, CancelMsg>;

// 访问
void process(const Message& msg) {
    std::visit([](auto&& arg) {
        using T = std::decay_t<decltype(arg)>;
        if constexpr (std::is_same_v<T, QuoteMsg>) {
            handle_quote(arg);
        } else if constexpr (std::is_same_v<T, TradeMsg>) {
            handle_trade(arg);
        }
    }, msg);
}

// 重载模式
template<class... Ts> struct overloaded : Ts... { using Ts::operator()...; };
template<class... Ts> overloaded(Ts...) -> overloaded<Ts...>;

void process_v2(const Message& msg) {
    std::visit(overloaded{
        [](const QuoteMsg& q) { handle_quote(q); },
        [](const TradeMsg& t) { handle_trade(t); },
        [](const OrderMsg& o) { handle_order(o); },
        [](const CancelMsg& c) { handle_cancel(c); }
    }, msg);
}

// 性能：variant 通常比虚函数快
// 因为 visit 可以被优化为跳转表
```

```cpp
    return x * x;

    requires Numeric<T> && (sizeof(T) >= 4)
```

### 2.3 协程（Coroutines）

```cpp
#include <coroutine>
#include <optional>

// Generator 协程
template<typename T>
class Generator {
public:
    struct promise_type {
        T current_value;
        
        Generator get_return_object() {
            return Generator{Handle::from_promise(*this)};
        }
        
        std::suspend_always initial_suspend() { return {}; }
        std::suspend_always final_suspend() noexcept { return {}; }
        std::suspend_always yield_value(T value) {
            current_value = value;
            return {};
        }
        void return_void() {}
        void unhandled_exception() { throw; }
    };
    
    using Handle = std::coroutine_handle<promise_type>;
    
    Generator(Handle h) : handle_(h) {}
    ~Generator() { if (handle_) handle_.destroy(); }
    
    // 移动语义
    Generator(Generator&& other) : handle_(other.handle_) {
        other.handle_ = nullptr;
    }
    
    bool next() {
        handle_.resume();
        return !handle_.done();
    }
    
    T value() const { return handle_.promise().current_value; }
    
private:
    Handle handle_;
};

// 使用 generator
Generator<int> range(int start, int end) {
    for (int i = start; i < end; i++) {
        co_yield i;
    }
}

void use_generator() {
    auto gen = range(1, 5);
    while (gen.next()) {
        std::cout << gen.value() << " ";  // 1 2 3 4
    }
}

// HFT 应用：异步消息处理
template<typename T>
class Task {
public:
    struct promise_type {
        std::optional<T> result;
        std::exception_ptr exception;
        
        Task get_return_object() {
            return Task{Handle::from_promise(*this)};
        }
        
        std::suspend_never initial_suspend() { return {}; }
        std::suspend_always final_suspend() noexcept { return {}; }
        
        void return_value(T value) { result = std::move(value); }
        void unhandled_exception() { exception = std::current_exception(); }
    };
    
    using Handle = std::coroutine_handle<promise_type>;
    
    T get() {
        if (handle_.promise().exception) {
            std::rethrow_exception(handle_.promise().exception);
        }
        return *handle_.promise().result;
    }
    
private:
    Handle handle_;
};

// 异步订单处理
Task<OrderResult> submit_order_async(Order order) {
    // 验证订单
    auto validation = co_await validate_async(order);
    if (!validation.success) {
        co_return OrderResult{false, 0, validation.error};
    }
    
    // 风控检查
    auto risk = co_await risk_check_async(order);
    if (!risk.approved) {
        co_return OrderResult{false, 0, "Risk limit exceeded"};
    }
    
    // 提交到交易所
    auto result = co_await send_to_exchange_async(order);
    co_return result;
}
```

### 2.5 三向比较（Spaceship Operator）

```cpp
#include <compare>

struct Order {
    uint64_t id;
    double price;
    int64_t timestamp;
    
    // 自动生成所有比较运算符
    auto operator<=>(const Order&) const = default;
};

// 自定义比较
struct PriceLevel {
    double price;
    int quantity;
    
    std::strong_ordering operator<=>(const PriceLevel& other) const {
        // 先比较价格
        if (auto cmp = price <=> other.price; cmp != 0) {
            return cmp;
        }
        // 再比较数量
        return quantity <=> other.quantity;
    }
    
    bool operator==(const PriceLevel&) const = default;
};

// 比较类别
// std::strong_ordering  - 完全有序（相等意味着相同）
// std::weak_ordering    - 弱序（等价不意味着相同）
// std::partial_ordering - 部分序（可能不可比较，如 NaN）

// 使用
PriceLevel a{100.0, 50};
PriceLevel b{100.0, 100};

if (a < b) { /* ... */ }
if (a > b) { /* ... */ }
if (a == b) { /* ... */ }
// 所有比较运算符都可用
```

### 2.6 constexpr 增强

```cpp
// C++20: constexpr 虚函数
struct Base {
    constexpr virtual int value() const { return 1; }
};

struct Derived : Base {
    constexpr int value() const override { return 2; }
};

consteval int get_value() {  // 必须编译期求值
    Derived d;
    Base& b = d;
    return b.value();  // 多态，编译期！
}

// constexpr 动态内存
constexpr int calculate() {
    std::vector<int> v = {1, 2, 3, 4, 5};  // C++20 允许
    return std::accumulate(v.begin(), v.end(), 0);
}

static_assert(calculate() == 15);

// constexpr std::string
constexpr std::string make_greeting() {
    std::string s = "Hello, ";
    s += "World!";
    return s;
}

// constinit - 编译期初始化，但不是 const
constinit int global_value = calculate();  // 编译期初始化
// 运行时可以修改 global_value

// HFT 应用：编译期查找表
constexpr std::array<int, 256> make_char_map() {
    std::array<int, 256> map{};
    for (int i = 0; i < 256; i++) {
        if (i >= '0' && i <= '9') map[i] = i - '0';
        else map[i] = -1;
    }
    return map;
}

constexpr auto CHAR_MAP = make_char_map();

int parse_digit(char c) {
    return CHAR_MAP[static_cast<unsigned char>(c)];
```

### 3.2 小对象优化（SSO）

```cpp
// std::string 的 SSO
// 小字符串（通常 < 16 字节）存储在栈上

// 自定义 SSO
template<typename T, size_t N = 16>
class SmallVector {
public:
    void push_back(const T& value) {
        if (size_ < N) {
            new (&stack_storage_[size_]) T(value);
        } else {
            if (size_ == N) {
                // 首次溢出，迁移到堆
                heap_storage_ = new T[N * 2];
                for (size_t i = 0; i < N; i++) {
                    new (&heap_storage_[i]) T(std::move(stack_storage_[i]));
                    stack_storage_[i].~T();
                }
                capacity_ = N * 2;
            }
            // ... 堆存储逻辑
        }
        size_++;
    }
    
private:
    alignas(T) std::byte stack_storage_[sizeof(T) * N];
    T* heap_storage_ = nullptr;
    size_t size_ = 0;
    size_t capacity_ = N;
};
```

## 四、面试常见问题

**Q: if constexpr 和普通 if 的区别？**

| 特性 | if constexpr | if |
|------|--------------|-----|
| 求值时机 | 编译期 | 运行时 |
| 分支实例化 | 只实例化选中分支 | 所有分支都实例化 |
| 条件要求 | 常量表达式 | 任意表达式 |
| 典型应用 | 模板特化 | 普通条件判断 |

**Q: std::variant 和 union 的区别？**

| 特性 | std::variant | union |
|------|--------------|-------|
| 类型安全 | 是 | 否 |
| 活跃成员追踪 | 自动 | 手动 |
| 非平凡类型 | 支持 | C++11 受限 |
| 异常安全 | 是 | 需手动处理 |

**Q: Concepts 的优势？**

1. 更清晰的错误信息
2. 约束文档化
3. 重载决议更直观
4. 替代复杂的 SFINAE
