## io包

### Reader和Writer

```go
// io.Reader接口
type Reader interface {
    Read(p []byte) (n int, err error)
}

// io.Writer接口
type Writer interface {
    Write(p []byte) (n int, err error)
}
```

### 常用函数

```go
// 复制
io.Copy(dst, src)
io.CopyN(dst, src, n)
io.CopyBuffer(dst, src, buf)

// 读取全部
data, err := io.ReadAll(reader)

// 限制读取
limited := io.LimitReader(reader, 1024)

// 组合Reader
multi := io.MultiReader(r1, r2, r3)

// 同时写多个Writer
multi := io.MultiWriter(w1, w2, w3)

// 管道
pr, pw := io.Pipe()
```

### bufio缓冲IO

```go
// 缓冲读取
reader := bufio.NewReader(file)
line, err := reader.ReadString('\n')
bytes, err := reader.ReadBytes('\n')

// Scanner
scanner := bufio.NewScanner(file)
for scanner.Scan() {
    line := scanner.Text()
    fmt.Println(line)
}

// 缓冲写入
writer := bufio.NewWriter(file)
writer.WriteString("hello")
writer.Flush()  // 刷新缓冲
```

## net/http包

### HTTP客户端

```go
// 简单GET
resp, err := http.Get("https://api.example.com/data")
if err != nil {
    return err
}
defer resp.Body.Close()
body, _ := io.ReadAll(resp.Body)

// 简单POST
resp, err := http.Post(url, "application/json", bytes.NewReader(data))

// 自定义请求
req, err := http.NewRequest("GET", url, nil)
req.Header.Set("Authorization", "Bearer token")
req.Header.Set("Content-Type", "application/json")

client := &http.Client{
    Timeout: 10 * time.Second,
}
resp, err := client.Do(req)
```

### HTTP服务器

```go
// 简单服务器
http.HandleFunc("/", func(w http.ResponseWriter, r *http.Request) {
    fmt.Fprintf(w, "Hello, World!")
})
http.ListenAndServe(":8080", nil)

// 自定义Handler
type MyHandler struct{}

func (h *MyHandler) ServeHTTP(w http.ResponseWriter, r *http.Request) {
    w.WriteHeader(http.StatusOK)
    w.Write([]byte("Hello"))
}

// 路由器
mux := http.NewServeMux()
mux.HandleFunc("/api/users", usersHandler)
mux.HandleFunc("/api/posts", postsHandler)
http.ListenAndServe(":8080", mux)

// 自定义Server
server := &http.Server{
    Addr:         ":8080",
    Handler:      mux,
    ReadTimeout:  5 * time.Second,
    WriteTimeout: 10 * time.Second,
}
server.ListenAndServe()
```

### 中间件

```go
func loggingMiddleware(next http.Handler) http.Handler {
    return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
        start := time.Now()
        next.ServeHTTP(w, r)
        log.Printf("%s %s %v", r.Method, r.URL.Path, time.Since(start))
    })
}

// 使用
handler := loggingMiddleware(myHandler)
```

## encoding/json包

### 序列化

```go
type User struct {
    ID       int    `json:"id"`
    Name     string `json:"name"`
    Email    string `json:"email,omitempty"`
    Password string `json:"-"`  // 忽略
}

user := User{ID: 1, Name: "Alice"}

// 序列化
data, err := json.Marshal(user)

// 格式化输出
data, err := json.MarshalIndent(user, "", "  ")

// 写入Writer
encoder := json.NewEncoder(writer)
encoder.Encode(user)
```

### 反序列化

```go
jsonStr := `{"id":1,"name":"Alice"}`

var user User
err := json.Unmarshal([]byte(jsonStr), &user)

// 从Reader读取
decoder := json.NewDecoder(reader)
err := decoder.Decode(&user)
```

### 动态JSON

```go
// map处理
var data map[string]interface{}
json.Unmarshal(jsonBytes, &data)
name := data["name"].(string)

// json.RawMessage延迟解析
type Response struct {
    Type string          `json:"type"`
    Data json.RawMessage `json:"data"`
}
```

## time包

### 时间操作

```go
// 当前时间
now := time.Now()

// 创建时间
t := time.Date(2024, time.January, 1, 12, 0, 0, 0, time.UTC)

// 时间戳
unix := t.Unix()           // 秒
unixMilli := t.UnixMilli() // 毫秒
unixNano := t.UnixNano()   // 纳秒

// 从时间戳创建
t = time.Unix(1704067200, 0)

// 时间运算
t.Add(time.Hour * 24)
t.AddDate(1, 0, 0)  // 加1年
t.Sub(t2)           // 时间差
```

### 格式化

```go
// Go使用特殊的参考时间：2006-01-02 15:04:05
t := time.Now()

// 格式化
s := t.Format("2006-01-02 15:04:05")
s := t.Format(time.RFC3339)

// 解析
t, err := time.Parse("2006-01-02", "2024-01-01")
t, err := time.ParseInLocation("2006-01-02", "2024-01-01", time.Local)
```

### 定时器

```go
// 定时器
timer := time.NewTimer(2 * time.Second)
<-timer.C  // 阻塞直到触发

// 周期性
ticker := time.NewTicker(time.Second)
for t := range ticker.C {
    fmt.Println("Tick at", t)
}
ticker.Stop()

// 延时执行
time.AfterFunc(time.Second, func() {
    fmt.Println("Delayed")
})
```

## os包

### 文件操作

```go
// 创建文件
f, err := os.Create("file.txt")
defer f.Close()

// 打开文件
f, err := os.Open("file.txt")         // 只读
f, err := os.OpenFile("file.txt", os.O_RDWR|os.O_CREATE, 0644)

// 读写
n, err := f.Read(buf)
n, err := f.Write(data)
n, err := f.WriteString("hello")

// 读取全部
data, err := os.ReadFile("file.txt")

// 写入全部
err := os.WriteFile("file.txt", data, 0644)
```

### 目录操作

```go
// 创建目录
os.Mkdir("dir", 0755)
os.MkdirAll("path/to/dir", 0755)

// 删除
os.Remove("file.txt")
os.RemoveAll("dir")

// 重命名/移动
os.Rename("old.txt", "new.txt")

// 遍历目录
entries, _ := os.ReadDir(".")
for _, entry := range entries {
    fmt.Println(entry.Name(), entry.IsDir())
}
```

### 环境变量

```go
// 获取
value := os.Getenv("HOME")
value, exists := os.LookupEnv("HOME")

// 设置
os.Setenv("MY_VAR", "value")

// 所有环境变量
for _, env := range os.Environ() {
    fmt.Println(env)
}
```

### 进程

```go
// 执行命令
cmd := exec.Command("ls", "-la")
output, err := cmd.Output()

// 获取输出和错误
cmd := exec.Command("cmd")
cmd.Stdout = os.Stdout
cmd.Stderr = os.Stderr
err := cmd.Run()

// 退出
os.Exit(0)

// 信号处理
sigChan := make(chan os.Signal, 1)
signal.Notify(sigChan, syscall.SIGINT, syscall.SIGTERM)
<-sigChan
```

## strings包

```go
// 查找
strings.Contains(s, "substr")
strings.HasPrefix(s, "prefix")
strings.HasSuffix(s, "suffix")
strings.Index(s, "substr")

// 转换
strings.ToLower(s)
strings.ToUpper(s)
strings.TrimSpace(s)
strings.Trim(s, " \n\t")
strings.TrimPrefix(s, "prefix")
strings.TrimSuffix(s, "suffix")

// 分割和连接
parts := strings.Split(s, ",")
s := strings.Join(parts, ",")

// 替换
strings.Replace(s, "old", "new", -1)
strings.ReplaceAll(s, "old", "new")

// Builder（高效拼接）
var builder strings.Builder
builder.WriteString("hello")
builder.WriteString(" world")
result := builder.String()
```

## path/filepath包

```go
// 路径操作
filepath.Join("a", "b", "c")  // a/b/c
filepath.Dir("/a/b/c")        // /a/b
filepath.Base("/a/b/c")       // c
filepath.Ext("file.txt")      // .txt
filepath.Abs(".")             // 绝对路径

// 遍历
filepath.Walk(".", func(path string, info os.FileInfo, err error) error {
    fmt.Println(path)
    return nil
})

// 匹配
matches, _ := filepath.Glob("*.txt")
```
