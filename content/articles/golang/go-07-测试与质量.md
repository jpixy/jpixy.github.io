## 单元测试

### 基本测试

```go
// calculator.go
func Add(a, b int) int {
    return a + b
}

// calculator_test.go
func TestAdd(t *testing.T) {
    result := Add(2, 3)
    if result != 5 {
        t.Errorf("Add(2, 3) = %d; want 5", result)
    }
}
```

### 运行测试

```bash
go test                    # 运行当前包测试
go test ./...              # 运行所有包测试
go test -v                 # 详细输出
go test -run TestAdd       # 运行匹配的测试
go test -cover             # 覆盖率
go test -coverprofile=c.out # 覆盖率报告
go tool cover -html=c.out  # HTML覆盖率报告
```

### 测试辅助

```go
func TestSomething(t *testing.T) {
    // 跳过

        t.Skip("skipping in short mode")

        // ...
    })

        // 清理资源
    })
```

## 表驱动测试

### 标准模式

```go
func TestAdd(t *testing.T) {
    tests := []struct {
        name     string
        a, b     int
        expected int
    }{
        {"positive", 2, 3, 5},
        {"negative", -1, -1, -2},
        {"zero", 0, 0, 0},
        {"mixed", -1, 1, 0},
    }
    
    for _, tt := range tests {
        t.Run(tt.name, func(t *testing.T) {
            result := Add(tt.a, tt.b)
            if result != tt.expected {
                t.Errorf("Add(%d, %d) = %d; want %d",
                    tt.a, tt.b, result, tt.expected)
            }
        })
    }
}
```

### 使用Map

```go
func TestParseStatus(t *testing.T) {
    tests := map[string]struct {
        input    string
        expected int
        wantErr  bool
    }{
        "valid":   {"200", 200, false},
        "invalid": {"abc", 0, true},
        "empty":   {"", 0, true},
    }
    
    for name, tt := range tests {
        t.Run(name, func(t *testing.T) {
            result, err := ParseStatus(tt.input)
            if (err != nil) != tt.wantErr {
                t.Errorf("unexpected error: %v", err)
            }
            if result != tt.expected {
                t.Errorf("got %d; want %d", result, tt.expected)
            }
        })
    }
}
```

## 测试断言

### testify库

```go
import "github.com/stretchr/testify/assert"
import "github.com/stretchr/testify/require"

func TestUser(t *testing.T) {
    user := GetUser(1)
    
    // assert: 失败继续执行
    assert.NotNil(t, user)
    assert.Equal(t, "Alice", user.Name)
    assert.True(t, user.Active)
    assert.Contains(t, user.Email, "@")
    assert.Len(t, user.Roles, 2)
    
    // require: 失败立即停止
    require.NotNil(t, user)
    require.NoError(t, err)
}

// 套件
type UserTestSuite struct {
    suite.Suite
    db *Database
}

func (s *UserTestSuite) SetupTest() {
    s.db = NewDatabase()
}

func (s *UserTestSuite) TearDownTest() {
    s.db.Close()
}

func (s *UserTestSuite) TestGetUser() {
    user := s.db.GetUser(1)
    s.NotNil(user)
}

func TestUserSuite(t *testing.T) {
    suite.Run(t, new(UserTestSuite))
}
```

## Mock测试

### 接口Mock

```go
// 定义接口
type UserRepository interface {
    GetUser(id int) (*User, error)
    SaveUser(user *User) error
}

// 实现Mock
type MockUserRepository struct {
    users map[int]*User
}

func (m *MockUserRepository) GetUser(id int) (*User, error) {
    if user, ok := m.users[id]; ok {
        return user, nil
    }
    return nil, errors.New("not found")
}

// 测试
func TestUserService(t *testing.T) {
    mock := &MockUserRepository{
        users: map[int]*User{1: {ID: 1, Name: "Alice"}},
    }
    
    service := NewUserService(mock)
    user, err := service.GetUser(1)
    
    assert.NoError(t, err)
    assert.Equal(t, "Alice", user.Name)
}
```

### 使用mockgen

```go
//go:generate mockgen -source=repository.go -destination=mock_repository.go -package=user

// 生成的Mock使用
func TestWithMockgen(t *testing.T) {
    ctrl := gomock.NewController(t)
    defer ctrl.Finish()
    
    mock := NewMockUserRepository(ctrl)
    mock.EXPECT().GetUser(1).Return(&User{Name: "Alice"}, nil)
    
    service := NewUserService(mock)
    user, _ := service.GetUser(1)
    
    assert.Equal(t, "Alice", user.Name)
}
```

### HTTP Mock

```go
func TestHTTPClient(t *testing.T) {
    server := httptest.NewServer(http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
        w.WriteHeader(http.StatusOK)
        w.Write([]byte(`{"name":"Alice"}`))
    }))
    defer server.Close()
    
    client := NewClient(server.URL)
    user, err := client.GetUser(1)
    
    assert.NoError(t, err)
    assert.Equal(t, "Alice", user.Name)
}
```

## 模糊测试（Go 1.18+）

```go
func FuzzParse(f *testing.F) {
    // 种子语料
    f.Add("hello")
    f.Add("world")
    f.Add("")
    
    f.Fuzz(func(t *testing.T, input string) {
        result, err := Parse(input)
        if err != nil {
            return
        }
        
        // 验证不变性
        encoded := Encode(result)
        result2, err := Parse(encoded)
        if err != nil {
            t.Errorf("re-parse failed: %v", err)
        }
        if !reflect.DeepEqual(result, result2) {
            t.Errorf("round-trip failed")
        }
    })
}
```

```bash
go test -fuzz=FuzzParse
go test -fuzz=FuzzParse -fuzztime=30s
```

## 测试最佳实践

### 命名规范

```go
// 函数：Test + 被测函数名
func TestAdd(t *testing.T) {}

// 方法：Test + 类型名 + 方法名
func TestCalculator_Add(t *testing.T) {}

// 场景：描述性名称
t.Run("returns error for empty input", func(t *testing.T) {})
```

### 测试文件组织

```
package/
├── calculator.go
├── calculator_test.go      # 单元测试
├── calculator_bench_test.go # 性能测试
└── testdata/               # 测试数据
    └── input.json
```

### Golden文件

```go
func TestRender(t *testing.T) {
    result := Render(data)
    
    golden := filepath.Join("testdata", t.Name()+".golden")
    
    if *update {
        os.WriteFile(golden, []byte(result), 0644)
    }
    
    expected, _ := os.ReadFile(golden)
    if result != string(expected) {
        t.Errorf("mismatch")
    }
}
```
